     1→'use client';
     2→
     3→import * as React from 'react';
     4→import {
     5→  supabase,
     6→  isSupabaseConfigured,
     7→  ensureAuthenticated,
     8→  getCurrentUserEmail,
     9→  invokeEdgeFunction,
    10→  apiFetch,
    11→  seedEntries,
    12→  seedRules,
    13→  seedLogs,
    14→  calculateConfidence,
    15→} from '@/lib/supabase';
    16→import type {
    17→  ShipmentEntry,
    18→  Exception,
    19→  ExtractedField,
    20→  OperationalRules,
    21→  AuditLog,
    22→  ShipmentStatus,
    23→  ExceptionStatus,
    24→  ReviewerAction,
    25→  ExceptionHistoryEntry,
    26→} from '@/lib/clearport-types';
    27→import {
    28→  type UserRole,
    29→} from '@/lib/services/rbac.service';
    30→
    31→// ============================================================================
    32→// Context Type
    33→// ============================================================================
    34→
    35→export type SupabaseStatus = 'connected' | 'unconfigured' | 'error_tables' | 'loading';
    36→export type EdgeFunctionStatus = 'live' | 'fallback' | 'unknown';
    37→
    38→interface ClearPortContextType {
    39→  entries: ShipmentEntry[];
    40→  selectedEntryId: string;
    41→  selectedEntry: ShipmentEntry | undefined;
    42→  selectedExceptionId: string;
    43→  selectedException: Exception | undefined;
    44→  activeTab: string;
    45→  rules: OperationalRules;
    46→  undoStack: { entryId: string; exceptionId: string; previousState: Exception }[];
    47→  auditLogs: AuditLog[];
    48→  theme: 'dark' | 'light';
    49→  supabaseStatus: SupabaseStatus;
    50→  edgeFunctionStatus: EdgeFunctionStatus;
    51→  currentUser: string;
    52→  currentTime: string;
    53→  // RBAC role for the current user — fetched from /api/organizations on
    54→  // load (reflects the user's role in their active org). Defaults to
    55→  // 'operator' so the no-login UX continues to work before the role is
    56→  // resolved. Components use the can* helpers (canUpload, canResolve,
    57→  // canManageRules, canExport) from rbac.service to gate their UI.
    58→  userRole: UserRole;
    59→  // Organizations the user belongs to (populated from
    60→  // GET /api/organizations). Empty until the first successful fetch.
    61→  userOrgs: Array<{ org_id: string; org_name: string; role: UserRole }>;
    62→  // The currently active org id. Sent as the X-Org-Id header on every
    63→  // API call so the backend resolves the same org context via
    64→  // requireOrgRole(). Null until the first org list load completes.
    65→  currentOrgId: string | null;
    66→
    67→  // API helper — same as apiFetch but injects X-Org-Id header
    68→  apiFetchOrg: <T = any>(path: string, options?: RequestInit & { raw?: boolean }) => Promise<T>;
    69→
    70→  // Actions
    71→  selectEntry: (id: string) => void;
    72→  selectException: (id: string) => void;
    73→  setActiveTab: (tab: string) => void;
    74→  updateException: (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => void;
    75→  undoLastAction: () => void;
    76→  acceptAllHighConfidence: (entryId: string) => void;
    77→  uploadDocuments: (files: File[]) => Promise<{ shipmentId: string; success: boolean; error?: string }>;
    78→  updateRules: (newRules: Partial<OperationalRules>) => void;
    79→  exportToCSV: (entryId: string) => Promise<void>;
    80→  toggleTheme: () => void;
    81→  refreshData: () => Promise<void>;
    82→  // Switch the active org context. Updates currentOrgId + userRole, then
    83→  // reloads all org-scoped data (shipments, rules, logs) via the API.
    84→  switchOrg: (orgId: string) => void;
    85→}
    86→
    87→const ClearPortContext = React.createContext<ClearPortContextType | undefined>(undefined);
    88→
    89→// ============================================================================
    90→// Provider
    91→// ============================================================================
    92→
    93→export const ClearPortProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
    94→  const [entries, setEntries] = React.useState<ShipmentEntry[]>(seedEntries);
    95→  const [selectedEntryId, setSelectedEntryId] = React.useState<string>('SHIP-2026-8802');
    96→  const [selectedExceptionId, setSelectedExceptionId] = React.useState<string>('8802-hts');
    97→  const [activeTab, setActiveTab] = React.useState<string>('exception-desk');
    98→  const [theme, setTheme] = React.useState<'dark' | 'light'>('dark');
    99→  const [rules, setRules] = React.useState<OperationalRules>(seedRules);
   100→  const [undoStack, setUndoStack] = React.useState<{ entryId: string; exceptionId: string; previousState: Exception }[]>([]);
   101→  const [auditLogs, setAuditLogs] = React.useState<AuditLog[]>(seedLogs);
   102→  const [supabaseStatus, setSupabaseStatus] = React.useState<SupabaseStatus>(
   103→    isSupabaseConfigured() ? 'loading' : 'unconfigured'
   104→  );
   105→  const [edgeFunctionStatus, setEdgeFunctionStatus] = React.useState<EdgeFunctionStatus>('unknown');
   106→  const [currentUser, setCurrentUser] = React.useState<string>('Broker');
   107→  const [currentTime, setCurrentTime] = React.useState<string>('');
   108→  // Initial value 'operator' preserves the no-login UX. The real role is
   109→  // fetched from /api/organizations on load (see loadData below).
   110→  const [userRole, setUserRole] = React.useState<UserRole>('operator');
   111→  // Orgs the user belongs to + the currently active org id. Populated by
   112→  // loadData on first run; switchOrg() updates currentOrgId and reloads.
   113→  const [userOrgs, setUserOrgs] = React.useState<Array<{ org_id: string; org_name: string; role: UserRole }>>([]);
   114→  const [currentOrgId, setCurrentOrgId] = React.useState<string | null>(null);
   115→
   116→  // Refs mirror the org state so loadData / apiFetchOrg / switchOrg can read
   117→  // the latest values without being recreated on every state change. This
   118→  // keeps loadData's deps stable ([]) so the initial mount effect only fires
   119→  // once, and switchOrg can explicitly trigger a reload.
   120→  const currentOrgIdRef = React.useRef<string | null>(null);
   121→  currentOrgIdRef.current = currentOrgId;
   122→  const userOrgsRef = React.useRef<typeof userOrgs>([]);
   123→  userOrgsRef.current = userOrgs;
   124→
   125→  // --- apiFetch wrapper that injects the X-Org-Id header for org-scoped routes ---
   126→  // Uses the ref so the callback identity is stable (no dependency on state).
   127→  // Routes that DON'T need an org context (e.g. /api/organizations itself)
   128→  // should call the base `apiFetch` directly.
   129→  const apiFetchOrg = React.useCallback(
   130→    <T = any>(path: string, options: RequestInit & { raw?: boolean } = {}): Promise<T> => {
   131→      const headers: Record<string, string> = {
   132→        ...((options.headers as Record<string, string>) || {}),
   133→      };
   134→      if (currentOrgIdRef.current) {
   135→        headers['X-Org-Id'] = currentOrgIdRef.current;
   136→      }
   137→      return apiFetch<T>(path, { ...options, headers });
   138→    },
   139→    [],
   140→  );
   141→
   142→  // --- Real-time clock ---
   143→  React.useEffect(() => {
   144→    const update = () => {
   145→      const now = new Date();
   146→      const utc = now.toISOString().slice(0, 16).replace('T', ' ') + ' UTC';
   147→      setCurrentTime(utc);
   148→    };
   149→    update();
   150→    const interval = setInterval(update, 1000);
   151→    return () => clearInterval(interval);
   152→  }, []);
   153→
   154→  // --- Load user email ---
   155→  React.useEffect(() => {
   156→    getCurrentUserEmail().then(email => setCurrentUser(email));
   157→  }, []);
   158→
   159→  // --- Persist theme ---
   160→  React.useEffect(() => {
   161→    const saved = typeof window !== 'undefined' ? localStorage.getItem('clearport-theme') : null;
   162→    if (saved === 'light' || saved === 'dark') setTheme(saved);
   163→  }, []);
   164→
   165→  const toggleTheme = React.useCallback(() => {
   166→    setTheme(prev => {
   167→      const next = prev === 'dark' ? 'light' : 'dark';
   168→      if (typeof window !== 'undefined') localStorage.setItem('clearport-theme', next);
   169→      return next;
   170→    });
   171→  }, []);
   172→
   173→  // --- Load data ---
   174→  const selectedEntryIdRef = React.useRef(selectedEntryId);
   175→  selectedEntryIdRef.current = selectedEntryId;
   176→
   177→  const loadData = React.useCallback(async () => {
   178→    if (!isSupabaseConfigured()) {
   179→      setSupabaseStatus('unconfigured');
   180→      setEdgeFunctionStatus('fallback');
   181→      return;
   182→    }
   183→
   184→    setSupabaseStatus('loading');
   185→
   186→    try {
   187→      await ensureAuthenticated();
   188→      const email = await getCurrentUserEmail();
   189→      setCurrentUser(email);
   190→
   191→      // --- Bootstrap org context (only on first run, when currentOrgId is null) ---
   192→      // Subsequent reloads (e.g. after switchOrg) skip this block and reuse
   193→      // the already-selected org. Uses the base apiFetch (no X-Org-Id header)
   194→      // because GET /api/organizations is the bootstrap path that doesn't
   195→      // require an org context.
   196→      if (!currentOrgIdRef.current) {
   197→        try {
   198→          const orgsRes = await apiFetch<{
   199→            organizations: Array<{ org_id: string; org_name: string; role: UserRole }>;
   200→          }>('/api/organizations');
   201→
   202→          const orgs = orgsRes?.organizations ?? [];
   203→          if (orgs.length > 0) {
   204→            setUserOrgs(orgs);
   205→            const first = orgs[0];
   206→            setCurrentOrgId(first.org_id);
   207→            currentOrgIdRef.current = first.org_id;
   208→            setUserRole(first.role);
   209→          } else {
   210→            // No org memberships — fall back to seed data with a warning.
   211→            // The user can still see the demo data; they just can't mutate it.
   212→            console.warn('[ctx] No org memberships found — falling back to seed data. Create or join an organization to enable live mode.');
   213→            setSupabaseStatus('connected');
   214→            setEdgeFunctionStatus('fallback');
   215→            return;
   216→          }
   217→        } catch (orgErr) {
   218→          // 403 (no org membership) or network / schema error — fall back to
   219→          // seed data so the demo UX keeps working.
   220→          console.warn('[ctx] /api/organizations failed, falling back to seed data:', orgErr);
   221→          setSupabaseStatus('connected');
   222→          setEdgeFunctionStatus('fallback');
   223→          return;
   224→        }
   225→      }
   226→
   227→      // --- Fetch shipments for the active org (uses X-Org-Id header via apiFetchOrg) ---
   228→      try {
   229→        const response = await apiFetchOrg<{
   230→          data: ShipmentEntry[];
   231→          pagination: { total: number };
   232→        }>('/api/shipments?page=1&limit=100');
   233→
   234→        if (response?.data && Array.isArray(response.data)) {
   235→          // Only use DB shipments if they have fields/exceptions data (new
   236→          // schema). Old-schema shipments have empty arrays — fall back to
   237→          // seed data so the demo UX is preserved.
   238→          const hasRealData = response.data.some(
   239→            (s) => s.fields.length > 0 || s.exceptions.length > 0,
   240→          );
   241→
   242→          if (hasRealData) {
   243→            setEntries(response.data);
   244→            // Select first shipment if current selection doesn't exist
   245→            if (!response.data.find((s) => s.id === selectedEntryIdRef.current)) {
   246→              const first = response.data[0];
   247→              setSelectedEntryId(first.id);
   248→              if (first.exceptions?.length > 0) {
   249→                const unresolved = first.exceptions.find((e) => e.status === 'Unresolved');
   250→                setSelectedExceptionId(unresolved ? unresolved.id : first.exceptions[0].id);
   251→              }
   252→            }
   253→          }
   254→          // else: keep seed entries (demo mode)
   255→
   256→          setEdgeFunctionStatus('live');
   257→          setSupabaseStatus('connected');
   258→
   259→          // Fetch rules + logs in parallel via the org-scoped API routes.
   260→          const [rulesRes, logsRes] = await Promise.all([
   261→            apiFetchOrg<{ rules: OperationalRules }>('/api/rules').catch(() => null),
   262→            apiFetchOrg<{ logs: AuditLog[] }>('/api/audit-logs').catch(() => null),
   263→          ]);
   264→          if (rulesRes?.rules) setRules(rulesRes.rules);
   265→          if (logsRes?.logs && logsRes.logs.length > 0) setAuditLogs(logsRes.logs);
   266→          return;
   267→        }
   268→      } catch (apiErr) {
   269→        console.warn('[ctx] /api/shipments failed, falling back to seed data:', apiErr);
   270→      }
   271→
   272→      // Fallback: seed data (demo mode)
   273→      setSupabaseStatus('connected');
   274→      setEdgeFunctionStatus('fallback');
   275→    } catch (err) {
   276→      console.error('[ctx] loadData error:', err);
   277→      setSupabaseStatus('error_tables');
   278→      setEdgeFunctionStatus('fallback');
   279→    }
   280→  }, [apiFetchOrg]);
   281→
   282→  // --- switchOrg: change the active org context and reload data ---
   283→  const switchOrg = React.useCallback((orgId: string) => {
   284→    const org = userOrgsRef.current.find((o) => o.org_id === orgId);
   285→    if (!org) {
   286→      console.warn('[ctx] switchOrg: org not found in userOrgs', { orgId });
   287→      return;
   288→    }
   289→    // Update both state + ref so apiFetchOrg picks up the new org immediately
   290→    // (the ref is read synchronously; the state triggers a re-render).
   291→    setCurrentOrgId(orgId);
   292→    currentOrgIdRef.current = orgId;
   293→    setUserRole(org.role);
   294→    // Reload all org-scoped data for the new org.
   295→    loadData();
   296→  }, [loadData]);
   297→
   298→  React.useEffect(() => {
   299→    loadData();
   300→  }, [loadData]);
   301→
   302→  const refreshData = React.useCallback(async () => {
   303→    await loadData();
   304→  }, [loadData]);
   305→
   306→  // --- Computed ---
   307→  const selectedEntry = React.useMemo(
   308→    () => entries.find(e => e.id === selectedEntryId),
   309→    [entries, selectedEntryId]
   310→  );
   311→
   312→  const selectedException = React.useMemo(() => {
   313→    if (!selectedEntry) return undefined;
   314→    return selectedEntry.exceptions.find(ex => ex.id === selectedExceptionId);
   315→  }, [selectedEntry, selectedExceptionId]);
   316→
   317→  // --- Selection (fixes stale closure — uses functional update) ---
   318→  const selectEntry = React.useCallback((id: string) => {
   319→    setSelectedEntryId(id);
   320→    setEntries(prev => {
   321→      const entry = prev.find(e => e.id === id);
   322→      if (entry && entry.exceptions.length > 0) {
   323→        const unresolved = entry.exceptions.find(e => e.status === 'Unresolved');
   324→        setSelectedExceptionId(unresolved ? unresolved.id : entry.exceptions[0].id);
   325→      } else {
   326→        setSelectedExceptionId('');
   327→      }
   328→      return prev;
   329→    });
   330→  }, []);
   331→
   332→  const selectException = React.useCallback((id: string) => {
   333→    setSelectedExceptionId(id);
   334→  }, []);
   335→
   336→  // --- Audit log helper ---
   337→  const addAuditLog = React.useCallback((text: string, type: AuditLog['type'] = 'info', shipmentId?: string) => {
   338→    const newLog: AuditLog = {
   339→      id: typeof crypto !== 'undefined' ? crypto.randomUUID() : `log-${Date.now()}`,
   340→      text,
   341→      timestamp: new Date().toISOString(),
   342→      type,
   343→      shipmentId,
   344→    };
   345→    setAuditLogs(prev => [newLog, ...prev]);
   346→
   347→    if (isSupabaseConfigured() && supabase) {
   348→      supabase.from('audit_logs').insert({
   349→        id: newLog.id,
   350→        text: newLog.text,
   351→        timestamp: newLog.timestamp,
   352→        type: newLog.type,
   353→        shipment_id: shipmentId || null,
   354→      }).then(({ error }) => {
   355→        if (error) console.warn('[ctx] audit log insert failed:', error.message);
   356→      });
   357→    }
   358→  }, []);
   359→
   360→  // --- Update exception (fixes setState-in-setState) ---
   361→  const updateException = React.useCallback(
   362→    (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => {
   363→      let previousException: Exception | undefined;
   364→      let updatedEntry: ShipmentEntry | undefined;
   365→
   366→      setEntries(prevEntries => {
   367→        return prevEntries.map(entry => {
   368→          if (entry.id !== entryId) return entry;
   369→
   370→          const prev = entry.exceptions.find(ex => ex.id === exceptionId);
   371→          if (!prev) return entry;
   372→          previousException = { ...prev };
   373→
   374→          const updatedExceptions = entry.exceptions.map(ex => {
   375→            if (ex.id !== exceptionId) return ex;
   376→
   377→            const oldVal = ex.correctedValue || ex.extractedValue;
   378→            const updatedVal = newValue !== undefined ? newValue : ex.extractedValue;
   379→
   380→            const historyItem: ExceptionHistoryEntry = {
   381→              user: currentUser,
   382→              oldValue: oldVal,
   383→              newValue: updatedVal,
   384→              timestamp: new Date().toISOString(),
   385→              action: status,
   386→            };
   387→
   388→            return {
   389→              ...ex,
   390→              status: status as ExceptionStatus,
   391→              correctedValue: status === 'Corrected' ? newValue : undefined,
   392→              history: [historyItem, ...ex.history],
   393→              resolvedAt: new Date().toISOString(),
   394→              resolvedBy: currentUser,
   395→            };
   396→          });
   397→
   398→          const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   399→          const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   400→          const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   401→
   402→          const updatedFields = entry.fields.map(f => {
   403→            if (f.exceptionId === exceptionId) {
   404→              const val = status === 'Corrected' ? (newValue || f.value) : f.value;
   405→              return { ...f, value: val, isFlagged: false };
   406→            }
   407→            return f;
   408→          });
   409→
   410→          updatedEntry = {
   411→            ...entry,
   412→            status: newStatus,
   413→            exceptions: updatedExceptions,
   414→            fields: updatedFields,
   415→            currentConfidence: newConfidence,
   416→          };
   417→
   418→          return updatedEntry;
   419→        });
   420→      });
   421→
   422→      // Side effects AFTER setState
   423→      if (previousException && updatedEntry) {
   424→        setUndoStack(prevStack => [
   425→          { entryId, exceptionId, previousState: { ...previousException! } },
   426→          ...prevStack,
   427→        ]);
   428→
   429→        const logText = `Exception "${previousException.fieldName}" for ${entryId} was ${status.toLowerCase()}${
   430→          status === 'Corrected' ? ` to "${newValue}"` : ''
   431→        } by ${currentUser}.`;
   432→        addAuditLog(logText, status === 'Rejected' ? 'warning' : 'success', entryId);
   433→
   434→        // Persist via the /api/exceptions/:id route (handles exception update,
   435→        // document_field sync, shipment confidence recompute, and audit log
   436→        // in one server-side transaction). Uses the org-scoped wrapper so
   437→        // the X-Org-Id header is sent.
   438→        if (isSupabaseConfigured()) {
   439→          apiFetchOrg(`/api/exceptions/${exceptionId}`, {
   440→            method: 'PATCH',
   441→            body: JSON.stringify({
   442→              status,
   443→              correctedValue: status === 'Corrected' ? newValue : undefined,
   444→            }),
   445→          }).catch((err) => {
   446→            console.warn('[ctx] exception update failed:', err instanceof Error ? err.message : err);
   447→          });
   448→        }
   449→      }
   450→    },
   451→    [currentUser, addAuditLog, apiFetchOrg]
   452→  );
   453→
   454→  // --- Undo ---
   455→  const undoLastAction = React.useCallback(() => {
   456→    if (undoStack.length === 0) return;
   457→
   458→    const [lastAction, ...remainingStack] = undoStack;
   459→    setUndoStack(remainingStack);
   460→
   461→    setEntries(prevEntries => {
   462→      return prevEntries.map(entry => {
   463→        if (entry.id !== lastAction.entryId) return entry;
   464→
   465→        const restored = lastAction.previousState;
   466→        const updatedExceptions = entry.exceptions.map(ex =>
   467→          ex.id === lastAction.exceptionId ? restored : ex
   468→        );
   469→
   470→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   471→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   472→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   473→
   474→        const updatedFields = entry.fields.map(f => {
   475→          if (f.exceptionId === lastAction.exceptionId) {
   476→            return {
   477→              ...f,
   478→              value: restored.status === 'Corrected' ? (restored.correctedValue || f.value) : restored.extractedValue,
   479→              isFlagged: restored.status === 'Unresolved',
   480→            };
   481→          }
   482→          return f;
   483→        });
   484→
   485→        const updatedEntry = {
   486→          ...entry,
   487→          status: newStatus,
   488→          exceptions: updatedExceptions,
   489→          fields: updatedFields,
   490→          currentConfidence: newConfidence,
   491→        };
   492→
   493→        if (isSupabaseConfigured() && supabase) {
   494→          supabase.from('exceptions').update({
   495→            status: restored.status,
   496→            corrected_value: restored.correctedValue || null,
   497→            resolved_at: restored.resolvedAt || null,
   498→            resolved_by: restored.resolvedBy || null,
   499→            history: restored.history,
   500→          }).eq('id', lastAction.exceptionId).then(({ error }) => {
   501→            if (error) console.warn('[ctx] undo exception failed:', error.message);
   502→          });
   503→
   504→          supabase.from('shipments').update({
   505→            status: newStatus,
   506→            current_confidence: newConfidence,
   507→            updated_at: new Date().toISOString(),
   508→          }).eq('id', entry.id).then(({ error }) => {
   509→            if (error) console.warn('[ctx] undo shipment failed:', error.message);
   510→          });
   511→        }
   512→
   513→        return updatedEntry;
   514→      });
   515→    });
   516→
   517→    addAuditLog(`Undo applied for last review action on shipment ${lastAction.entryId}`, 'info', lastAction.entryId);
   518→  }, [undoStack, addAuditLog]);
   519→
   520→  // --- Batch accept (uses configured threshold) ---
   521→  const acceptAllHighConfidence = React.useCallback((entryId: string) => {
   522→    setEntries(prevEntries => {
   523→      return prevEntries.map(entry => {
   524→        if (entry.id !== entryId) return entry;
   525→
   526→        const cutoff = rules.invoiceThreshold;
   527→        let acceptedCount = 0;
   528→
   529→        const updatedExceptions = entry.exceptions.map(ex => {
   530→          if (ex.status === 'Unresolved' && ex.confidence >= cutoff) {
   531→            acceptedCount++;
   532→            const historyItem: ExceptionHistoryEntry = {
   533→              user: currentUser,
   534→              oldValue: ex.extractedValue,
   535→              newValue: ex.extractedValue,
   536→              timestamp: new Date().toISOString(),
   537→              action: 'Accepted' as ReviewerAction,
   538→            };
   539→            return {
   540→              ...ex,
   541→              status: 'Accepted' as ExceptionStatus,
   542→              history: [historyItem, ...ex.history],
   543→              resolvedAt: new Date().toISOString(),
   544→              resolvedBy: currentUser,
   545→            };
   546→          }
   547→          return ex;
   548→        });
   549→
   550→        if (acceptedCount === 0) return entry;
   551→
   552→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');

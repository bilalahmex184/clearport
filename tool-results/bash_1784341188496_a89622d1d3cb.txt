     1→'use client';
     2→
     3→import * as React from 'react';
     4→import { useClearPort } from '@/context/ClearPortContext';
     5→import {
     6→  AlertCircle, CheckCircle2, ChevronRight, Eye, RefreshCw, XCircle, FileText,
     7→  ZoomIn, ZoomOut, RotateCw, Sparkles, Undo2, Loader2, FileSpreadsheet,
     8→} from 'lucide-react';
     9→import { motion, AnimatePresence } from 'framer-motion';
    10→import { supabase, invokeEdgeFunction } from '@/lib/supabase';
    11→import { canResolve, roleLabel } from '@/lib/services/rbac.service';
    12→
    13→export default function ExceptionDesk() {
    14→  const {
    15→    entries,
    16→    selectedEntry,
    17→    selectedExceptionId,
    18→    selectedException,
    19→    selectException,
    20→    updateException,
    21→    undoLastAction,
    22→    acceptAllHighConfidence,
    23→    rules,
    24→    userRole,
    25→    exportToCSV,
    26→  } = useClearPort();
    27→
    28→  // RBAC: viewer role cannot resolve exceptions. They can still browse the
    29→  // document viewer + exception list (read-only), but the Accept / Modify /
    30→  // Reject buttons are hidden and the keyboard shortcuts that mutate state
    31→  // are disabled below.
    32→  const canResolveExceptions = canResolve(userRole);
    33→
    34→  const [severityFilter, setSeverityFilter] = React.useState<'All' | 'High' | 'Medium' | 'Low'>('All');
    35→  const [sortBy, setSortBy] = React.useState<'severity' | 'confidence' | 'fieldName'>('severity');
    36→  const [isEditing, setIsEditing] = React.useState(false);
    37→  const [editValue, setEditValue] = React.useState('');
    38→  const [activeDocTab, setActiveDocTab] = React.useState('');
    39→  const [zoomLevel, setZoomLevel] = React.useState(100);
    40→  const [rotation, setRotation] = React.useState(0);
    41→  const [showHistory, setShowHistory] = React.useState(false);
    42→  const [documentUrl, setDocumentUrl] = React.useState<string | null>(null);
    43→  const [documentMime, setDocumentMime] = React.useState<string>('');
    44→  const [isLoadingUrl, setIsLoadingUrl] = React.useState(false);
    45→  const inputRef = React.useRef<HTMLInputElement>(null);
    46→
    47→  // Fetch signed URL for the active document when tab changes
    48→  React.useEffect(() => {
    49→    if (!selectedEntry || !activeDocTab) {
    50→      setDocumentUrl(null);
    51→      setDocumentMime('');
    52→      return;
    53→    }
    54→
    55→    // Find the document matching the active tab
    56→    const doc = selectedEntry.documents.find(d => d.docType === activeDocTab);
    57→    if (!doc) {
    58→      setDocumentUrl(null);
    59→      setDocumentMime('');
    60→      return;
    61→    }
    62→
    63→    let cancelled = false;
    64→    setIsLoadingUrl(true);
    65→    setDocumentUrl(null);
    66→    setDocumentMime(doc.mimeType || '');
    67→
    68→    (async () => {
    69→      try {
    70→        // Try the get-document-url edge function first
    71→        const response = await invokeEdgeFunction<any>('get-document-url', {
    72→          storagePath: doc.storagePath,
    73→        });
    74→        if (!cancelled && response?.success && response.signedUrl) {
    75→          setDocumentUrl(response.signedUrl);
    76→        }
    77→      } catch (err) {
    78→        // Fallback: try direct Supabase storage signed URL
    79→        if (supabase && !cancelled) {
    80→          try {
    81→            const { data } = await supabase.storage
    82→              .from('documents')
    83→              .createSignedUrl(doc.storagePath, 3600);
    84→            if (data?.signedUrl) {
    85→              setDocumentUrl(data.signedUrl);
    86→            }
    87→          } catch (e) {
    88→            console.warn('[doc-url] failed to get signed URL:', e);
    89→          }
    90→        }
    91→      } finally {
    92→        if (!cancelled) setIsLoadingUrl(false);
    93→      }
    94→    })();
    95→
    96→    return () => { cancelled = true; };
    97→  }, [selectedEntry?.id, activeDocTab]);
    98→
    99→  // Fix: use useEffect instead of setState-during-render
   100→  React.useEffect(() => {
   101→    if (selectedException) {
   102→      setEditValue(selectedException.correctedValue ?? selectedException.extractedValue);
   103→      setActiveDocTab(selectedException.docType);
   104→      setIsEditing(false);
   105→    }
   106→  }, [selectedException?.id]);
   107→
   108→  // Keyboard shortcuts (scoped — only when not typing in input)
   109→  React.useEffect(() => {
   110→    const handleKeyDown = (e: KeyboardEvent) => {
   111→      const activeEl = document.activeElement;
   112→      if (activeEl?.tagName === 'INPUT' || activeEl?.tagName === 'TEXTAREA') {
   113→        if (e.key === 'Escape') {
   114→          inputRef.current?.blur();
   115→          setIsEditing(false);
   116→        }
   117→        return;
   118→      }
   119→
   120→      if (!selectedException || !selectedEntry) return;
   121→
   122→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'z') {
   123→        // Undo is a resolve-adjacent action — gated by the same RBAC check.
   124→        if (!canResolveExceptions) return;
   125→        e.preventDefault();
   126→        undoLastAction();
   127→        return;
   128→      }
   129→
   130→      // Accept / Modify / Reject shortcuts are no-ops for viewer role.
   131→      if (!canResolveExceptions) return;
   132→
   133→      if (e.key === ' ') {
   134→        e.preventDefault();
   135→        updateException(selectedEntry.id, selectedException.id, 'Accepted');
   136→      }
   137→
   138→      if (e.key.toLowerCase() === 'e') {
   139→        e.preventDefault();
   140→        setIsEditing(true);
   141→        setTimeout(() => inputRef.current?.focus(), 50);
   142→      }
   143→
   144→      if (e.key.toLowerCase() === 'r') {
   145→        e.preventDefault();
   146→        updateException(selectedEntry.id, selectedException.id, 'Rejected');
   147→      }
   148→    };
   149→
   150→    window.addEventListener('keydown', handleKeyDown);
   151→    return () => window.removeEventListener('keydown', handleKeyDown);
   152→  }, [selectedException, selectedEntry, updateException, undoLastAction, canResolveExceptions]);
   153→
   154→  if (!selectedEntry) {
   155→    return (
   156→      <div className="h-full flex flex-col items-center justify-center text-gray-500 p-6">
   157→        <AlertCircle className="w-12 h-12 mb-4 text-gray-600 animate-pulse" />
   158→        <p className="text-sm font-medium">No shipment selected</p>
   159→        <p className="text-xs text-gray-600 mt-1">Please select an active batch from the Dashboard or sidebar.</p>
   160→      </div>
   161→    );
   162→  }
   163→
   164→  const exceptions = selectedEntry.exceptions;
   165→
   166→  // Filter & Sort
   167→  const filteredExceptions = exceptions.filter(ex => {
   168→    if (severityFilter === 'All') return true;
   169→    if (severityFilter === 'High') return ex.confidence < 60;
   170→    if (severityFilter === 'Medium') return ex.confidence >= 60 && ex.confidence < 85;
   171→    if (severityFilter === 'Low') return ex.confidence >= 85;
   172→    return true;
   173→  });
   174→
   175→  const sortedExceptions = [...filteredExceptions].sort((a, b) => {
   176→    if (sortBy === 'severity') return a.confidence - b.confidence;
   177→    if (sortBy === 'confidence') return b.confidence - a.confidence;
   178→    return a.fieldName.localeCompare(b.fieldName);
   179→  });
   180→
   181→  const totalExceptions = exceptions.length;
   182→  const resolvedExceptions = exceptions.filter(ex => ex.status !== 'Unresolved').length;
   183→  const highConfidenceCount = exceptions.filter(ex => ex.status === 'Unresolved' && ex.confidence >= rules.invoiceThreshold).length;
   184→
   185→  const getConfidenceColor = (conf: number) => {
   186→    if (conf < 60) return 'text-red-400 bg-red-950/40 border-red-900/50';
   187→    if (conf < 85) return 'text-amber-400 bg-amber-950/40 border-amber-900/50';
   188→    return 'text-green-400 bg-green-950/40 border-green-900/50';
   189→  };
   190→
   191→  const getConfidenceBadge = (conf: number) => {
   192→    if (conf < 60) return 'RED / CRITICAL';
   193→    if (conf < 85) return 'AMBER / WARNING';
   194→    return 'GREEN / COMPLIANT';
   195→  };
   196→
   197→  const handleSaveEdit = () => {
   198→    if (selectedException && selectedEntry) {
   199→      updateException(selectedEntry.id, selectedException.id, 'Corrected', editValue);
   200→      setIsEditing(false);
   201→    }
   202→  };
   203→
   204→  // Data-driven document tabs (derive from shipment fields, not hardcoded IDs)
   205→  // Derive available doc types from BOTH fields' sourceDoc AND uploaded documents
   206→  const availableDocTypes = Array.from(new Set([
   207→    ...selectedEntry.fields.map(f => f.sourceDoc).filter(Boolean),
   208→    ...selectedEntry.documents.map(d => d.docType).filter(Boolean),
   209→  ]));
   210→
   211→  // Filter fields by the active doc tab (if multiple doc types exist)
   212→  const fieldsForActiveTab = availableDocTypes.length > 1
   213→    ? selectedEntry.fields.filter(f => f.sourceDoc === activeDocTab || !activeDocTab)
   214→    : selectedEntry.fields;
   215→
   216→  return (
   217→    <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 h-full overflow-hidden p-6 font-sans">
   218→      {/* LEFT PANEL: EXCEPTION LIST */}
   219→      <div className="lg:col-span-3 bg-[#0c0d12] border border-gray-900 rounded-xl flex flex-col overflow-hidden">
   220→        <div className="p-4 border-b border-gray-900 bg-[#0e1017]">
   221→          <div className="flex items-center justify-between mb-2">
   222→            <span className="font-mono text-xs text-gray-500 tracking-wider">EXCEPTIONS DESK</span>
   223→            <span className="font-mono text-xs text-emerald-400 bg-emerald-950/30 px-2 py-0.5 rounded border border-emerald-900/40">
   224→              {resolvedExceptions} / {totalExceptions} RESOLVED
   225→            </span>
   226→          </div>
   227→          <h2 className="text-sm font-medium text-gray-200 tracking-tight flex items-center gap-1.5">
   228→            <FileText className="w-4 h-4 text-gray-400" />
   229→            {selectedEntry.id}
   230→          </h2>
   231→          <p className="text-xs text-gray-500 mt-1 truncate">{selectedEntry.shipper}</p>
   232→
   233→          {/* CSV Export Button */}
   234→          <button
   235→            onClick={() => exportToCSV(selectedEntry.id)}
   236→            className="mt-3 w-full flex items-center justify-center gap-1.5 text-[10px] bg-emerald-600 hover:bg-emerald-500 text-black font-bold px-2 py-1.5 rounded-lg transition-all cursor-pointer uppercase tracking-wider"
   237→          >
   238→            <FileSpreadsheet className="w-3 h-3" />
   239→            Export CSV
   240→          </button>
   241→
   242→          {/* (§4) Validation status banner — reads validation_status, not just totalExceptions */}
   243→          {selectedEntry.validationStatus === 'completed' && totalExceptions === 0 && (
   244→            <div className="mt-2 p-2 rounded-lg border text-[10px] font-mono flex items-center gap-1.5 bg-emerald-950/30 border-emerald-900/40 text-emerald-400">
   245→              <CheckCircle2 className="w-3 h-3 shrink-0" />
   246→              <span>VALIDATED — zero exceptions found. Shipment is clean.</span>
   247→            </div>
   248→          )}
   249→          {selectedEntry.validationStatus === 'completed' && totalExceptions > 0 && (
   250→            <div className="mt-2 p-2 rounded-lg border text-[10px] font-mono flex items-center gap-1.5 bg-amber-950/30 border-amber-900/40 text-amber-400">
   251→              <AlertCircle className="w-3 h-3 shrink-0" />
   252→              <span>VALIDATED — {totalExceptions} exceptions require review.</span>
   253→            </div>
   254→          )}
   255→          {(selectedEntry.validationStatus === 'failed' || selectedEntry.validationStatus === 'degraded') && (
   256→            <div className="mt-2 p-2 rounded-lg border text-[10px] font-mono flex items-center gap-1.5 bg-red-950/30 border-red-900/40 text-red-400">
   257→              <AlertCircle className="w-3 h-3 shrink-0" />
   258→              <span>VALIDATION INCOMPLETE — pipeline error, retry required.</span>
   259→            </div>
   260→          )}
   261→          {(selectedEntry.validationStatus === 'pending' || selectedEntry.validationStatus === 'running' || !selectedEntry.validationStatus) && (
   262→            <div className="mt-2 p-2 rounded-lg border text-[10px] font-mono flex items-center gap-1.5 bg-blue-950/30 border-blue-900/40 text-blue-400">
   263→              <Loader2 className="w-3 h-3 shrink-0 animate-spin" />
   264→              <span>VALIDATION IN PROGRESS — not yet validated.</span>
   265→            </div>
   266→          )}
   267→        </div>
   268→
   269→        {/* Filters */}
   270→        <div className="p-3 border-b border-gray-900 bg-[#090a0f] space-y-2.5">
   271→          <div className="flex items-center gap-1">
   272→            {(['All', 'High', 'Medium', 'Low'] as const).map(filter => (
   273→              <button
   274→                key={filter}
   275→                onClick={() => setSeverityFilter(filter)}
   276→                className={`text-[11px] px-2.5 py-1 rounded-md font-medium transition-all ${
   277→                  severityFilter === filter
   278→                    ? 'bg-gray-800 text-white'
   279→                    : 'text-gray-500 hover:text-gray-300 hover:bg-gray-950'
   280→                }`}
   281→              >
   282→                {filter}
   283→              </button>
   284→            ))}
   285→          </div>
   286→
   287→          <div className="flex items-center justify-between gap-2 pt-1">
   288→            <span className="text-[10px] font-mono text-gray-500 uppercase tracking-widest">SORT BY</span>
   289→            <select
   290→              value={sortBy}
   291→              onChange={e => setSortBy(e.target.value as any)}
   292→              className="bg-black border border-gray-800 text-gray-300 rounded px-1.5 py-0.5 text-xs focus:outline-none focus:border-gray-700 font-medium"
   293→            >
   294→              <option value="severity">Severity (Low Conf)</option>
   295→              <option value="confidence">Confidence (High)</option>
   296→              <option value="fieldName">Field Name</option>
   297→            </select>
   298→          </div>
   299→
   300→          {highConfidenceCount > 0 && canResolveExceptions && (
   301→            <button
   302→              onClick={() => acceptAllHighConfidence(selectedEntry.id)}
   303→              className="w-full flex items-center justify-center gap-1.5 text-xs bg-emerald-950/30 text-emerald-400 border border-emerald-900/50 hover:bg-emerald-900/20 py-2 rounded-md transition-all cursor-pointer font-medium mt-1"
   304→            >
   305→              <Sparkles className="w-3.5 h-3.5" />
   306→              Accept {highConfidenceCount} Fields ≥ {rules.invoiceThreshold}% Conf
   307→            </button>
   308→          )}
   309→
   310→          {!canResolveExceptions && (
   311→            <div className="mt-1 flex items-start gap-1.5 text-[10px] text-amber-500/90 bg-amber-950/20 border border-amber-900/40 rounded-md p-2 font-mono leading-relaxed">
   312→              <AlertCircle className="w-3 h-3 shrink-0 mt-0.5" />
   313→              <span>
   314→                READ-ONLY ({roleLabel(userRole)}) — Accept / Modify / Reject are
   315→                disabled. Switch to an operator / admin session to resolve exceptions.
   316→              </span>
   317→            </div>
   318→          )}
   319→        </div>
   320→
   321→        {/* Exceptions list */}
   322→        <div className="flex-1 overflow-y-auto divide-y divide-gray-950 p-2 space-y-1.5">
   323→          <AnimatePresence mode="popLayout">
   324→            {sortedExceptions.map(ex => {
   325→              const isSelected = ex.id === selectedExceptionId;
   326→              const severityClass = getConfidenceColor(ex.confidence);
   327→              const isDone = ex.status !== 'Unresolved';
   328→
   329→              return (
   330→                <motion.div
   331→                  key={ex.id}
   332→                  layout
   333→                  initial={{ opacity: 0, y: 10 }}
   334→                  animate={{ opacity: 1, y: 0 }}
   335→                  exit={{ opacity: 0, scale: 0.95 }}
   336→                  transition={{ duration: 0.15 }}
   337→                  onClick={() => selectException(ex.id)}
   338→                  className={`p-3 rounded-lg cursor-pointer transition-all border ${
   339→                    isSelected
   340→                      ? 'bg-gray-950 border-gray-700 shadow-lg shadow-black/40'
   341→                      : 'bg-transparent border-transparent hover:bg-gray-950/40 hover:border-gray-900'
   342→                  }`}
   343→                >
   344→                  <div className="flex items-start justify-between gap-1.5 mb-1.5">
   345→                    <span className="text-xs font-semibold text-gray-200 tracking-tight leading-tight">
   346→                      {ex.fieldName}
   347→                    </span>
   348→                    {isDone ? (
   349→                      <span className={`text-[9px] font-mono font-bold px-1.5 py-0.5 rounded flex items-center gap-1 shrink-0 ${
   350→                        ex.status === 'Accepted' ? 'bg-emerald-950 text-emerald-400 border border-emerald-900/30' :
   351→                        ex.status === 'Rejected' ? 'bg-red-950 text-red-400 border border-red-900/30' :
   352→                        'bg-blue-950 text-blue-400 border border-blue-900/30'
   353→                      }`}>
   354→                        {ex.status.toUpperCase()}
   355→                      </span>
   356→                    ) : (
   357→                      <span className={`text-[10px] font-mono font-bold px-1.5 py-0.5 rounded border shrink-0 ${severityClass}`}>
   358→                        {ex.confidence}%
   359→                      </span>
   360→                    )}
   361→                  </div>
   362→
   363→                  <p className="text-[11px] text-gray-400 leading-normal line-clamp-2">{ex.reason}</p>
   364→
   365→                  <div className="flex items-center justify-between mt-2 pt-2 border-t border-gray-900/30 text-[10px] text-gray-500 font-mono">
   366→                    <span>{ex.docType}</span>
   367→                    <span className="text-gray-600 uppercase">{ex.exceptionType.replace(/_/g, ' ')}</span>
   368→                  </div>
   369→                </motion.div>
   370→              );
   371→            })}
   372→          </AnimatePresence>
   373→
   374→          {sortedExceptions.length === 0 && (
   375→            <div className="p-8 text-center text-gray-500 text-xs">
   376→              No exceptions match the selected filter.
   377→            </div>
   378→          )}
   379→        </div>
   380→      </div>
   381→
   382→      {/* CENTER PANEL: DOCUMENT VIEWER */}
   383→      <div className="lg:col-span-5 bg-[#0c0d12] border border-gray-900 rounded-xl flex flex-col overflow-hidden relative">
   384→        {/* Document Tabs (data-driven) */}
   385→        <div className="px-4 py-2 bg-[#0e1017] border-b border-gray-900 flex items-center justify-between shrink-0">
   386→          <div className="flex items-center gap-1.5 overflow-x-auto">
   387→            {availableDocTypes.map(doc => (
   388→              <button
   389→                key={doc}
   390→                onClick={() => setActiveDocTab(doc)}
   391→                className={`text-xs px-3 py-1.5 rounded-md font-medium transition-all whitespace-nowrap ${
   392→                  activeDocTab === doc
   393→                    ? 'bg-gray-800 text-gray-100'
   394→                    : 'text-gray-500 hover:text-gray-300'
   395→                }`}
   396→              >
   397→                {doc === 'Commercial Invoice' ? 'Invoice' : doc === 'Bill of Lading' ? 'BOL' : doc}
   398→              </button>
   399→            ))}
   400→            {availableDocTypes.length === 0 && (
   401→              <span className="text-xs text-gray-600">No documents available</span>
   402→            )}
   403→          </div>
   404→
   405→          <div className="flex items-center gap-1 shrink-0">
   406→            <button
   407→              onClick={() => setZoomLevel(prev => Math.max(50, prev - 10))}
   408→              title="Zoom Out"
   409→              className="p-1 rounded text-gray-500 hover:text-gray-300 hover:bg-gray-950 transition-all cursor-pointer"
   410→            >
   411→              <ZoomOut className="w-4 h-4" />
   412→            </button>
   413→            <span className="text-xs font-mono text-gray-500 w-10 text-center">{zoomLevel}%</span>
   414→            <button
   415→              onClick={() => setZoomLevel(prev => Math.min(200, prev + 10))}
   416→              title="Zoom In"
   417→              className="p-1 rounded text-gray-500 hover:text-gray-300 hover:bg-gray-950 transition-all cursor-pointer"
   418→            >
   419→              <ZoomIn className="w-4 h-4" />
   420→            </button>
   421→            <button
   422→              onClick={() => setRotation(prev => (prev + 90) % 360)}
   423→              title="Rotate 90°"
   424→              className="p-1 rounded text-gray-500 hover:text-gray-300 hover:bg-gray-950 transition-all cursor-pointer"
   425→            >
   426→              <RotateCw className="w-4 h-4" />
   427→            </button>
   428→          </div>
   429→        </div>
   430→
   431→        {/* Document Board — shows real uploaded file or extracted data view */}
   432→        <div className="flex-1 overflow-auto bg-[#040406] p-6 flex items-start justify-center relative">
   433→          {documentUrl ? (
   434→            /* Real file viewer — show the actual uploaded document based on MIME type */
   435→            <div
   436→              style={{
   437→                transform: `scale(${zoomLevel / 100}) rotate(${rotation}deg)`,
   438→                transformOrigin: 'top center',
   439→                transition: 'transform 0.15s ease-out',
   440→              }}
   441→              className="rounded border border-gray-700 shadow-2xl overflow-hidden bg-white relative"
   442→            >
   443→              {documentMime.startsWith('text/') ? (
   444→                /* Text file — show in iframe with monospace styling */
   445→                <iframe
   446→                  src={documentUrl}
   447→                  className="w-[500px] h-[600px] bg-white"
   448→                  title="Document text"
   449→                  style={{ fontFamily: 'monospace', fontSize: '12px' }}
   450→                />
   451→              ) : documentMime === 'application/pdf' ? (
   452→                /* PDF — show in iframe (browser's built-in PDF viewer) */
   453→                <iframe
   454→                  src={documentUrl}
   455→                  className="w-[500px] h-[600px] bg-white"
   456→                  title="Document PDF"
   457→                />
   458→              ) : documentMime.startsWith('image/') ? (
   459→                /* Image (PNG, JPEG, TIFF) — show directly with zoom/rotate */
   460→                <img
   461→                  src={documentUrl}
   462→                  alt={`Document: ${activeDocTab}`}
   463→                  className="max-w-[500px] max-h-[600px] object-contain"
   464→                  onError={(e) => {
   465→                    console.warn('[doc-viewer] image failed to load:', documentUrl);
   466→                  }}
   467→                />
   468→              ) : (
   469→                /* Unknown type — provide download link */
   470→                <div className="w-[500px] h-[600px] flex flex-col items-center justify-center bg-gray-50 p-8 text-center">
   471→                  <FileText className="w-16 h-16 text-gray-400 mb-4" />
   472→                  <p className="text-sm font-semibold text-gray-700 mb-2">
   473→                    {activeDocTab || 'Document'}
   474→                  </p>
   475→                  <p className="text-xs text-gray-500 mb-4">
   476→                    Preview not available for this file type ({documentMime || 'unknown'})
   477→                  </p>
   478→                  <a
   479→                    href={documentUrl}
   480→                    target="_blank"
   481→                    rel="noreferrer"
   482→                    className="text-xs bg-amber-500 hover:bg-amber-600 text-black font-bold px-4 py-2 rounded-lg transition-all"
   483→                  >
   484→                    Download File
   485→                  </a>
   486→                </div>
   487→              )}
   488→
   489→              {/* Highlighted bounding box overlay */}
   490→              {selectedException && activeDocTab === selectedException.docType && (

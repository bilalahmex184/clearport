    70→    70→
    71→    71→export const ClearPortProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
    72→    72→  const [entries, setEntries] = React.useState<ShipmentEntry[]>(seedEntries);
    73→    73→  const [selectedEntryId, setSelectedEntryId] = React.useState<string>('SHIP-2026-8802');
    74→    74→  const [selectedExceptionId, setSelectedExceptionId] = React.useState<string>('8802-hts');
    75→    75→  const [activeTab, setActiveTab] = React.useState<string>('exception-desk');
    76→    76→  const [theme, setTheme] = React.useState<'dark' | 'light'>('dark');
    77→    77→  const [rules, setRules] = React.useState<OperationalRules>(seedRules);
    78→    78→  const [undoStack, setUndoStack] = React.useState<{ entryId: string; exceptionId: string; previousState: Exception }[]>([]);
    79→    79→  const [auditLogs, setAuditLogs] = React.useState<AuditLog[]>(seedLogs);
    80→    80→  const [supabaseStatus, setSupabaseStatus] = React.useState<SupabaseStatus>(
    81→    81→    isSupabaseConfigured() ? 'loading' : 'unconfigured'
    82→    82→  );
    83→    83→  const [edgeFunctionStatus, setEdgeFunctionStatus] = React.useState<EdgeFunctionStatus>('unknown');
    84→    84→  const [currentUser, setCurrentUser] = React.useState<string>('Broker');
    85→    85→  const [currentTime, setCurrentTime] = React.useState<string>('');
    86→    86→
    87→    87→  // --- Real-time clock ---
    88→    88→  React.useEffect(() => {
    89→    89→    const update = () => {
    90→    90→      const now = new Date();
    91→    91→      const utc = now.toISOString().slice(0, 16).replace('T', ' ') + ' UTC';
    92→    92→      setCurrentTime(utc);
    93→    93→    };
    94→    94→    update();
    95→    95→    const interval = setInterval(update, 1000);
    96→    96→    return () => clearInterval(interval);
    97→    97→  }, []);
    98→    98→
    99→    99→  // --- Load user email ---
   100→   100→  React.useEffect(() => {
   101→   101→    getCurrentUserEmail().then(email => setCurrentUser(email));
   102→   102→  }, []);
   103→   103→
   104→   104→  // --- Persist theme ---
   105→   105→  React.useEffect(() => {
   106→   106→    const saved = typeof window !== 'undefined' ? localStorage.getItem('clearport-theme') : null;
   107→   107→    if (saved === 'light' || saved === 'dark') setTheme(saved);
   108→   108→  }, []);
   109→   109→
   110→   110→  const toggleTheme = React.useCallback(() => {
   111→   111→    setTheme(prev => {
   112→   112→      const next = prev === 'dark' ? 'light' : 'dark';
   113→   113→      if (typeof window !== 'undefined') localStorage.setItem('clearport-theme', next);
   114→   114→      return next;
   115→   115→    });
   116→   116→  }, []);
   117→   117→
   118→   118→  // --- Load data ---
   119→   119→  const selectedEntryIdRef = React.useRef(selectedEntryId);
   120→   120→  selectedEntryIdRef.current = selectedEntryId;
   121→   121→
   122→   122→  const loadData = React.useCallback(async () => {
   123→   123→    if (!isSupabaseConfigured()) {
   124→   124→      setSupabaseStatus('unconfigured');
   125→   125→      setEdgeFunctionStatus('fallback');
   126→   126→      return;
   127→   127→    }
   128→   128→
   129→   129→    setSupabaseStatus('loading');
   130→   130→
   131→   131→    try {
   132→   132→      await ensureAuthenticated();
   133→   133→      const email = await getCurrentUserEmail();
   134→   134→      setCurrentUser(email);
   135→   135→
   136→   136→      // Try the new /api/shipments route (replaces both the get-shipments
   137→   137→      // edge function call AND the fetchShipmentsDirect fallback).
   138→   138→      try {
   139→   139→        const response = await apiFetch<{
   140→   140→          data: ShipmentEntry[];
   141→   141→          pagination: { total: number };
   142→   142→        }>('/api/shipments?page=1&limit=100');
   143→   143→
   144→   144→        if (response?.data && Array.isArray(response.data)) {
   145→   145→          // Only use DB shipments if they have fields/exceptions data (new
   146→   146→          // schema). Old-schema shipments have empty arrays — fall back to
   147→   147→          // seed data so the demo UX is preserved.
   148→   148→          const hasRealData = response.data.some(
   149→   149→            (s) => s.fields.length > 0 || s.exceptions.length > 0,
   150→   150→          );
   151→   151→
   152→   152→          if (hasRealData) {
   153→   153→            setEntries(response.data);
   154→   154→            // Select first shipment if current selection doesn't exist
   155→   155→            if (!response.data.find((s) => s.id === selectedEntryIdRef.current)) {
   156→   156→              const first = response.data[0];
   157→   157→              setSelectedEntryId(first.id);
   158→   158→              if (first.exceptions?.length > 0) {
   159→   159→                const unresolved = first.exceptions.find((e) => e.status === 'Unresolved');
   160→   160→                setSelectedExceptionId(unresolved ? unresolved.id : first.exceptions[0].id);
   161→   161→              }
   162→   162→            }
   163→   163→          }
   164→   164→          // else: keep seed entries (demo mode)
   165→   165→
   166→   166→          setEdgeFunctionStatus('live');
   167→   167→          setSupabaseStatus('connected');
   168→   168→
   169→   169→          // Fetch rules + logs in parallel via the new API routes.
   170→   170→          const [rulesRes, logsRes] = await Promise.all([
   171→   171→            apiFetch<{ rules: OperationalRules }>('/api/rules').catch(() => null),
   172→   172→            apiFetch<{ logs: AuditLog[] }>('/api/audit-logs').catch(() => null),
   173→   173→          ]);
   174→   174→          if (rulesRes?.rules) setRules(rulesRes.rules);
   175→   175→          if (logsRes?.logs && logsRes.logs.length > 0) setAuditLogs(logsRes.logs);
   176→   176→          return;
   177→   177→        }
   178→   178→      } catch (apiErr) {
   179→   179→        console.warn('[ctx] /api/shipments failed, falling back to seed data:', apiErr);
   180→   180→      }
   181→   181→
   182→   182→      // Fallback: seed data (demo mode)
   183→   183→      setSupabaseStatus('connected');
   184→   184→      setEdgeFunctionStatus('fallback');
   185→   185→    } catch (err) {
   186→   186→      console.error('[ctx] loadData error:', err);
   187→   187→      setSupabaseStatus('error_tables');
   188→   188→      setEdgeFunctionStatus('fallback');
   189→   189→    }
   190→   190→  }, []);
   191→   191→
   192→   192→  React.useEffect(() => {
   193→   193→    loadData();
   194→   194→  }, [loadData]);
   195→   195→
   196→   196→  const refreshData = React.useCallback(async () => {
   197→   197→    await loadData();
   198→   198→  }, [loadData]);
   199→   199→
   200→   200→  // --- Computed ---
   201→   201→  const selectedEntry = React.useMemo(
   202→   202→    () => entries.find(e => e.id === selectedEntryId),
   203→   203→    [entries, selectedEntryId]
   204→   204→  );
   205→   205→
   206→   206→  const selectedException = React.useMemo(() => {
   207→   207→    if (!selectedEntry) return undefined;
   208→   208→    return selectedEntry.exceptions.find(ex => ex.id === selectedExceptionId);
   209→   209→  }, [selectedEntry, selectedExceptionId]);
   210→   210→
   211→   211→  // --- Selection (fixes stale closure — uses functional update) ---
   212→   212→  const selectEntry = React.useCallback((id: string) => {
   213→   213→    setSelectedEntryId(id);
   214→   214→    setEntries(prev => {
   215→   215→      const entry = prev.find(e => e.id === id);
   216→   216→      if (entry && entry.exceptions.length > 0) {
   217→   217→        const unresolved = entry.exceptions.find(e => e.status === 'Unresolved');
   218→   218→        setSelectedExceptionId(unresolved ? unresolved.id : entry.exceptions[0].id);
   219→   219→      } else {
   220→   220→        setSelectedExceptionId('');
   221→   221→      }
   222→   222→      return prev;
   223→   223→    });
   224→   224→  }, []);
   225→   225→
   226→   226→  const selectException = React.useCallback((id: string) => {
   227→   227→    setSelectedExceptionId(id);
   228→   228→  }, []);
   229→   229→
   230→   230→  // --- Audit log helper ---
   231→   231→  const addAuditLog = React.useCallback((text: string, type: AuditLog['type'] = 'info', shipmentId?: string) => {
   232→   232→    const newLog: AuditLog = {
   233→   233→      id: typeof crypto !== 'undefined' ? crypto.randomUUID() : `log-${Date.now()}`,
   234→   234→      text,
   235→   235→      timestamp: new Date().toISOString(),
   236→   236→      type,
   237→   237→      shipmentId,
   238→   238→    };
   239→   239→    setAuditLogs(prev => [newLog, ...prev]);
   240→   240→
   241→   241→    if (isSupabaseConfigured() && supabase) {
   242→   242→      supabase.from('audit_logs').insert({
   243→   243→        id: newLog.id,
   244→   244→        text: newLog.text,
   245→   245→        timestamp: newLog.timestamp,
   246→   246→        type: newLog.type,
   247→   247→        shipment_id: shipmentId || null,
   248→   248→      }).then(({ error }) => {
   249→   249→        if (error) console.warn('[ctx] audit log insert failed:', error.message);
   250→   250→      });
   251→   251→    }
   252→   252→  }, []);
   253→   253→
   254→   254→  // --- Update exception (fixes setState-in-setState) ---
   255→   255→  const updateException = React.useCallback(
   256→   256→    (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => {
   257→   257→      let previousException: Exception | undefined;
   258→   258→      let updatedEntry: ShipmentEntry | undefined;
   259→   259→
   260→   260→      setEntries(prevEntries => {
   261→   261→        return prevEntries.map(entry => {
   262→   262→          if (entry.id !== entryId) return entry;
   263→   263→
   264→   264→          const prev = entry.exceptions.find(ex => ex.id === exceptionId);
   265→   265→          if (!prev) return entry;
   266→   266→          previousException = { ...prev };
   267→   267→
   268→   268→          const updatedExceptions = entry.exceptions.map(ex => {
   269→   269→            if (ex.id !== exceptionId) return ex;
   270→   270→
   271→   271→            const oldVal = ex.correctedValue || ex.extractedValue;
   272→   272→            const updatedVal = newValue !== undefined ? newValue : ex.extractedValue;
   273→   273→
   274→   274→            const historyItem: ExceptionHistoryEntry = {
   275→   275→              user: currentUser,
   276→   276→              oldValue: oldVal,
   277→   277→              newValue: updatedVal,
   278→   278→              timestamp: new Date().toISOString(),
   279→   279→              action: status,
   280→   280→            };
   281→   281→
   282→   282→            return {
   283→   283→              ...ex,
   284→   284→              status: status as ExceptionStatus,
   285→   285→              correctedValue: status === 'Corrected' ? newValue : undefined,
   286→   286→              history: [historyItem, ...ex.history],
   287→   287→              resolvedAt: new Date().toISOString(),
   288→   288→              resolvedBy: currentUser,
   289→   289→            };
   290→   290→          });
   291→   291→
   292→   292→          const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   293→   293→          const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   294→   294→          const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   295→   295→
   296→   296→          const updatedFields = entry.fields.map(f => {
   297→   297→            if (f.exceptionId === exceptionId) {
   298→   298→              const val = status === 'Corrected' ? (newValue || f.value) : f.value;
   299→   299→              return { ...f, value: val, isFlagged: false };
   300→   300→            }
   301→   301→            return f;
   302→   302→          });
   303→   303→
   304→   304→          updatedEntry = {
   305→   305→            ...entry,
   306→   306→            status: newStatus,
   307→   307→            exceptions: updatedExceptions,
   308→   308→            fields: updatedFields,
   309→   309→            currentConfidence: newConfidence,
   310→   310→          };
   311→   311→
   312→   312→          return updatedEntry;
   313→   313→        });
   314→   314→      });
   315→   315→
   316→   316→      // Side effects AFTER setState
   317→   317→      if (previousException && updatedEntry) {
   318→   318→        setUndoStack(prevStack => [
   319→   319→          { entryId, exceptionId, previousState: { ...previousException! } },
   320→   320→          ...prevStack,
   321→   321→        ]);
   322→   322→
   323→   323→        const logText = `Exception "${previousException.fieldName}" for ${entryId} was ${status.toLowerCase()}${
   324→   324→          status === 'Corrected' ? ` to "${newValue}"` : ''
   325→   325→        } by ${currentUser}.`;
   326→   326→        addAuditLog(logText, status === 'Rejected' ? 'warning' : 'success', entryId);
   327→   327→
   328→   328→        // Persist via the /api/exceptions/:id route (handles exception update,
   329→   329→        // document_field sync, shipment confidence recompute, and audit log
   330→   330→        // in one server-side transaction).
   331→   331→        if (isSupabaseConfigured()) {
   332→   332→          apiFetch(`/api/exceptions/${exceptionId}`, {
   333→   333→            method: 'PATCH',
   334→   334→            body: JSON.stringify({
   335→   335→              status,
   336→   336→              correctedValue: status === 'Corrected' ? newValue : undefined,
   337→   337→            }),
   338→   338→          }).catch((err) => {
   339→   339→            console.warn('[ctx] exception update failed:', err instanceof Error ? err.message : err);
   340→   340→          });
   341→   341→        }
   342→   342→      }
   343→   343→    },
   344→   344→    [currentUser, addAuditLog]
   345→   345→  );
   346→   346→
   347→   347→  // --- Undo ---
   348→   348→  const undoLastAction = React.useCallback(() => {
   349→   349→    if (undoStack.length === 0) return;
   350→   350→
   351→   351→    const [lastAction, ...remainingStack] = undoStack;
   352→   352→    setUndoStack(remainingStack);
   353→   353→
   354→   354→    setEntries(prevEntries => {
   355→   355→      return prevEntries.map(entry => {
   356→   356→        if (entry.id !== lastAction.entryId) return entry;
   357→   357→
   358→   358→        const restored = lastAction.previousState;
   359→   359→        const updatedExceptions = entry.exceptions.map(ex =>
   360→   360→          ex.id === lastAction.exceptionId ? restored : ex
   361→   361→        );
   362→   362→
   363→   363→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   364→   364→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   365→   365→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   366→   366→
   367→   367→        const updatedFields = entry.fields.map(f => {
   368→   368→          if (f.exceptionId === lastAction.exceptionId) {
   369→   369→            return {
   370→   370→              ...f,
   371→   371→              value: restored.status === 'Corrected' ? (restored.correctedValue || f.value) : restored.extractedValue,
   372→   372→              isFlagged: restored.status === 'Unresolved',
   373→   373→            };
   374→   374→          }
   375→   375→          return f;
   376→   376→        });
   377→   377→
   378→   378→        const updatedEntry = {
   379→   379→          ...entry,
   380→   380→          status: newStatus,
   381→   381→          exceptions: updatedExceptions,
   382→   382→          fields: updatedFields,
   383→   383→          currentConfidence: newConfidence,
   384→   384→        };
   385→   385→
   386→   386→        if (isSupabaseConfigured() && supabase) {
   387→   387→          supabase.from('exceptions').update({
   388→   388→            status: restored.status,
   389→   389→            corrected_value: restored.correctedValue || null,
   390→   390→            resolved_at: restored.resolvedAt || null,
   391→   391→            resolved_by: restored.resolvedBy || null,
   392→   392→            history: restored.history,
   393→   393→          }).eq('id', lastAction.exceptionId).then(({ error }) => {
   394→   394→            if (error) console.warn('[ctx] undo exception failed:', error.message);
   395→   395→          });
   396→   396→
   397→   397→          supabase.from('shipments').update({
   398→   398→            status: newStatus,
   399→   399→            current_confidence: newConfidence,
   400→   400→            updated_at: new Date().toISOString(),
   401→   401→          }).eq('id', entry.id).then(({ error }) => {
   402→   402→            if (error) console.warn('[ctx] undo shipment failed:', error.message);
   403→   403→          });
   404→   404→        }
   405→   405→
   406→   406→        return updatedEntry;
   407→   407→      });
   408→   408→    });
   409→   409→
   410→   410→    addAuditLog(`Undo applied for last review action on shipment ${lastAction.entryId}`, 'info', lastAction.entryId);
   411→   411→  }, [undoStack, addAuditLog]);
   412→   412→
   413→   413→  // --- Batch accept (uses configured threshold) ---
   414→   414→  const acceptAllHighConfidence = React.useCallback((entryId: string) => {
   415→   415→    setEntries(prevEntries => {
   416→   416→      return prevEntries.map(entry => {
   417→   417→        if (entry.id !== entryId) return entry;
   418→   418→
   419→   419→        const cutoff = rules.invoiceThreshold;
   420→   420→        let acceptedCount = 0;
   421→   421→
   422→   422→        const updatedExceptions = entry.exceptions.map(ex => {
   423→   423→          if (ex.status === 'Unresolved' && ex.confidence >= cutoff) {
   424→   424→            acceptedCount++;
   425→   425→            const historyItem: ExceptionHistoryEntry = {
   426→   426→              user: currentUser,
   427→   427→              oldValue: ex.extractedValue,
   428→   428→              newValue: ex.extractedValue,
   429→   429→              timestamp: new Date().toISOString(),
   430→   430→              action: 'Accepted' as ReviewerAction,
   431→   431→            };
   432→   432→            return {
   433→   433→              ...ex,
   434→   434→              status: 'Accepted' as ExceptionStatus,
   435→   435→              history: [historyItem, ...ex.history],
   436→   436→              resolvedAt: new Date().toISOString(),
   437→   437→              resolvedBy: currentUser,
   438→   438→            };
   439→   439→          }
   440→   440→          return ex;
   441→   441→        });
   442→   442→
   443→   443→        if (acceptedCount === 0) return entry;
   444→   444→
   445→   445→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   446→   446→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   447→   447→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   448→   448→
   449→   449→        const updatedFields = entry.fields.map(f => {
   450→   450→          const exc = updatedExceptions.find(e => e.id === f.exceptionId);
   451→   451→          if (exc) return { ...f, isFlagged: exc.status === 'Unresolved' };
   452→   452→          return f;
   453→   453→        });
   454→   454→
   455→   455→        const updatedEntry = {
   456→   456→          ...entry,
   457→   457→          status: newStatus,
   458→   458→          exceptions: updatedExceptions,
   459→   459→          fields: updatedFields,
   460→   460→          currentConfidence: newConfidence,
   461→   461→        };
   462→   462→
   463→   463→        addAuditLog(
   464→   464→          `Batch action: Approved ${acceptedCount} high-confidence exceptions (≥${cutoff}%) in ${entryId}.`,
   465→   465→          'success',
   466→   466→          entryId
   467→   467→        );
   468→   468→
   469→   469→        // Persist via the /api/exceptions/batch-accept route (accepts each
   470→   470→        // qualifying exception, syncs document_fields, recomputes shipment
   471→   471→        // confidence + status, and writes a batch-level audit log).
   472→   472→        if (isSupabaseConfigured()) {
   473→   473→          apiFetch('/api/exceptions/batch-accept', {
   474→   474→            method: 'POST',
   475→   475→            body: JSON.stringify({
   476→   476→              shipmentId: entryId,
   477→   477→              threshold: cutoff,
   478→   478→            }),
   479→   479→          }).catch((err) => {
   480→   480→            console.warn('[ctx] batch accept failed:', err instanceof Error ? err.message : err);
   481→   481→          });
   482→   482→        }

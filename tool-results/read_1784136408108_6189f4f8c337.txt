    79→    79→
    80→    80→export const ClearPortProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
    81→    81→  const [entries, setEntries] = React.useState<ShipmentEntry[]>(seedEntries);
    82→    82→  const [selectedEntryId, setSelectedEntryId] = React.useState<string>('SHIP-2026-8802');
    83→    83→  const [selectedExceptionId, setSelectedExceptionId] = React.useState<string>('8802-hts');
    84→    84→  const [activeTab, setActiveTab] = React.useState<string>('exception-desk');
    85→    85→  const [theme, setTheme] = React.useState<'dark' | 'light'>('dark');
    86→    86→  const [rules, setRules] = React.useState<OperationalRules>(seedRules);
    87→    87→  const [undoStack, setUndoStack] = React.useState<{ entryId: string; exceptionId: string; previousState: Exception }[]>([]);
    88→    88→  const [auditLogs, setAuditLogs] = React.useState<AuditLog[]>(seedLogs);
    89→    89→  const [supabaseStatus, setSupabaseStatus] = React.useState<SupabaseStatus>(
    90→    90→    isSupabaseConfigured() ? 'loading' : 'unconfigured'
    91→    91→  );
    92→    92→  const [edgeFunctionStatus, setEdgeFunctionStatus] = React.useState<EdgeFunctionStatus>('unknown');
    93→    93→  const [currentUser, setCurrentUser] = React.useState<string>('Broker');
    94→    94→  const [currentTime, setCurrentTime] = React.useState<string>('');
    95→    95→  // Anonymous users default to 'operator'. In production this would be
    96→    96→  // fetched from the server (e.g. GET /api/me → role from user_roles table).
    97→    97→  const [userRole, setUserRole] = React.useState<UserRole>(getDefaultRole());
    98→    98→
    99→    99→  // --- Real-time clock ---
   100→   100→  React.useEffect(() => {
   101→   101→    const update = () => {
   102→   102→      const now = new Date();
   103→   103→      const utc = now.toISOString().slice(0, 16).replace('T', ' ') + ' UTC';
   104→   104→      setCurrentTime(utc);
   105→   105→    };
   106→   106→    update();
   107→   107→    const interval = setInterval(update, 1000);
   108→   108→    return () => clearInterval(interval);
   109→   109→  }, []);
   110→   110→
   111→   111→  // --- Load user email ---
   112→   112→  React.useEffect(() => {
   113→   113→    getCurrentUserEmail().then(email => setCurrentUser(email));
   114→   114→  }, []);
   115→   115→
   116→   116→  // --- Persist theme ---
   117→   117→  React.useEffect(() => {
   118→   118→    const saved = typeof window !== 'undefined' ? localStorage.getItem('clearport-theme') : null;
   119→   119→    if (saved === 'light' || saved === 'dark') setTheme(saved);
   120→   120→  }, []);
   121→   121→
   122→   122→  const toggleTheme = React.useCallback(() => {
   123→   123→    setTheme(prev => {
   124→   124→      const next = prev === 'dark' ? 'light' : 'dark';
   125→   125→      if (typeof window !== 'undefined') localStorage.setItem('clearport-theme', next);
   126→   126→      return next;
   127→   127→    });
   128→   128→  }, []);
   129→   129→
   130→   130→  // --- Load data ---
   131→   131→  const selectedEntryIdRef = React.useRef(selectedEntryId);
   132→   132→  selectedEntryIdRef.current = selectedEntryId;
   133→   133→
   134→   134→  const loadData = React.useCallback(async () => {
   135→   135→    if (!isSupabaseConfigured()) {
   136→   136→      setSupabaseStatus('unconfigured');
   137→   137→      setEdgeFunctionStatus('fallback');
   138→   138→      return;
   139→   139→    }
   140→   140→
   141→   141→    setSupabaseStatus('loading');
   142→   142→
   143→   143→    try {
   144→   144→      await ensureAuthenticated();
   145→   145→      const email = await getCurrentUserEmail();
   146→   146→      setCurrentUser(email);
   147→   147→
   148→   148→      // Try the new /api/shipments route (replaces both the get-shipments
   149→   149→      // edge function call AND the fetchShipmentsDirect fallback).
   150→   150→      try {
   151→   151→        const response = await apiFetch<{
   152→   152→          data: ShipmentEntry[];
   153→   153→          pagination: { total: number };
   154→   154→        }>('/api/shipments?page=1&limit=100');
   155→   155→
   156→   156→        if (response?.data && Array.isArray(response.data)) {
   157→   157→          // Only use DB shipments if they have fields/exceptions data (new
   158→   158→          // schema). Old-schema shipments have empty arrays — fall back to
   159→   159→          // seed data so the demo UX is preserved.
   160→   160→          const hasRealData = response.data.some(
   161→   161→            (s) => s.fields.length > 0 || s.exceptions.length > 0,
   162→   162→          );
   163→   163→
   164→   164→          if (hasRealData) {
   165→   165→            setEntries(response.data);
   166→   166→            // Select first shipment if current selection doesn't exist
   167→   167→            if (!response.data.find((s) => s.id === selectedEntryIdRef.current)) {
   168→   168→              const first = response.data[0];
   169→   169→              setSelectedEntryId(first.id);
   170→   170→              if (first.exceptions?.length > 0) {
   171→   171→                const unresolved = first.exceptions.find((e) => e.status === 'Unresolved');
   172→   172→                setSelectedExceptionId(unresolved ? unresolved.id : first.exceptions[0].id);
   173→   173→              }
   174→   174→            }
   175→   175→          }
   176→   176→          // else: keep seed entries (demo mode)
   177→   177→
   178→   178→          setEdgeFunctionStatus('live');
   179→   179→          setSupabaseStatus('connected');
   180→   180→
   181→   181→          // Fetch rules + logs in parallel via the new API routes.
   182→   182→          const [rulesRes, logsRes] = await Promise.all([
   183→   183→            apiFetch<{ rules: OperationalRules }>('/api/rules').catch(() => null),
   184→   184→            apiFetch<{ logs: AuditLog[] }>('/api/audit-logs').catch(() => null),
   185→   185→          ]);
   186→   186→          if (rulesRes?.rules) setRules(rulesRes.rules);
   187→   187→          if (logsRes?.logs && logsRes.logs.length > 0) setAuditLogs(logsRes.logs);
   188→   188→          return;
   189→   189→        }
   190→   190→      } catch (apiErr) {
   191→   191→        console.warn('[ctx] /api/shipments failed, falling back to seed data:', apiErr);
   192→   192→      }
   193→   193→
   194→   194→      // Fallback: seed data (demo mode)
   195→   195→      setSupabaseStatus('connected');
   196→   196→      setEdgeFunctionStatus('fallback');
   197→   197→    } catch (err) {
   198→   198→      console.error('[ctx] loadData error:', err);
   199→   199→      setSupabaseStatus('error_tables');
   200→   200→      setEdgeFunctionStatus('fallback');
   201→   201→    }
   202→   202→  }, []);
   203→   203→
   204→   204→  React.useEffect(() => {
   205→   205→    loadData();
   206→   206→  }, [loadData]);
   207→   207→
   208→   208→  const refreshData = React.useCallback(async () => {
   209→   209→    await loadData();
   210→   210→  }, [loadData]);
   211→   211→
   212→   212→  // --- Computed ---
   213→   213→  const selectedEntry = React.useMemo(
   214→   214→    () => entries.find(e => e.id === selectedEntryId),
   215→   215→    [entries, selectedEntryId]
   216→   216→  );
   217→   217→
   218→   218→  const selectedException = React.useMemo(() => {
   219→   219→    if (!selectedEntry) return undefined;
   220→   220→    return selectedEntry.exceptions.find(ex => ex.id === selectedExceptionId);
   221→   221→  }, [selectedEntry, selectedExceptionId]);
   222→   222→
   223→   223→  // --- Selection (fixes stale closure — uses functional update) ---
   224→   224→  const selectEntry = React.useCallback((id: string) => {
   225→   225→    setSelectedEntryId(id);
   226→   226→    setEntries(prev => {
   227→   227→      const entry = prev.find(e => e.id === id);
   228→   228→      if (entry && entry.exceptions.length > 0) {
   229→   229→        const unresolved = entry.exceptions.find(e => e.status === 'Unresolved');
   230→   230→        setSelectedExceptionId(unresolved ? unresolved.id : entry.exceptions[0].id);
   231→   231→      } else {
   232→   232→        setSelectedExceptionId('');
   233→   233→      }
   234→   234→      return prev;
   235→   235→    });
   236→   236→  }, []);
   237→   237→
   238→   238→  const selectException = React.useCallback((id: string) => {
   239→   239→    setSelectedExceptionId(id);
   240→   240→  }, []);
   241→   241→
   242→   242→  // --- Audit log helper ---
   243→   243→  const addAuditLog = React.useCallback((text: string, type: AuditLog['type'] = 'info', shipmentId?: string) => {
   244→   244→    const newLog: AuditLog = {
   245→   245→      id: typeof crypto !== 'undefined' ? crypto.randomUUID() : `log-${Date.now()}`,
   246→   246→      text,
   247→   247→      timestamp: new Date().toISOString(),
   248→   248→      type,
   249→   249→      shipmentId,
   250→   250→    };
   251→   251→    setAuditLogs(prev => [newLog, ...prev]);
   252→   252→
   253→   253→    if (isSupabaseConfigured() && supabase) {
   254→   254→      supabase.from('audit_logs').insert({
   255→   255→        id: newLog.id,
   256→   256→        text: newLog.text,
   257→   257→        timestamp: newLog.timestamp,
   258→   258→        type: newLog.type,
   259→   259→        shipment_id: shipmentId || null,
   260→   260→      }).then(({ error }) => {
   261→   261→        if (error) console.warn('[ctx] audit log insert failed:', error.message);
   262→   262→      });
   263→   263→    }
   264→   264→  }, []);
   265→   265→
   266→   266→  // --- Update exception (fixes setState-in-setState) ---
   267→   267→  const updateException = React.useCallback(
   268→   268→    (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => {
   269→   269→      let previousException: Exception | undefined;
   270→   270→      let updatedEntry: ShipmentEntry | undefined;
   271→   271→
   272→   272→      setEntries(prevEntries => {
   273→   273→        return prevEntries.map(entry => {
   274→   274→          if (entry.id !== entryId) return entry;
   275→   275→
   276→   276→          const prev = entry.exceptions.find(ex => ex.id === exceptionId);
   277→   277→          if (!prev) return entry;
   278→   278→          previousException = { ...prev };
   279→   279→
   280→   280→          const updatedExceptions = entry.exceptions.map(ex => {
   281→   281→            if (ex.id !== exceptionId) return ex;
   282→   282→
   283→   283→            const oldVal = ex.correctedValue || ex.extractedValue;
   284→   284→            const updatedVal = newValue !== undefined ? newValue : ex.extractedValue;
   285→   285→
   286→   286→            const historyItem: ExceptionHistoryEntry = {
   287→   287→              user: currentUser,
   288→   288→              oldValue: oldVal,
   289→   289→              newValue: updatedVal,
   290→   290→              timestamp: new Date().toISOString(),
   291→   291→              action: status,
   292→   292→            };
   293→   293→
   294→   294→            return {
   295→   295→              ...ex,
   296→   296→              status: status as ExceptionStatus,
   297→   297→              correctedValue: status === 'Corrected' ? newValue : undefined,
   298→   298→              history: [historyItem, ...ex.history],
   299→   299→              resolvedAt: new Date().toISOString(),
   300→   300→              resolvedBy: currentUser,
   301→   301→            };
   302→   302→          });
   303→   303→
   304→   304→          const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   305→   305→          const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   306→   306→          const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   307→   307→
   308→   308→          const updatedFields = entry.fields.map(f => {
   309→   309→            if (f.exceptionId === exceptionId) {
   310→   310→              const val = status === 'Corrected' ? (newValue || f.value) : f.value;
   311→   311→              return { ...f, value: val, isFlagged: false };
   312→   312→            }
   313→   313→            return f;
   314→   314→          });
   315→   315→
   316→   316→          updatedEntry = {
   317→   317→            ...entry,
   318→   318→            status: newStatus,
   319→   319→            exceptions: updatedExceptions,
   320→   320→            fields: updatedFields,
   321→   321→            currentConfidence: newConfidence,
   322→   322→          };
   323→   323→
   324→   324→          return updatedEntry;
   325→   325→        });
   326→   326→      });
   327→   327→
   328→   328→      // Side effects AFTER setState
   329→   329→      if (previousException && updatedEntry) {
   330→   330→        setUndoStack(prevStack => [
   331→   331→          { entryId, exceptionId, previousState: { ...previousException! } },
   332→   332→          ...prevStack,
   333→   333→        ]);
   334→   334→
   335→   335→        const logText = `Exception "${previousException.fieldName}" for ${entryId} was ${status.toLowerCase()}${
   336→   336→          status === 'Corrected' ? ` to "${newValue}"` : ''
   337→   337→        } by ${currentUser}.`;
   338→   338→        addAuditLog(logText, status === 'Rejected' ? 'warning' : 'success', entryId);
   339→   339→
   340→   340→        // Persist via the /api/exceptions/:id route (handles exception update,
   341→   341→        // document_field sync, shipment confidence recompute, and audit log
   342→   342→        // in one server-side transaction).
   343→   343→        if (isSupabaseConfigured()) {
   344→   344→          apiFetch(`/api/exceptions/${exceptionId}`, {
   345→   345→            method: 'PATCH',
   346→   346→            body: JSON.stringify({
   347→   347→              status,
   348→   348→              correctedValue: status === 'Corrected' ? newValue : undefined,
   349→   349→            }),
   350→   350→          }).catch((err) => {
   351→   351→            console.warn('[ctx] exception update failed:', err instanceof Error ? err.message : err);
   352→   352→          });
   353→   353→        }
   354→   354→      }
   355→   355→    },
   356→   356→    [currentUser, addAuditLog]
   357→   357→  );
   358→   358→
   359→   359→  // --- Undo ---
   360→   360→  const undoLastAction = React.useCallback(() => {
   361→   361→    if (undoStack.length === 0) return;
   362→   362→
   363→   363→    const [lastAction, ...remainingStack] = undoStack;
   364→   364→    setUndoStack(remainingStack);
   365→   365→
   366→   366→    setEntries(prevEntries => {
   367→   367→      return prevEntries.map(entry => {
   368→   368→        if (entry.id !== lastAction.entryId) return entry;
   369→   369→
   370→   370→        const restored = lastAction.previousState;
   371→   371→        const updatedExceptions = entry.exceptions.map(ex =>
   372→   372→          ex.id === lastAction.exceptionId ? restored : ex
   373→   373→        );
   374→   374→
   375→   375→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   376→   376→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   377→   377→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   378→   378→
   379→   379→        const updatedFields = entry.fields.map(f => {
   380→   380→          if (f.exceptionId === lastAction.exceptionId) {
   381→   381→            return {
   382→   382→              ...f,
   383→   383→              value: restored.status === 'Corrected' ? (restored.correctedValue || f.value) : restored.extractedValue,
   384→   384→              isFlagged: restored.status === 'Unresolved',
   385→   385→            };
   386→   386→          }
   387→   387→          return f;
   388→   388→        });
   389→   389→
   390→   390→        const updatedEntry = {
   391→   391→          ...entry,
   392→   392→          status: newStatus,
   393→   393→          exceptions: updatedExceptions,
   394→   394→          fields: updatedFields,
   395→   395→          currentConfidence: newConfidence,
   396→   396→        };
   397→   397→
   398→   398→        if (isSupabaseConfigured() && supabase) {
   399→   399→          supabase.from('exceptions').update({
   400→   400→            status: restored.status,
   401→   401→            corrected_value: restored.correctedValue || null,
   402→   402→            resolved_at: restored.resolvedAt || null,
   403→   403→            resolved_by: restored.resolvedBy || null,
   404→   404→            history: restored.history,
   405→   405→          }).eq('id', lastAction.exceptionId).then(({ error }) => {
   406→   406→            if (error) console.warn('[ctx] undo exception failed:', error.message);
   407→   407→          });
   408→   408→
   409→   409→          supabase.from('shipments').update({
   410→   410→            status: newStatus,
   411→   411→            current_confidence: newConfidence,
   412→   412→            updated_at: new Date().toISOString(),
   413→   413→          }).eq('id', entry.id).then(({ error }) => {
   414→   414→            if (error) console.warn('[ctx] undo shipment failed:', error.message);
   415→   415→          });
   416→   416→        }
   417→   417→
   418→   418→        return updatedEntry;
   419→   419→      });
   420→   420→    });
   421→   421→
   422→   422→    addAuditLog(`Undo applied for last review action on shipment ${lastAction.entryId}`, 'info', lastAction.entryId);
   423→   423→  }, [undoStack, addAuditLog]);
   424→   424→
   425→   425→  // --- Batch accept (uses configured threshold) ---
   426→   426→  const acceptAllHighConfidence = React.useCallback((entryId: string) => {
   427→   427→    setEntries(prevEntries => {
   428→   428→      return prevEntries.map(entry => {
   429→   429→        if (entry.id !== entryId) return entry;
   430→   430→
   431→   431→        const cutoff = rules.invoiceThreshold;
   432→   432→        let acceptedCount = 0;
   433→   433→
   434→   434→        const updatedExceptions = entry.exceptions.map(ex => {
   435→   435→          if (ex.status === 'Unresolved' && ex.confidence >= cutoff) {
   436→   436→            acceptedCount++;
   437→   437→            const historyItem: ExceptionHistoryEntry = {
   438→   438→              user: currentUser,
   439→   439→              oldValue: ex.extractedValue,
   440→   440→              newValue: ex.extractedValue,
   441→   441→              timestamp: new Date().toISOString(),
   442→   442→              action: 'Accepted' as ReviewerAction,
   443→   443→            };
   444→   444→            return {
   445→   445→              ...ex,
   446→   446→              status: 'Accepted' as ExceptionStatus,
   447→   447→              history: [historyItem, ...ex.history],
   448→   448→              resolvedAt: new Date().toISOString(),
   449→   449→              resolvedBy: currentUser,
   450→   450→            };
   451→   451→          }
   452→   452→          return ex;
   453→   453→        });
   454→   454→
   455→   455→        if (acceptedCount === 0) return entry;
   456→   456→
   457→   457→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   458→   458→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   459→   459→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   460→   460→
   461→   461→        const updatedFields = entry.fields.map(f => {
   462→   462→          const exc = updatedExceptions.find(e => e.id === f.exceptionId);
   463→   463→          if (exc) return { ...f, isFlagged: exc.status === 'Unresolved' };
   464→   464→          return f;
   465→   465→        });
   466→   466→
   467→   467→        const updatedEntry = {
   468→   468→          ...entry,
   469→   469→          status: newStatus,
   470→   470→          exceptions: updatedExceptions,
   471→   471→          fields: updatedFields,
   472→   472→          currentConfidence: newConfidence,
   473→   473→        };
   474→   474→
   475→   475→        addAuditLog(
   476→   476→          `Batch action: Approved ${acceptedCount} high-confidence exceptions (≥${cutoff}%) in ${entryId}.`,
   477→   477→          'success',
   478→   478→          entryId
   479→   479→        );

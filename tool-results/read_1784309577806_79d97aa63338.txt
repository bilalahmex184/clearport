   145→   145→// Convert ArrayBuffer to base64 string (UTF-8 safe via chunked encoding)
   146→   146→function bufToBase64(buf: ArrayBuffer): string {
   147→   147→  const bytes = new Uint8Array(buf);
   148→   148→  let binary = "";
   149→   149→  const chunkSize = 0x8000; // 32 KB
   150→   150→  for (let i = 0; i < bytes.length; i += chunkSize) {
   151→   151→    const chunk = bytes.subarray(i, i + chunkSize);
   152→   152→    binary += String.fromCharCode.apply(null, chunk as unknown as number[]);
   153→   153→  }
   154→   154→  return btoa(binary);
   155→   155→}

   157→   157→// Strip ```json fences and trailing prose from Gemini's reply.
   158→   158→// Handles arrays, single objects, and objects with a wrapper like { fields: [...] }.
   159→   159→function extractJsonArray(text: string): any[] {
   160→   160→  if (!text) return [];
   161→   161→  let cleaned = text.trim();
   162→   162→
   163→   163→  // Remove ```json ... ``` fences if present
   164→   164→  const fenceMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/i);
   165→   165→  if (fenceMatch) {
   166→   166→    cleaned = fenceMatch[1].trim();
   167→   167→  }
   168→   168→
   169→   169→  // Try direct parse first (fast path)
   170→   170→  try {
   171→   171→    const direct = JSON.parse(cleaned);
   172→   172→    if (Array.isArray(direct)) return direct;
   173→   173→    if (direct && typeof direct === "object") {
   174→   174→      // Could be a wrapper object
   175→   175→      if (Array.isArray(direct.fields)) return direct.fields;
   176→   176→      if (Array.isArray(direct.data)) return direct.data;
   177→   177→      // Single field object?
   178→   178→      if (direct.field_key) return [direct];
   179→   179→    }
   180→   180→  } catch {
   181→   181→    // Fall through to manual extraction
   182→   182→  }
   183→   183→
   184→   184→  // Find the first '[' and matching last ']'
   185→   185→  const start = cleaned.indexOf("[");
   186→   186→  const end = cleaned.lastIndexOf("]");
   187→   187→  if (start !== -1 && end !== -1 && end > start) {
   188→   188→    const slice = cleaned.slice(start, end + 1);
   189→   189→    try {
   190→   190→      const parsed = JSON.parse(slice);
   191→   191→      return Array.isArray(parsed) ? parsed : [];
   192→   192→    } catch {
   193→   193→      // Fall through
   194→   194→    }
   195→   195→  }
   196→   196→
   197→   197→  // Try finding a JSON object instead
   198→   198→  const objStart = cleaned.indexOf("{");
   199→   199→  const objEnd = cleaned.lastIndexOf("}");
   200→   200→  if (objStart !== -1 && objEnd !== -1 && objEnd > objStart) {
   201→   201→    const slice = cleaned.slice(objStart, objEnd + 1);
   202→   202→    try {
   203→   203→      const parsed = JSON.parse(slice);
   204→   204→      if (Array.isArray(parsed)) return parsed;
   205→   205→      if (parsed && typeof parsed === "object") {
   206→   206→        if (Array.isArray(parsed.fields)) return parsed.fields;
   207→   207→        if (parsed.field_key) return [parsed];
   208→   208→      }
   209→   209→    } catch {
   210→   210→      // Give up
   211→   211→    }
   212→   212→  }
   213→   213→
   214→   214→  return [];
   215→   215→}

   281→   281→// Call Gemini with the file bytes. Returns an array of field objects.
   282→   282→// Handles both binary files (via inlineData) and text files (via text prompt).
   283→   283→// Returns { fields, debug, model, rawResponse } so the caller can surface
   284→   284→// diagnostic info and persist the raw AI response for audit.
   285→   285→//
   286→   286→// MODEL CASCADE (most accurate first):
   287→   287→//   gemini-2.5-pro  → gemini-2.0-flash → gemini-1.5-flash → regex fallback
   288→   288→//
   289→   289→// NOTE: Gemini handles multi-page PDFs natively via inlineData — the entire
   290→   290→// PDF is sent as a single inlineData part and the model reads every page.
   291→   291→// No special chunking or per-page calls are required.
   292→   292→async function callGeminiExtraction(
   293→   293→  ai: GoogleGenAI,
   294→   294→  mimeType: string,
   295→   295→  base64Data: string,
   296→   296→  rawText?: string
   297→   297→): Promise<{ fields: any[]; debug: any; model: string | null; rawResponse: string | null }> {
   298→   298→  // Ordered model cascade — Pro first (most accurate), then Flash variants.
   299→   299→  const models = ["gemini-2.5-pro", "gemini-2.0-flash", "gemini-1.5-flash"];
   300→   300→  const debug: any = { modelsTried: [], errors: [], rawResponses: [], retries: [] };
   301→   301→
   302→   302→  // Strict, deterministic config — temperature 0 + topP 0 + topK 1 means the
   303→   303→  // model always picks the single highest-probability token, ensuring that
   304→   304→  // the same document yields the same extraction on every run.
   305→   305→  const generationConfig = { temperature: 0, topP: 0, topK: 1 };
   306→   306→
   307→   307→  for (const model of models) {
   308→   308→    debug.modelsTried.push(model);
   309→   309→    let lastErr: any = null;
   310→   310→
   311→   311→    // Retry up to 2 times on transient errors (429 rate-limit, 503 server
   312→   312→    // error). Exponential backoff: 1s, then 2s.
   313→   313→    for (let attempt = 0; attempt <= 2; attempt++) {
   314→   314→      try {
   315→   315→        const parts: any[] = [];
   316→   316→
   317→   317→        if (rawText && rawText.length > 0) {
   318→   318→          parts.push({ text: `Document text content:\n\n${rawText}\n\n---\n${GEMINI_PROMPT}` });
   319→   319→        } else {
   320→   320→          parts.push({ inlineData: { mimeType, data: base64Data } });
   321→   321→          parts.push({ text: GEMINI_PROMPT });
   322→   322→        }
   323→   323→
   324→   324→        // 30-second timeout — if the model is slow or hung, abort and fall
   325→   325→        // back to the next model in the cascade.
   326→   326→        const response = await Promise.race([
   327→   327→          ai.models.generateContent({
   328→   328→            model,
   329→   329→            contents: [{ role: "user", parts }],
   330→   330→            config: generationConfig,
   331→   331→          }),
   332→   332→          new Promise<never>((_, reject) =>
   333→   333→            setTimeout(() => reject(new Error(`TIMEOUT_15s after 15000ms`)), 15000)
   334→   334→          ),
   335→   335→        ]);
   336→   336→
   337→   337→        // Try multiple ways to get text from the response
   338→   338→        let text = "";
   339→   339→        if (typeof response.text === "string") {
   340→   340→          text = response.text;
   341→   341→        } else if (typeof response.text === "function") {
   342→   342→          text = response.text();
   343→   343→        } else if (response?.candidates?.[0]?.content?.parts?.[0]?.text) {
   344→   344→          text = response.candidates[0].content.parts[0].text;
   345→   345→        } else if (response?.response?.text) {
   346→   346→          text = response.response.text;
   347→   347→        }
   348→   348→
   349→   349→        debug.rawResponses.push({ model, attempt, textLength: text.length, preview: text.substring(0, 300) });
   350→   350→
   351→   351→        const parsed = extractJsonArray(text);
   352→   352→        if (parsed.length > 0) {
   353→   353→          debug.success = true;
   354→   354→          debug.successModel = model;
   355→   355→          return { fields: parsed, debug, model, rawResponse: text };
   356→   356→        } else {
   357→   357→          debug.errors.push(`${model}: 0 parseable fields from ${text.length} chars`);
   358→   358→          lastErr = new Error(`0 parseable fields from ${text.length} chars`);
   359→   359→          // Non-retryable: model responded but produced nothing useful.
   360→   360→          // Move on to the next model.
   361→   361→          break;
   362→   362→        }
   363→   363→      } catch (err: any) {
   364→   364→        const errMsg = err?.message || String(err);
   365→   365→        lastErr = err;
   366→   366→        debug.errors.push(`${model} (attempt ${attempt + 1}): ${errMsg}`);
   367→   367→        debug.rawResponses.push({ model, attempt, error: errMsg });
   368→   368→
   369→   369→        const isRetryable =
   370→   370→          /429/.test(errMsg) ||            // rate limit
   371→   371→          /503/.test(errMsg) ||            // server error
   372→   372→          /overloaded/i.test(errMsg) ||
   373→   373→          /rate.?limit/i.test(errMsg) ||
   374→   374→          /SERVICE_UNAVAILABLE/i.test(errMsg) ||
   375→   375→          /RESOURCE_EXHAUSTED/i.test(errMsg);
   376→   376→
   377→   377→        // If quota exhausted (429), skip ALL remaining models — they share the same quota.
   378→   378→        // Go straight to regex fallback to avoid wasting 15+ seconds on retries.
   379→   379→        const isQuotaExhausted = /429/.test(errMsg) || /RESOURCE_EXHAUSTED/i.test(errMsg) || /quota/i.test(errMsg);
   380→   380→        if (isQuotaExhausted) {
   381→   381→          debug.errors.push(`${model}: quota exhausted — skipping remaining models, using regex fallback`);
   382→   382→          debug.quotaExhausted = true;
   383→   383→          return { fields: [], debug, model: null, rawResponse: null };
   384→   384→        }
   385→   385→
   386→   386→        if (isRetryable && attempt < 1) {  // Reduced from 2 to 1 retry
   387→   387→          const delay = 1000; // Fixed 1s delay (was exponential)
   388→   388→          debug.retries.push({ model, attempt: attempt + 1, delayMs: delay, reason: errMsg });
   389→   389→          await new Promise((r) => setTimeout(r, delay));
   390→   390→          continue;
   391→   391→        }
   392→   392→
   393→   393→        // Non-retryable error (incl. TIMEOUT_30s) — break out of retry loop
   394→   394→        // and fall through to the next model.
   395→   395→        if (errMsg.startsWith("TIMEOUT_15s")) {
   396→   396→          debug.errors.push(`${model}: timed out after 30s, falling back to next model`);
   397→   397→        }
   398→   398→        break;
   399→   399→      }
   400→   400→    }
   401→   401→    if (lastErr) {
   402→   402→      // Already logged — continue to next model.
   403→   403→    }
   404→   404→  }
   405→   405→
   406→   406→  return { fields: [], debug, model: null, rawResponse: null };
   407→   407→}

   411→   411→// Sanitize a currency string: "$52,150.75" → "52150.75" (numeric only)
   412→   412→function sanitizeCurrency(val: string): { display: string; numeric: number | null } {
   413→   413→  const display = val.trim();
   414→   414→  const cleaned = display.replace(/[^0-9.]/g, "");
   415→   415→  const numeric = cleaned ? parseFloat(cleaned) : null;
   416→   416→  return { display, numeric: numeric && Number.isFinite(numeric) ? numeric : null };
   417→   417→}

   419→   419→// Normalize UTF-8 foreign characters for ASCII-only compliance systems.
   420→   420→function normalizeUtf8(val: string): { utf8: string; ascii: string } {
   421→   421→  const utf8 = val.trim();
   422→   422→  const ascii = utf8
   423→   423→    .replace(/ä/g, "ae").replace(/ö/g, "oe").replace(/ü/g, "ue")
   424→   424→    .replace(/Ä/g, "Ae").replace(/Ö/g, "Oe").replace(/Ü/g, "Ue")
   425→   425→    .replace(/ß/g, "ss")
   426→   426→    .replace(/é/g, "e").replace(/è/g, "e").replace(/ê/g, "e").replace(/ë/g, "e")
   427→   427→    .replace(/á/g, "a").replace(/à/g, "a").replace(/â/g, "a")
   428→   428→    .replace(/í/g, "i").replace(/ì/g, "i").replace(/î/g, "i")
   429→   429→    .replace(/ó/g, "o").replace(/ò/g, "o").replace(/ô/g, "o")
   430→   430→    .replace(/ú/g, "u").replace(/ù/g, "u").replace(/û/g, "u")
   431→   431→    .replace(/ñ/g, "n").replace(/ç/g, "c")
   432→   432→    .replace(/[^\x00-\x7F]/g, "")
   433→   433→    .trim();
   434→   434→  return { utf8, ascii };
   435→   435→}

   437→   437→// Parse multi-line table rows. Groups secondary lines with preceding line item.
   438→   438→function parseTableRows(text: string): { lineItems: any[]; totalValue: string | null } {
   439→   439→  const lines = text.split(/\n/).map(l => l.trim()).filter(l => l.length > 0);
   440→   440→  const lineItems: any[] = [];
   441→   441→  let totalValue: string | null = null;
   442→   442→
   443→   443→  const itemPattern = /^(.+?)\s+(\d+)\s+(\d{4}\.\d{2}\.\d{4}|\d{8,})\s+([$€£¥]?[\d,]+\.?\d*)\s*$/i;
   444→   444→  const totalPattern = /^(?:total|grand\s*total|invoice\s*total|total\s*value)\s*[:\-]?\s*([$€£¥][\d,]+\.?\d*)/i;
   445→   445→
   446→   446→  for (const line of lines) {
   447→   447→    const totalMatch = line.match(totalPattern);
   448→   448→    if (totalMatch && totalMatch[1]) {
   449→   449→      totalValue = totalMatch[1].trim();
   450→   450→      continue;
   451→   451→    }
   452→   452→    const itemMatch = line.match(itemPattern);
   453→   453→    if (itemMatch) {
   454→   454→      const [, desc, qty, hts, value] = itemMatch;
   455→   455→      lineItems.push({ description: desc.trim(), qty: parseInt(qty), htsCode: hts.trim(), value: value.trim(), secondaryLines: [] as string[] });
   456→   456→      continue;
   457→   457→    }
   458→   458→    const isSecondary = /^(shipping\s*(?:cost|fee)?|insurance(?:\s*cost)?|freight|handling|duty|tax|discount|subtotal|other)\s*[:\-]?\s*([$€£¥]?[\d,]+\.?\d*)?/i.test(line);
   459→   459→    if (isSecondary && lineItems.length > 0) {
   460→   460→      lineItems[lineItems.length - 1].secondaryLines.push(line);
   461→   461→    }
   462→   462→  }
   463→   463→  return { lineItems, totalValue };
   464→   464→}

   466→   466→// --- CSV Parser ----------------------------------------------------------------
   467→   467→// Parse CSV content as structured fields. Expects rows with headers or key-value pairs.
   468→   468→function parseCSV(text: string): any[] {
   469→   469→  const fields: any[] = [];
   470→   470→  const lines = text.split(/\n/).map(l => l.trim()).filter(l => l.length > 0);
   471→   471→  if (lines.length === 0) return fields;
   472→   472→
   473→   473→  // Detect delimiter (comma, semicolon, tab)
   474→   474→  const firstLine = lines[0];
   475→   475→  const delimiter = firstLine.includes('\t') ? '\t' : firstLine.includes(';') ? ';' : ',';
   476→   476→
   477→   477→  // Check if it's a header-based CSV
   478→   478→  const headers = firstLine.split(delimiter).map(h => h.trim().toLowerCase().replace(/[^a-z0-9]/g, ''));
   479→   479→  
   480→   480→  // Map CSV headers to our field keys
   481→   481→  const headerMap: Record<string, string> = {
   482→   482→    invoice: 'invoiceNo', invoicenumber: 'invoiceNo', invoiceno: 'invoiceNo', invoicenum: 'invoiceNo',
   483→   483→    date: 'invoiceDate', invoicedate: 'invoiceDate',
   484→   484→    shipper: 'shipper', exporter: 'shipper', shipperexporter: 'shipper',
   485→   485→    consignee: 'consignee', importer: 'consignee', consigneeimporter: 'consignee',
   486→   486→    value: 'declaredValue', totalvalue: 'declaredValue', declaredvalue: 'declaredValue', total: 'declaredValue', amount: 'declaredValue',
   487→   487→    hts: 'htsCode', htscode: 'htsCode', tariff: 'htsCode', hscode: 'htsCode',
   488→   488→    weight: 'netWeight', netweight: 'netWeight', net: 'netWeight',
   489→   489→    grossweight: 'grossWeight', gross: 'grossWeight',
   490→   490→    origin: 'countryOfOrigin', countryoforigin: 'countryOfOrigin', country: 'countryOfOrigin',
   491→   491→    carrier: 'carrier',
   492→   492→    port: 'portOfEntry', portofentry: 'portOfEntry',
   493→   493→    bol: 'billOfLading', billoflading: 'billOfLading',
   494→   494→    address: 'consigneeAddress', consigneeaddress: 'consigneeAddress',
   495→   495→  };
   496→   496→
   497→   497→  // If first line looks like headers
   498→   498→  const hasHeaders = headers.some(h => headerMap[h]);
   499→   499→  
   500→   500→  if (hasHeaders && lines.length >= 2) {
   501→   501→    // Header-based CSV: use first data row
   502→   502→    const values = lines[1].split(delimiter).map(v => v.trim().replace(/^["']|["']$/g, ''));
   503→   503→    headers.forEach((header, idx) => {
   504→   504→      const fieldKey = headerMap[header];
   505→   505→      if (fieldKey && values[idx]) {
   506→   506→        const def = FIELD_DEFINITIONS.find(d => d.key === fieldKey);
   507→   507→        fields.push({
   508→   508→          field_key: fieldKey,
   509→   509→          field_label: def?.label || fieldKey,
   510→   510→          extracted_value: values[idx],
   511→   511→          confidence: 85,
   512→   512→        });
   513→   513→      }
   514→   514→    });
   515→   515→  } else {
   516→   516→    // Key-value CSV: "Invoice Number,INV-001" format
   517→   517→    for (const line of lines) {
   518→   518→      const parts = line.split(delimiter).map(p => p.trim().replace(/^["']|["']$/g, ''));
   519→   519→      if (parts.length >= 2) {
   520→   520→        const key = parts[0].toLowerCase().replace(/[^a-z0-9]/g, '');
   521→   521→        const value = parts[1];
   522→   522→        const fieldKey = headerMap[key];
   523→   523→        if (fieldKey && value) {
   524→   524→          const def = FIELD_DEFINITIONS.find(d => d.key === fieldKey);
   525→   525→          fields.push({
   526→   526→            field_key: fieldKey,
   527→   527→            field_label: def?.label || fieldKey,
   528→   528→            extracted_value: value,
   529→   529→            confidence: 80,
   530→   530→          });
   531→   531→        }
   532→   532→      }
   533→   533→    }
   534→   534→  }
   535→   535→
   536→   536→  return fields;
   537→   537→}

   789→   789→        fields.push({ field_key: "countryOfOrigin", field_label: "Country of Origin", extracted_value: countryMatch[1], confidence: 60 });
   790→   790→      }
   791→   791→    }
   792→   792→
   793→   793→    // Look for bare weight
   794→   794→    if (!fields.find(f => f.field_key === "netWeight")) {
   795→   795→      const weightMatch = fullText.match(/(\d[\d,]*\.?\d*\s*(?:lbs?|kg|kgs|pounds?))/i);
   796→   796→      if (weightMatch) {
   797→   797→        fields.push({ field_key: "netWeight", field_label: "Net Weight", extracted_value: weightMatch[1], confidence: 65 });
   798→   798→      }
   799→   799→    }
   800→   800→
   801→   801→    // Look for company name (first line that's not a keyword/number)
   802→   802→    if (!fields.find(f => f.field_key === "shipper")) {
   803→   803→      for (const line of lines) {
   804→   804→        const trimmed = line.trim();
   805→   805→        // Skip empty lines, numbers, dates, currency values, keywords
   806→   806→        if (trimmed.length < 3 || trimmed.length > 60) continue;
   807→   807→        if (/^\d/.test(trimmed)) continue;
   808→   808→        if (/[$€£¥]/.test(trimmed)) continue;
   809→   809→        if (/^(invoice|inv|date|shipper|consignee|total|hts|net|gross|country|carrier|port|bill)/i.test(trimmed)) continue;
   810→   810→        if (/^(INV|SHIP|DOC)/i.test(trimmed)) continue;
   811→   811→        // Looks like a company name
   812→   812→        if (/^[A-Z][a-zA-Z\s&.,]+$/.test(trimmed)) {
   813→   813→          fields.push({ field_key: "shipper", field_label: "Shipper/Exporter", extracted_value: trimmed, confidence: 55 });
   814→   814→          break;
   815→   815→        }
   816→   816→      }
   817→   817→    }
   818→   818→  }
   819→   819→
   820→   820→  return fields;
   821→   821→}

   823→   823→// --- PDF text-layer extraction (Tier 2) ---------------------------------------
   824→   824→// Extracts embedded text from PDF files that have a real text layer
   825→   825→// (not scanned images). Uses a simple regex-based approach to find text
   826→   826→// between BT/ET markers in the PDF content stream.
   827→   827→function extractPdfTextLayer(arrayBuffer: ArrayBuffer): string | null {
   828→   828→  try {
   829→   829→    const bytes = new Uint8Array(arrayBuffer);
   830→   830→    // Check if it's actually a PDF
   831→   831→    const header = new TextDecoder().decode(bytes.slice(0, 5));
   832→   832→    if (header !== '%PDF-') return null;
   833→   833→
   834→   834→    // Extract text from PDF content streams
   835→   835→    // Look for text in parentheses within BT...ET blocks
   836→   836→    const text = new TextDecoder('latin1').decode(bytes);
   837→   837→    const textMatches: string[] = [];
   838→   838→
   839→   839→    // Match text in parentheses (PDF text objects)
   840→   840→    const textRegex = /\(([^)]+)\)/g;
   841→   841→    let match;
   842→   842→    while ((match = textRegex.exec(text)) !== null) {
   843→   843→      const t = match[1];
   844→   844→      // Filter out non-text content (binary data, short strings)
   845→   845→      if (t.length > 2 && /[a-zA-Z]/.test(t) && !/\\[nrt]/.test(t)) {
   846→   846→        textMatches.push(t);
   847→   847→      }
   848→   848→    }
   849→   849→
   850→   850→    if (textMatches.length < 3) return null; // Not enough text to be useful
   851→   851→
   852→   852→    // Join and clean up
   853→   853→    const result = textMatches.join('\n');
   854→   854→    return result.length > 50 ? result : null;
   855→   855→  } catch {
   856→   856→    return null;
   857→   857→  }
   858→   858→}

    76→const GEMINI_PROMPT = `You are a strict customs document OCR engine. Extract structured fields from this document.
    77→
    78→STRICT OUTPUT CONTRACT — read carefully:
    79→1. Return ONLY a JSON array. Do NOT wrap it in markdown fences. Do NOT add any prose, explanation, or commentary before or after the array.
    80→2. Every object in the array MUST have EXACTLY these four keys (no more, no less):
    81→   - "field_key": string (one of the allowed keys listed below)
    82→   - "field_label": string (the human-readable label listed below)
    83→   - "extracted_value": string (the verbatim value read from the document) OR null
    84→   - "confidence": integer between 0 and 100
    85→3. If a field is NOT present in the document, OMIT it from the array entirely. Do NOT include it with a null value.
    86→4. If you cannot clearly read a value, OMIT that field. Do NOT guess, infer, hallucinate, or fabricate values. Only extract fields that are clearly visible in the document.
    87→5. Confidence must reflect your actual certainty about BOTH the field's presence AND the accuracy of the extracted value:
    88→   - 95-100: very clear, unambiguous, legible text, no possible doubt
    89→   - 80-94:  clear, but minor ambiguity (e.g. partial obscuring, slight blur)
    90→   - 60-79:  somewhat clear, value readable but with some uncertainty
    91→   - below 60: uncertain — you should usually OMIT the field rather than return it
    92→6. Use the EXACT field_key strings below (case-sensitive). Do not invent new keys.
    93→
    94→SPECIAL HANDLING RULES:
    95→
    96→A. MULTI-LINE TABLE ROWS:
    97→   - Commercial invoices often have line-item tables where a single item spans multiple rows.
    98→   - Secondary rows under a line item (e.g. "Shipping Cost", "Insurance", "Handling Fee") belong to the PRECEDING line item, NOT as separate items.
    99→   - If a table row has no QTY or HS Code, it is a secondary row — merge it with the line item above.
   100→   - Extract only the PRIMARY line item's HTS Code (field_key: "htsCode"), not secondary rows.
   101→
   102→B. CURRENCY & NUMBERS:
   103→   - Always include the currency symbol in extracted_value (e.g. "$52,150.75", not "52150.75").
   104→   - Accept $, €, £, ¥ as valid currency prefixes.
   105→   - If the total is labeled "Total", "Grand Total", "Invoice Total", or "Total Declared Value", extract it as field_key "declaredValue".
   106→
   107→C. FOREIGN CHARACTERS (UTF-8):
   108→   - Documents may contain foreign characters: German ß, ü, ö, ä; French é, è; Spanish ñ; etc.
   109→   - Preserve the original UTF-8 characters in extracted_value (e.g. "Industriestraße 14" stays as-is).
   110→   - Do NOT transliterate or strip foreign characters — the system handles UTF-8 normalization downstream.
   111→   - If a name contains special characters, extract it verbatim.

   113→D. BOUNDING BOX ISOLATION — "OFFICIAL CBP USE" AREA:
   114→   - Many customs documents have an "Official CBP Use Only" box (usually bottom-left or bottom-right).

   116→   - Do NOT extract signatures, stamps, or handwriting from the "Official CBP Use" box.

   145→// Convert ArrayBuffer to base64 string (UTF-8 safe via chunked encoding)
   146→function bufToBase64(buf: ArrayBuffer): string {
   147→  const bytes = new Uint8Array(buf);
   148→  let binary = "";
   149→  const chunkSize = 0x8000; // 32 KB
   150→  for (let i = 0; i < bytes.length; i += chunkSize) {
   151→    const chunk = bytes.subarray(i, i + chunkSize);
   152→    binary += String.fromCharCode.apply(null, chunk as unknown as number[]);
   153→  }
   154→  return btoa(binary);
   155→}

   157→// Strip ```json fences and trailing prose from Gemini's reply.
   158→// Handles arrays, single objects, and objects with a wrapper like { fields: [...] }.
   159→function extractJsonArray(text: string): any[] {
   160→  if (!text) return [];
   161→  let cleaned = text.trim();
   162→
   163→  // Remove ```json ... ``` fences if present
   164→  const fenceMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/i);
   165→  if (fenceMatch) {
   166→    cleaned = fenceMatch[1].trim();
   167→  }
   168→
   169→  // Try direct parse first (fast path)
   170→  try {
   171→    const direct = JSON.parse(cleaned);
   172→    if (Array.isArray(direct)) return direct;
   173→    if (direct && typeof direct === "object") {
   174→      // Could be a wrapper object
   175→      if (Array.isArray(direct.fields)) return direct.fields;
   176→      if (Array.isArray(direct.data)) return direct.data;
   177→      // Single field object?
   178→      if (direct.field_key) return [direct];
   179→    }
   180→  } catch {
   181→    // Fall through to manual extraction
   182→  }
   183→
   184→  // Find the first '[' and matching last ']'
   185→  const start = cleaned.indexOf("[");
   186→  const end = cleaned.lastIndexOf("]");
   187→  if (start !== -1 && end !== -1 && end > start) {
   188→    const slice = cleaned.slice(start, end + 1);
   189→    try {
   190→      const parsed = JSON.parse(slice);
   191→      return Array.isArray(parsed) ? parsed : [];
   192→    } catch {
   193→      // Fall through
   194→    }
   195→  }
   196→
   197→  // Try finding a JSON object instead
   198→  const objStart = cleaned.indexOf("{");
   199→  const objEnd = cleaned.lastIndexOf("}");
   200→  if (objStart !== -1 && objEnd !== -1 && objEnd > objStart) {
   201→    const slice = cleaned.slice(objStart, objEnd + 1);
   202→    try {
   203→      const parsed = JSON.parse(slice);
   204→      if (Array.isArray(parsed)) return parsed;
   205→      if (parsed && typeof parsed === "object") {
   206→        if (Array.isArray(parsed.fields)) return parsed.fields;
   207→        if (parsed.field_key) return [parsed];
   208→      }
   209→    } catch {
   210→      // Give up
   211→    }
   212→  }
   213→
   214→  return [];
   215→}

   281→// Call Gemini with the file bytes. Returns an array of field objects.
   282→// Handles both binary files (via inlineData) and text files (via text prompt).
   283→// Returns { fields, debug, model, rawResponse } so the caller can surface
   284→// diagnostic info and persist the raw AI response for audit.
   285→//
   286→// MODEL CASCADE (most accurate first):
   287→//   gemini-2.5-pro  → gemini-2.0-flash → gemini-1.5-flash → regex fallback
   288→//
   289→// NOTE: Gemini handles multi-page PDFs natively via inlineData — the entire
   290→// PDF is sent as a single inlineData part and the model reads every page.
   291→// No special chunking or per-page calls are required.
   292→async function callGeminiExtraction(
   293→  ai: GoogleGenAI,
   294→  mimeType: string,
   295→  base64Data: string,
   296→  rawText?: string
   297→): Promise<{ fields: any[]; debug: any; model: string | null; rawResponse: string | null }> {
   298→  // Ordered model cascade — Pro first (most accurate), then Flash variants.
   299→  const models = ["gemini-2.5-pro", "gemini-2.0-flash", "gemini-1.5-flash"];
   300→  const debug: any = { modelsTried: [], errors: [], rawResponses: [], retries: [] };
   301→
   302→  // Strict, deterministic config — temperature 0 + topP 0 + topK 1 means the
   303→  // model always picks the single highest-probability token, ensuring that
   304→  // the same document yields the same extraction on every run.
   305→  const generationConfig = { temperature: 0, topP: 0, topK: 1 };
   306→
   307→  for (const model of models) {
   308→    debug.modelsTried.push(model);
   309→    let lastErr: any = null;
   310→
   311→    // Retry up to 2 times on transient errors (429 rate-limit, 503 server
   312→    // error). Exponential backoff: 1s, then 2s.
   313→    for (let attempt = 0; attempt <= 2; attempt++) {
   314→      try {
   315→        const parts: any[] = [];
   316→
   317→        if (rawText && rawText.length > 0) {
   318→          parts.push({ text: `Document text content:\n\n${rawText}\n\n---\n${GEMINI_PROMPT}` });
   319→        } else {
   320→          parts.push({ inlineData: { mimeType, data: base64Data } });
   321→          parts.push({ text: GEMINI_PROMPT });
   322→        }
   323→
   324→        // 30-second timeout — if the model is slow or hung, abort and fall
   325→        // back to the next model in the cascade.
   326→        const response = await Promise.race([
   327→          ai.models.generateContent({
   328→            model,
   329→            contents: [{ role: "user", parts }],
   330→            config: generationConfig,
   331→          }),
   332→          new Promise<never>((_, reject) =>
   333→            setTimeout(() => reject(new Error(`TIMEOUT_15s after 15000ms`)), 15000)
   334→          ),
   335→        ]);
   336→
   337→        // Try multiple ways to get text from the response
   338→        let text = "";
   339→        if (typeof response.text === "string") {
   340→          text = response.text;
   341→        } else if (typeof response.text === "function") {
   342→          text = response.text();
   343→        } else if (response?.candidates?.[0]?.content?.parts?.[0]?.text) {
   344→          text = response.candidates[0].content.parts[0].text;
   345→        } else if (response?.response?.text) {
   346→          text = response.response.text;
   347→        }
   348→
   349→        debug.rawResponses.push({ model, attempt, textLength: text.length, preview: text.substring(0, 300) });
   350→
   351→        const parsed = extractJsonArray(text);
   352→        if (parsed.length > 0) {
   353→          debug.success = true;
   354→          debug.successModel = model;
   355→          return { fields: parsed, debug, model, rawResponse: text };
   356→        } else {
   357→          debug.errors.push(`${model}: 0 parseable fields from ${text.length} chars`);
   358→          lastErr = new Error(`0 parseable fields from ${text.length} chars`);
   359→          // Non-retryable: model responded but produced nothing useful.
   360→          // Move on to the next model.
   361→          break;
   362→        }
   363→      } catch (err: any) {
   364→        const errMsg = err?.message || String(err);
   365→        lastErr = err;
   366→        debug.errors.push(`${model} (attempt ${attempt + 1}): ${errMsg}`);
   367→        debug.rawResponses.push({ model, attempt, error: errMsg });
   368→
   369→        const isRetryable =
   370→          /429/.test(errMsg) ||            // rate limit
   371→          /503/.test(errMsg) ||            // server error
   372→          /overloaded/i.test(errMsg) ||
   373→          /rate.?limit/i.test(errMsg) ||
   374→          /SERVICE_UNAVAILABLE/i.test(errMsg) ||
   375→          /RESOURCE_EXHAUSTED/i.test(errMsg);
   376→
   377→        // If quota exhausted (429), skip ALL remaining models — they share the same quota.
   378→        // Go straight to regex fallback to avoid wasting 15+ seconds on retries.
   379→        const isQuotaExhausted = /429/.test(errMsg) || /RESOURCE_EXHAUSTED/i.test(errMsg) || /quota/i.test(errMsg);
   380→        if (isQuotaExhausted) {
   381→          debug.errors.push(`${model}: quota exhausted — skipping remaining models, using regex fallback`);
   382→          debug.quotaExhausted = true;
   383→          return { fields: [], debug, model: null, rawResponse: null };
   384→        }
   385→
   386→        if (isRetryable && attempt < 1) {  // Reduced from 2 to 1 retry
   387→          const delay = 1000; // Fixed 1s delay (was exponential)
   388→          debug.retries.push({ model, attempt: attempt + 1, delayMs: delay, reason: errMsg });
   389→          await new Promise((r) => setTimeout(r, delay));
   390→          continue;
   391→        }
   392→
   393→        // Non-retryable error (incl. TIMEOUT_30s) — break out of retry loop
   394→        // and fall through to the next model.
   395→        if (errMsg.startsWith("TIMEOUT_15s")) {
   396→          debug.errors.push(`${model}: timed out after 30s, falling back to next model`);
   397→        }
   398→        break;
   399→      }
   400→    }
   401→    if (lastErr) {
   402→      // Already logged — continue to next model.
   403→    }
   404→  }
   405→
   406→  return { fields: [], debug, model: null, rawResponse: null };
   407→}

   411→// Sanitize a currency string: "$52,150.75" → "52150.75" (numeric only)
   412→function sanitizeCurrency(val: string): { display: string; numeric: number | null } {
   413→  const display = val.trim();
   414→  const cleaned = display.replace(/[^0-9.]/g, "");
   415→  const numeric = cleaned ? parseFloat(cleaned) : null;
   416→  return { display, numeric: numeric && Number.isFinite(numeric) ? numeric : null };
   417→}

   419→// Normalize UTF-8 foreign characters for ASCII-only compliance systems.
   420→function normalizeUtf8(val: string): { utf8: string; ascii: string } {
   421→  const utf8 = val.trim();
   422→  const ascii = utf8
   423→    .replace(/ä/g, "ae").replace(/ö/g, "oe").replace(/ü/g, "ue")
   424→    .replace(/Ä/g, "Ae").replace(/Ö/g, "Oe").replace(/Ü/g, "Ue")
   425→    .replace(/ß/g, "ss")
   426→    .replace(/é/g, "e").replace(/è/g, "e").replace(/ê/g, "e").replace(/ë/g, "e")
   427→    .replace(/á/g, "a").replace(/à/g, "a").replace(/â/g, "a")
   428→    .replace(/í/g, "i").replace(/ì/g, "i").replace(/î/g, "i")
   429→    .replace(/ó/g, "o").replace(/ò/g, "o").replace(/ô/g, "o")
   430→    .replace(/ú/g, "u").replace(/ù/g, "u").replace(/û/g, "u")
   431→    .replace(/ñ/g, "n").replace(/ç/g, "c")
   432→    .replace(/[^\x00-\x7F]/g, "")
   433→    .trim();
   434→  return { utf8, ascii };
   435→}

   437→// Parse multi-line table rows. Groups secondary lines with preceding line item.
   438→function parseTableRows(text: string): { lineItems: any[]; totalValue: string | null } {
   439→  const lines = text.split(/\n/).map(l => l.trim()).filter(l => l.length > 0);
   440→  const lineItems: any[] = [];
   441→  let totalValue: string | null = null;
   442→
   443→  const itemPattern = /^(.+?)\s+(\d+)\s+(\d{4}\.\d{2}\.\d{4}|\d{8,})\s+([$€£¥]?[\d,]+\.?\d*)\s*$/i;
   444→  const totalPattern = /^(?:total|grand\s*total|invoice\s*total|total\s*value)\s*[:\-]?\s*([$€£¥][\d,]+\.?\d*)/i;
   445→
   446→  for (const line of lines) {
   447→    const totalMatch = line.match(totalPattern);
   448→    if (totalMatch && totalMatch[1]) {
   449→      totalValue = totalMatch[1].trim();
   450→      continue;
   451→    }
   452→    const itemMatch = line.match(itemPattern);
   453→    if (itemMatch) {
   454→      const [, desc, qty, hts, value] = itemMatch;
   455→      lineItems.push({ description: desc.trim(), qty: parseInt(qty), htsCode: hts.trim(), value: value.trim(), secondaryLines: [] as string[] });
   456→      continue;
   457→    }
   458→    const isSecondary = /^(shipping\s*(?:cost|fee)?|insurance(?:\s*cost)?|freight|handling|duty|tax|discount|subtotal|other)\s*[:\-]?\s*([$€£¥]?[\d,]+\.?\d*)?/i.test(line);
   459→    if (isSecondary && lineItems.length > 0) {
   460→      lineItems[lineItems.length - 1].secondaryLines.push(line);
   461→    }
   462→  }
   463→  return { lineItems, totalValue };
   464→}

   466→// --- CSV Parser ----------------------------------------------------------------
   467→// Parse CSV content as structured fields. Expects rows with headers or key-value pairs.
   468→function parseCSV(text: string): any[] {
   469→  const fields: any[] = [];
   470→  const lines = text.split(/\n/).map(l => l.trim()).filter(l => l.length > 0);
   471→  if (lines.length === 0) return fields;
   472→
   473→  // Detect delimiter (comma, semicolon, tab)
   474→  const firstLine = lines[0];
   475→  const delimiter = firstLine.includes('\t') ? '\t' : firstLine.includes(';') ? ';' : ',';
   476→
   477→  // Check if it's a header-based CSV
   478→  const headers = firstLine.split(delimiter).map(h => h.trim().toLowerCase().replace(/[^a-z0-9]/g, ''));
   479→  
   480→  // Map CSV headers to our field keys
   481→  const headerMap: Record<string, string> = {
   482→    invoice: 'invoiceNo', invoicenumber: 'invoiceNo', invoiceno: 'invoiceNo', invoicenum: 'invoiceNo',
   483→    date: 'invoiceDate', invoicedate: 'invoiceDate',
   484→    shipper: 'shipper', exporter: 'shipper', shipperexporter: 'shipper',
   485→    consignee: 'consignee', importer: 'consignee', consigneeimporter: 'consignee',
   486→    value: 'declaredValue', totalvalue: 'declaredValue', declaredvalue: 'declaredValue', total: 'declaredValue', amount: 'declaredValue',
   487→    hts: 'htsCode', htscode: 'htsCode', tariff: 'htsCode', hscode: 'htsCode',
   488→    weight: 'netWeight', netweight: 'netWeight', net: 'netWeight',
   489→    grossweight: 'grossWeight', gross: 'grossWeight',
   490→    origin: 'countryOfOrigin', countryoforigin: 'countryOfOrigin', country: 'countryOfOrigin',
   491→    carrier: 'carrier',
   492→    port: 'portOfEntry', portofentry: 'portOfEntry',
   493→    bol: 'billOfLading', billoflading: 'billOfLading',
   494→    address: 'consigneeAddress', consigneeaddress: 'consigneeAddress',
   495→  };
   496→
   497→  // If first line looks like headers
   498→  const hasHeaders = headers.some(h => headerMap[h]);
   499→  
   500→  if (hasHeaders && lines.length >= 2) {
   501→    // Header-based CSV: use first data row
   502→    const values = lines[1].split(delimiter).map(v => v.trim().replace(/^["']|["']$/g, ''));
   503→    headers.forEach((header, idx) => {
   504→      const fieldKey = headerMap[header];
   505→      if (fieldKey && values[idx]) {
   506→        const def = FIELD_DEFINITIONS.find(d => d.key === fieldKey);
   507→        fields.push({
   508→          field_key: fieldKey,
   509→          field_label: def?.label || fieldKey,
   510→          extracted_value: values[idx],
   511→          confidence: 85,
   512→        });
   513→      }
   514→    });
   515→  } else {
   516→    // Key-value CSV: "Invoice Number,INV-001" format
   517→    for (const line of lines) {
   518→      const parts = line.split(delimiter).map(p => p.trim().replace(/^["']|["']$/g, ''));
   519→      if (parts.length >= 2) {
   520→        const key = parts[0].toLowerCase().replace(/[^a-z0-9]/g, '');
   521→        const value = parts[1];
   522→        const fieldKey = headerMap[key];
   523→        if (fieldKey && value) {
   524→          const def = FIELD_DEFINITIONS.find(d => d.key === fieldKey);
   525→          fields.push({
   526→            field_key: fieldKey,
   527→            field_label: def?.label || fieldKey,
   528→            extracted_value: value,
   529→            confidence: 80,
   530→          });
   531→        }
   532→      }
   533→    }
   534→  }
   535→
   536→  return fields;
   537→}

   790→      }

   798→      }
   799→    }
   800→
   801→    // Look for company name (first line that's not a keyword/number)
   802→    if (!fields.find(f => f.field_key === "shipper")) {
   803→      for (const line of lines) {

   806→        if (trimmed.length < 3 || trimmed.length > 60) continue;

   813→          fields.push({ field_key: "shipper", field_label: "Shipper/Exporter", extracted_value: trimmed, confidence: 55 });
   814→          break;
   815→        }
   816→      }
   817→    }
   818→  }
   819→
   820→  return fields;

   823→// --- PDF text-layer extraction (Tier 2) ---------------------------------------
   824→// Extracts embedded text from PDF files that have a real text layer
   825→// (not scanned images). Uses a simple regex-based approach to find text
   826→// between BT/ET markers in the PDF content stream.
   827→function extractPdfTextLayer(arrayBuffer: ArrayBuffer): string | null {
   828→  try {
   829→    const bytes = new Uint8Array(arrayBuffer);
   830→    // Check if it's actually a PDF
   831→    const header = new TextDecoder().decode(bytes.slice(0, 5));
   832→    if (header !== '%PDF-') return null;
   833→
   834→    // Extract text from PDF content streams
   835→    // Look for text in parentheses within BT...ET blocks
   836→    const text = new TextDecoder('latin1').decode(bytes);
   837→    const textMatches: string[] = [];
   838→
   839→    // Match text in parentheses (PDF text objects)
   840→    const textRegex = /\(([^)]+)\)/g;
   841→    let match;
   842→    while ((match = textRegex.exec(text)) !== null) {
   843→      const t = match[1];
   844→      // Filter out non-text content (binary data, short strings)
   845→      if (t.length > 2 && /[a-zA-Z]/.test(t) && !/\\[nrt]/.test(t)) {
   846→        textMatches.push(t);
   847→      }
   848→    }
   849→
   850→    if (textMatches.length < 3) return null; // Not enough text to be useful
   851→
   852→    // Join and clean up
   853→    const result = textMatches.join('\n');
   854→    return result.length > 50 ? result : null;
   855→  } catch {
   856→    return null;
   857→  }
   858→}

   948→    const ai = getGeminiClient();
   949→
   950→    if (!ai) {
   951→      console.warn(
   952→        "[extract-document] GEMINI_API_KEY not set — using mock extraction"
   953→      );
   954→    }
   955→
   956→    const allFields: any[] = [];
   957→    const perDocResults: any[] = [];
   958→    let latestShipper: string | null = null;
   959→    let latestConsignee: string | null = null;
   960→    let geminiDebug: any = null;
   961→    // Track whether extraction failed completely for ANY document so we can
   962→    // return a clear error response at the end (rather than silently succeeding
   963→    // with zero fields).
   964→    let totalExtractionFailures = 0;
   965→    const failureDetails: any[] = [];
   966→
   967→    // 4. Process each document through the 5-tier extraction fallback chain
   968→    for (const doc of docs) {
   969→      let extracted: any[] = [];
   970→      let usedModel: string | null = null;
   971→      let rawAiResponse: string | null = null;
   972→      let extractionSource: string = "none";
   973→      let extractionTier: number = 0;
   974→
   975→      // Mark document as 'extracting'
   976→      await admin.from("documents").update({
   977→        processing_status: "extracting",
   978→        processing_started_at: new Date().toISOString(),
   979→      }).eq("id", doc.id);
   980→
   981→      // 4a. Download the file from Storage
   982→      const { data: fileData, error: downloadErr } = await admin.storage
   983→        .from("documents")
   984→        .download(doc.storage_path);
   985→
   986→      if (downloadErr || !fileData) {
   987→        console.error("[extract-document] file download failed for", doc.storage_path, downloadErr);
   988→        await admin.from("documents").update({ processing_status: "failed" }).eq("id", doc.id);
   989→        perDocResults.push({ documentId: doc.id, file_name: doc.file_name, status: "download_failed", fields: [] });
   990→        continue;
   991→      }
   992→
   993→      const ab = await fileData.arrayBuffer();
   994→      const base64 = bufToBase64(ab);
   995→      const mimeType = doc.mime_type || "application/octet-stream";
   996→
   997→      // Extract raw text for text-based files (used by regex fallback)
   998→      let rawText: string | undefined;
   999→      if (mimeType.startsWith("text/")) {
  1000→        try { rawText = new TextDecoder().decode(ab); } catch {}
  1001→      }
  1002→
  1003→      // ─── TIER 1: Gemini Vision (primary) ───
  1004→      if (ai) {
  1005→        extractionTier = 1;
  1006→        const geminiResult = await callGeminiExtraction(ai, mimeType, base64, rawText);
  1007→        extracted = geminiResult.fields;
  1008→        geminiDebug = geminiResult.debug;
  1009→        usedModel = geminiResult.model;
  1010→        rawAiResponse = geminiResult.rawResponse;
  1011→        if (extracted.length > 0) {
  1012→          extractionSource = usedModel || "gemini";
  1013→        }
  1014→      }
  1015→
  1016→      // ─── TIER 2: PDF text-layer extraction (if Gemini failed and it's a PDF) ───
  1017→      if (extracted.length === 0 && mimeType === "application/pdf") {
  1018→        extractionTier = 2;
  1019→        console.log("[extract-document] Tier 2: trying PDF text-layer extraction");
  1020→        const pdfText = extractPdfTextLayer(ab);
  1021→        if (pdfText) {
  1022→          rawText = pdfText; // Use the extracted PDF text for regex
  1023→          console.log("[extract-document] PDF text-layer found, running regex on it");
  1024→        }
  1025→      }
  1026→
  1027→      // ─── TIER 3: Cloud Vision OCR placeholder ───
  1028→      // (Not implemented in this environment — would require a Google Cloud Vision API key
  1029→      // separate from Gemini. Falls through to Tier 4.)
  1030→      if (extracted.length === 0 && !rawText) {
  1031→        extractionTier = 3;
  1032→        // Cloud Vision would go here: const visionText = await callCloudVision(base64, mimeType);
  1033→        // if (visionText) rawText = visionText;
  1034→      }
  1035→
  1036→      // ─── TIER 4: Tesseract OCR (local, zero quota) ───
  1037→      if (extracted.length === 0 && !rawText) {
  1038→        extractionTier = 4;
  1039→        console.log("[extract-document] Tier 4: trying Tesseract OCR");
  1040→        const tesseractText = await tesseractOCR(ab, mimeType);
  1041→        if (tesseractText) {
  1042→          rawText = tesseractText;
  1043→          console.log("[extract-document] Tesseract produced text, running regex on it");
  1044→        }
  1045→      }
  1046→
  1047→      // ─── Regex extraction on whatever text we have (works for Tiers 2, 3, 4) ───
  1048→      if (extracted.length === 0 && rawText) {
  1049→        console.log(`[extract-document] Running regex extraction (tier ${extractionTier})`);
  1050→        extracted = regexExtract(rawText);
  1051→        if (extracted.length > 0) {
  1052→          extractionSource = extractionTier === 2 ? "pdf_text_layer" :
  1053→                             extractionTier === 4 ? "tesseract" : "regex_fallback";
  1054→        }
  1055→      }
  1056→
  1057→      // ─── TIER 5: Mark as 'needs_manual_review' — NEVER silent zero ───
  1058→      if (extracted.length === 0) {
  1059→        extractionTier = 5;
  1060→        extractionSource = "needs_manual_review";
  1061→
  1062→        // Mark the document for manual review
  1063→        await admin.from("documents").update({
  1064→          processing_status: "needs_manual_review",
  1065→          extraction_source: "needs_manual_review",
  1066→        }).eq("id", doc.id);
  1067→
  1068→        // Create an exception for manual review
  1069→        await userClient.from("exceptions").insert({
  1070→          shipment_id: shipmentId,
  1071→          org_id: orgMember?.org_id || null,
  1072→          user_id: user.id,
  1073→          field_key: "_document",
  1074→          field_name: `Document needs manual review: ${doc.file_name}`,
  1075→          original_value: "",
  1076→          extracted_value: "",
  1077→          confidence: 0,
  1078→          reason: `All extraction tiers failed for ${doc.file_name}. Document requires manual review.`,
  1079→          exception_type: "missing_field",
  1080→          doc_type: doc.doc_type || "Unknown",
  1081→          status: "Unresolved",
  1082→          history: [],
  1083→        }).then(({ error }: any) => {
  1084→          if (error) console.warn("[extract-document] failed to create manual review exception:", error.message);
  1085→        });
  1086→
  1087→        totalExtractionFailures++;
  1088→        failureDetails.push({
  1089→          documentId: doc.id,
  1090→          file_name: doc.file_name,
  1091→          reason: "All 5 extraction tiers failed — document marked for manual review",
  1092→          tiersTried: [1, 2, 3, 4],
  1093→          modelsTried: geminiDebug?.modelsTried || [],
  1094→          errors: geminiDebug?.errors || [],
  1095→        });
  1096→
  1097→        perDocResults.push({
  1098→          documentId: doc.id,
  1099→          file_name: doc.file_name,
  1100→          doc_type: doc.doc_type,
  1101→          status: "needs_manual_review",
  1102→          extractionSource: "needs_manual_review",
  1103→          extractionTier: 5,
  1104→          fieldsCount: 0,
  1105→          fields: [],
  1106→        });
  1107→
  1108→        // Audit log the manual review flag
  1109→        await userClient.from("audit_logs").insert({
  1110→          shipment_id: shipmentId,
  1111→          user_id: user.id,
  1112→          text: `[extraction] Document ${doc.file_name} marked for manual review — all 5 tiers failed`,
  1113→          type: "warning",
  1114→        }).then(({ error }: any) => { if (error) console.warn("[extract-document] audit log failed:", error.message); });
  1115→
  1116→        continue;
  1117→      }
  1118→
  1119→      // ─── Success: mark document as completed ───
  1120→      await admin.from("documents").update({
  1121→        processing_status: "completed",
  1122→        extraction_source: extractionSource,
  1123→      }).eq("id", doc.id);
  1124→
  1125→      // 4a-bis. Persist raw AI response to audit_logs for compliance/debug.
  1126→      // We truncate to 500 chars to avoid bloating the audit log.
  1127→      if (rawAiResponse) {
  1128→        const truncated = rawAiResponse.length > 500
  1129→          ? rawAiResponse.substring(0, 500) + `…(+${rawAiResponse.length - 500} chars)`
  1130→          : rawAiResponse;
  1131→        await userClient.from("audit_logs").insert({
  1132→          shipment_id: shipmentId,
  1133→          user_id: user.id,
  1134→          text: `Gemini raw response (model: ${usedModel}, fields: ${extracted.length}) — doc ${doc.file_name}: ${truncated}`,
  1135→          type: "info",
  1136→        }).then(({ error: alErr }: any) => {
  1137→          if (alErr) {
  1138→            console.warn("[extract-document] audit_log insert failed for raw response:", alErr.message);
  1139→          }
  1140→        });
  1141→      }
  1142→
  1143→      // 4a-ter. If Gemini AND regex both returned nothing, record the failure.
  1144→      // We still proceed to write whatever fields we DID get (likely none for
  1145→      // this doc), and continue with the next document. We surface the
  1146→      // aggregate failure at the end of the response.
  1147→      if (extracted.length === 0) {
  1148→        totalExtractionFailures++;
  1149→        failureDetails.push({
  1150→          documentId: doc.id,
  1151→          file_name: doc.file_name,
  1152→          reason: "no fields extracted from any model or regex",
  1153→          modelsTried: geminiDebug?.modelsTried || [],
  1154→          errors: geminiDebug?.errors || [],
  1155→        });
  1156→        perDocResults.push({
  1157→          documentId: doc.id,
  1158→          file_name: doc.file_name,
  1159→          doc_type: doc.doc_type,
  1160→          status: "extraction_failed",
  1161→          fieldsCount: 0,
  1162→          fields: [],
  1163→        });
  1164→        continue;
  1165→      }
  1166→
  1167→      // 4b. Normalize + write to document_fields
  1168→      const writtenFields: any[] = [];
  1169→      for (const f of extracted) {
  1170→        const key = String(f.field_key || "").trim();
  1171→        const value = String(f.extracted_value ?? "").trim();
  1172→        if (!key || !value) continue;
  1173→
  1174→        // Map label if missing
  1175→        const def = FIELD_DEFINITIONS.find((d) => d.key === key);
  1176→        const label = String(f.field_label || def?.label || key);
  1177→        const conf = Math.max(
  1178→          0,
  1179→          Math.min(100, Number(f.confidence ?? 70) || 70)
  1180→        );
  1181→
  1182→        const insertRow = {
  1183→          document_id: doc.id,
  1184→          shipment_id: shipmentId,
  1185→          user_id: user.id,
  1186→          field_key: key,
  1187→          field_label: label,
  1188→          extracted_value: value,
  1189→          confidence: conf,
  1190→          is_flagged: false,
  1191→          validation_errors: [],
  1192→          bounding_box: f.bounding_box || null,
  1193→          extraction_source: extractionSource,
  1194→        };
  1195→
  1196→        const { data: inserted, error: insErr } = await userClient
  1197→          .from("document_fields")
  1198→          .insert(insertRow)
  1199→          .select("id, field_key, field_label, extracted_value, confidence")
  1200→          .single();
  1201→
  1202→        if (insErr) {
  1203→          console.warn(
  1204→            "[extract-document] field insert failed:",
  1205→            key,
  1206→            insErr.message
  1207→          );
  1208→          continue;
  1209→        }
  1210→
  1211→        writtenFields.push(inserted);
  1212→        allFields.push(inserted);
  1213→
  1214→        // Track shipper / consignee for shipment row update
  1215→        if (key === "shipper" && !latestShipper) latestShipper = value;
  1216→        if (key === "consignee" && !latestConsignee) latestConsignee = value;
  1217→      }
  1218→
  1219→      perDocResults.push({
  1220→        documentId: doc.id,
  1221→        file_name: doc.file_name,
  1222→        doc_type: doc.doc_type,
  1223→        status: "extracted",
  1224→        extractionSource,
  1225→        model: usedModel,
  1226→        fieldsCount: writtenFields.length,
  1227→        fields: writtenFields,
  1228→      });
  1229→    }
  1230→
  1231→    // 5. Update shipment shipper / consignee if extracted
  1232→    const shipUpdate: any = {};
  1233→    if (latestShipper) shipUpdate.shipper = latestShipper;
  1234→    if (latestConsignee) shipUpdate.consignee = latestConsignee;
  1235→
  1236→    if (Object.keys(shipUpdate).length > 0) {
  1237→      const { error: shipUpdErr } = await userClient
  1238→        .from("shipments")
  1239→        .update(shipUpdate)
  1240→        .eq("id", shipmentId);
  1241→      if (shipUpdErr) {
  1242→        console.warn(
  1243→          "[extract-document] shipment shipper/consignee update failed:",
  1244→          shipUpdErr
  1245→        );
  1246→      }
  1247→    }
  1248→
  1249→    // 6. Audit log
  1250→    await userClient.from("audit_logs").insert({
  1251→      shipment_id: shipmentId,
  1252→      user_id: user.id,
  1253→      text: `Gemini extraction completed for ${docs.length} document(s) — ${allFields.length} fields extracted.`,
  1254→      type: "success",
  1255→    });
  1256→
  1257→    // 7. Respond — the response mirrors ExtractDocumentResponse shape.
  1258→    // If ALL models failed AND regex returned nothing for every document,
  1259→    // return a clear error instead of silently succeeding with 0 fields.
  1260→    const totalDocsAttempted = docs.length;
  1261→    if (
  1262→      ai &&
  1263→      allFields.length === 0 &&
  1264→      totalExtractionFailures === totalDocsAttempted &&
  1265→      totalDocsAttempted > 0
  1266→    ) {
  1267→      // Every document failed extraction — surface a clear, actionable error.
  1268→      await userClient.from("audit_logs").insert({
  1269→        shipment_id: shipmentId,
  1270→        user_id: user.id,
  1271→        text: `Extraction failed: all models unavailable across ${totalDocsAttempted} document(s).`,
  1272→        type: "error",
  1273→      });
  1274→
  1275→      return jsonRes(
  1276→        {
  1277→          success: false,
  1278→          error: "Extraction failed: all models unavailable",
  1279→          details: {
  1280→            documentsAttempted: totalDocsAttempted,
  1281→            failures: failureDetails,
  1282→            debug: geminiDebug,
  1283→          },
  1284→        },
  1285→        502
  1286→      );
  1287→    }
  1288→
  1289→    return jsonRes({
  1290→      success: true,
  1291→      shipmentId,
  1292→      documentId: documentId || docs[0]?.id,
  1293→      fields: allFields.map((f) => ({
  1294→        field_key: f.field_key,
  1295→        field_label: f.field_label,
  1296→        extracted_value: f.extracted_value,
  1297→        confidence: f.confidence,
  1298→        bounding_box: f.bounding_box || null,
  1299→      })),
  1300→      documents: perDocResults,
  1301→      shipper: latestShipper || undefined,
  1302→      consignee: latestConsignee || undefined,
  1303→      geminiUsed: !!ai,
  1304→      debug: geminiDebug,
  1305→      ...(totalExtractionFailures > 0
  1306→        ? {
  1307→            partialFailure: {
  1308→              failedDocuments: totalExtractionFailures,
  1309→              totalDocuments: totalDocsAttempted,
  1310→              details: failureDetails,
  1311→            },
  1312→          }
  1313→        : {}),
  1314→    });
  1315→  } catch (err: any) {
  1316→    console.error("[extract-document] unhandled error:", err);
  1317→    return jsonRes(
  1318→      { error: "Internal server error", detail: String(err?.message || err) },
  1319→      500
  1320→    );
  1321→  }
  1322→});

    32→// ============================================================================
    33→// Context Type
    34→// ============================================================================
    35→
    36→export type SupabaseStatus = 'connected' | 'unconfigured' | 'error_tables' | 'loading';
    37→export type EdgeFunctionStatus = 'live' | 'fallback' | 'unknown';
    38→
    39→interface ClearPortContextType {
    40→  entries: ShipmentEntry[];
    41→  selectedEntryId: string;
    42→  selectedEntry: ShipmentEntry | undefined;
    43→  selectedExceptionId: string;
    44→  selectedException: Exception | undefined;
    45→  activeTab: string;
    46→  rules: OperationalRules;
    47→  undoStack: { entryId: string; exceptionId: string; previousState: Exception }[];
    48→  auditLogs: AuditLog[];
    49→  theme: 'dark' | 'light';
    50→  supabaseStatus: SupabaseStatus;
    51→  edgeFunctionStatus: EdgeFunctionStatus;
    52→  currentUser: string;
    53→  currentTime: string;
    54→  // RBAC role for the current user — anonymous users default to 'operator'
    55→  // so the no-login UX continues to work. Components use the can* helpers
    56→  // (canUpload, canResolve, canManageRules, canExport) from rbac.service
    57→  // to gate their interactive UI.
    58→  userRole: UserRole;
    59→
    60→  // Actions
    61→  selectEntry: (id: string) => void;
    62→  selectException: (id: string) => void;
    63→  setActiveTab: (tab: string) => void;
    64→  updateException: (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => void;
    65→  undoLastAction: () => void;
    66→  acceptAllHighConfidence: (entryId: string) => void;
    67→  uploadDocuments: (files: File[]) => Promise<{ shipmentId: string; success: boolean; error?: string }>;
    68→  updateRules: (newRules: Partial<OperationalRules>) => void;
    69→  exportToCSV: (entryId: string) => Promise<void>;
    70→  toggleTheme: () => void;
    71→  refreshData: () => Promise<void>;
    72→}
    73→
    74→const ClearPortContext = React.createContext<ClearPortContextType | undefined>(undefined);

    76→// ============================================================================
    77→// Provider

    79→
    80→export const ClearPortProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
    81→  const [entries, setEntries] = React.useState<ShipmentEntry[]>(seedEntries);
    82→  const [selectedEntryId, setSelectedEntryId] = React.useState<string>('SHIP-2026-8802');
    83→  const [selectedExceptionId, setSelectedExceptionId] = React.useState<string>('8802-hts');
    84→  const [activeTab, setActiveTab] = React.useState<string>('exception-desk');
    85→  const [theme, setTheme] = React.useState<'dark' | 'light'>('dark');
    86→  const [rules, setRules] = React.useState<OperationalRules>(seedRules);
    87→  const [undoStack, setUndoStack] = React.useState<{ entryId: string; exceptionId: string; previousState: Exception }[]>([]);
    88→  const [auditLogs, setAuditLogs] = React.useState<AuditLog[]>(seedLogs);
    89→  const [supabaseStatus, setSupabaseStatus] = React.useState<SupabaseStatus>(
    90→    isSupabaseConfigured() ? 'loading' : 'unconfigured'
    91→  );
    92→  const [edgeFunctionStatus, setEdgeFunctionStatus] = React.useState<EdgeFunctionStatus>('unknown');
    93→  const [currentUser, setCurrentUser] = React.useState<string>('Broker');
    94→  const [currentTime, setCurrentTime] = React.useState<string>('');
    95→  // Anonymous users default to 'operator'. In production this would be
    96→  // fetched from the server (e.g. GET /api/me → role from user_roles table).
    97→  const [userRole, setUserRole] = React.useState<UserRole>(getDefaultRole());
    98→
    99→  // --- Real-time clock ---
   100→  React.useEffect(() => {
   101→    const update = () => {
   102→      const now = new Date();
   103→      const utc = now.toISOString().slice(0, 16).replace('T', ' ') + ' UTC';
   104→      setCurrentTime(utc);
   105→    };
   106→    update();
   107→    const interval = setInterval(update, 1000);
   108→    return () => clearInterval(interval);
   109→  }, []);
   110→
   111→  // --- Load user email ---
   112→  React.useEffect(() => {
   113→    getCurrentUserEmail().then(email => setCurrentUser(email));
   114→  }, []);
   115→
   116→  // --- Persist theme ---
   117→  React.useEffect(() => {
   118→    const saved = typeof window !== 'undefined' ? localStorage.getItem('clearport-theme') : null;
   119→    if (saved === 'light' || saved === 'dark') setTheme(saved);
   120→  }, []);
   121→
   122→  const toggleTheme = React.useCallback(() => {
   123→    setTheme(prev => {
   124→      const next = prev === 'dark' ? 'light' : 'dark';
   125→      if (typeof window !== 'undefined') localStorage.setItem('clearport-theme', next);
   126→      return next;
   127→    });
   128→  }, []);
   129→
   130→  // --- Load data ---
   131→  const selectedEntryIdRef = React.useRef(selectedEntryId);
   132→  selectedEntryIdRef.current = selectedEntryId;
   133→
   134→  const loadData = React.useCallback(async () => {
   135→    if (!isSupabaseConfigured()) {
   136→      setSupabaseStatus('unconfigured');
   137→      setEdgeFunctionStatus('fallback');
   138→      return;
   139→    }
   140→
   141→    setSupabaseStatus('loading');
   142→
   143→    try {
   144→      await ensureAuthenticated();
   145→      const email = await getCurrentUserEmail();
   146→      setCurrentUser(email);
   147→
   148→      // Try the new /api/shipments route (replaces both the get-shipments
   149→      // edge function call AND the fetchShipmentsDirect fallback).
   150→      try {
   151→        const response = await apiFetch<{
   152→          data: ShipmentEntry[];
   153→          pagination: { total: number };
   154→        }>('/api/shipments?page=1&limit=100');
   155→
   156→        if (response?.data && Array.isArray(response.data)) {
   157→          // Only use DB shipments if they have fields/exceptions data (new
   158→          // schema). Old-schema shipments have empty arrays — fall back to
   159→          // seed data so the demo UX is preserved.
   160→          const hasRealData = response.data.some(
   161→            (s) => s.fields.length > 0 || s.exceptions.length > 0,
   162→          );
   163→
   164→          if (hasRealData) {
   165→            setEntries(response.data);
   166→            // Select first shipment if current selection doesn't exist
   167→            if (!response.data.find((s) => s.id === selectedEntryIdRef.current)) {
   168→              const first = response.data[0];
   169→              setSelectedEntryId(first.id);
   170→              if (first.exceptions?.length > 0) {
   171→                const unresolved = first.exceptions.find((e) => e.status === 'Unresolved');
   172→                setSelectedExceptionId(unresolved ? unresolved.id : first.exceptions[0].id);
   173→              }
   174→            }
   175→          }
   176→          // else: keep seed entries (demo mode)
   177→
   178→          setEdgeFunctionStatus('live');
   179→          setSupabaseStatus('connected');
   180→
   181→          // Fetch rules + logs in parallel via the new API routes.
   182→          const [rulesRes, logsRes] = await Promise.all([
   183→            apiFetch<{ rules: OperationalRules }>('/api/rules').catch(() => null),
   184→            apiFetch<{ logs: AuditLog[] }>('/api/audit-logs').catch(() => null),
   185→          ]);
   186→          if (rulesRes?.rules) setRules(rulesRes.rules);
   187→          if (logsRes?.logs && logsRes.logs.length > 0) setAuditLogs(logsRes.logs);
   188→          return;
   189→        }
   190→      } catch (apiErr) {
   191→        console.warn('[ctx] /api/shipments failed, falling back to seed data:', apiErr);
   192→      }
   193→
   194→      // Fallback: seed data (demo mode)
   195→      setSupabaseStatus('connected');
   196→      setEdgeFunctionStatus('fallback');
   197→    } catch (err) {
   198→      console.error('[ctx] loadData error:', err);
   199→      setSupabaseStatus('error_tables');
   200→      setEdgeFunctionStatus('fallback');
   201→    }
   202→  }, []);
   203→
   204→  React.useEffect(() => {
   205→    loadData();
   206→  }, [loadData]);
   207→
   208→  const refreshData = React.useCallback(async () => {
   209→    await loadData();
   210→  }, [loadData]);
   211→
   212→  // --- Computed ---
   213→  const selectedEntry = React.useMemo(
   214→    () => entries.find(e => e.id === selectedEntryId),
   215→    [entries, selectedEntryId]
   216→  );
   217→
   218→  const selectedException = React.useMemo(() => {
   219→    if (!selectedEntry) return undefined;
   220→    return selectedEntry.exceptions.find(ex => ex.id === selectedExceptionId);
   221→  }, [selectedEntry, selectedExceptionId]);
   222→
   223→  // --- Selection (fixes stale closure — uses functional update) ---
   224→  const selectEntry = React.useCallback((id: string) => {
   225→    setSelectedEntryId(id);
   226→    setEntries(prev => {
   227→      const entry = prev.find(e => e.id === id);
   228→      if (entry && entry.exceptions.length > 0) {
   229→        const unresolved = entry.exceptions.find(e => e.status === 'Unresolved');
   230→        setSelectedExceptionId(unresolved ? unresolved.id : entry.exceptions[0].id);
   231→      } else {
   232→        setSelectedExceptionId('');
   233→      }
   234→      return prev;
   235→    });
   236→  }, []);
   237→
   238→  const selectException = React.useCallback((id: string) => {
   239→    setSelectedExceptionId(id);
   240→  }, []);
   241→
   242→  // --- Audit log helper ---
   243→  const addAuditLog = React.useCallback((text: string, type: AuditLog['type'] = 'info', shipmentId?: string) => {
   244→    const newLog: AuditLog = {
   245→      id: typeof crypto !== 'undefined' ? crypto.randomUUID() : `log-${Date.now()}`,
   246→      text,
   247→      timestamp: new Date().toISOString(),
   248→      type,
   249→      shipmentId,
   250→    };
   251→    setAuditLogs(prev => [newLog, ...prev]);
   252→
   253→    if (isSupabaseConfigured() && supabase) {
   254→      supabase.from('audit_logs').insert({
   255→        id: newLog.id,
   256→        text: newLog.text,
   257→        timestamp: newLog.timestamp,
   258→        type: newLog.type,
   259→        shipment_id: shipmentId || null,
   260→      }).then(({ error }) => {
   261→        if (error) console.warn('[ctx] audit log insert failed:', error.message);
   262→      });
   263→    }
   264→  }, []);
   265→
   266→  // --- Update exception (fixes setState-in-setState) ---
   267→  const updateException = React.useCallback(
   268→    (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => {
   269→      let previousException: Exception | undefined;
   270→      let updatedEntry: ShipmentEntry | undefined;
   271→
   272→      setEntries(prevEntries => {
   273→        return prevEntries.map(entry => {
   274→          if (entry.id !== entryId) return entry;
   275→
   276→          const prev = entry.exceptions.find(ex => ex.id === exceptionId);
   277→          if (!prev) return entry;
   278→          previousException = { ...prev };
   279→
   280→          const updatedExceptions = entry.exceptions.map(ex => {
   281→            if (ex.id !== exceptionId) return ex;
   282→
   283→            const oldVal = ex.correctedValue || ex.extractedValue;
   284→            const updatedVal = newValue !== undefined ? newValue : ex.extractedValue;
   285→
   286→            const historyItem: ExceptionHistoryEntry = {
   287→              user: currentUser,
   288→              oldValue: oldVal,
   289→              newValue: updatedVal,
   290→              timestamp: new Date().toISOString(),
   291→              action: status,
   292→            };
   293→
   294→            return {
   295→              ...ex,
   296→              status: status as ExceptionStatus,
   297→              correctedValue: status === 'Corrected' ? newValue : undefined,
   298→              history: [historyItem, ...ex.history],
   299→              resolvedAt: new Date().toISOString(),
   300→              resolvedBy: currentUser,
   301→            };
   302→          });
   303→
   304→          const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   305→          const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   306→          const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   307→
   308→          const updatedFields = entry.fields.map(f => {
   309→            if (f.exceptionId === exceptionId) {
   310→              const val = status === 'Corrected' ? (newValue || f.value) : f.value;
   311→              return { ...f, value: val, isFlagged: false };
   312→            }
   313→            return f;
   314→          });
   315→
   316→          updatedEntry = {
   317→            ...entry,
   318→            status: newStatus,
   319→            exceptions: updatedExceptions,
   320→            fields: updatedFields,
   321→            currentConfidence: newConfidence,
   322→          };
   323→
   324→          return updatedEntry;
   325→        });
   326→      });
   327→
   328→      // Side effects AFTER setState
   329→      if (previousException && updatedEntry) {
   330→        setUndoStack(prevStack => [
   331→          { entryId, exceptionId, previousState: { ...previousException! } },
   332→          ...prevStack,
   333→        ]);
   334→
   335→        const logText = `Exception "${previousException.fieldName}" for ${entryId} was ${status.toLowerCase()}${
   336→          status === 'Corrected' ? ` to "${newValue}"` : ''
   337→        } by ${currentUser}.`;
   338→        addAuditLog(logText, status === 'Rejected' ? 'warning' : 'success', entryId);
   339→
   340→        // Persist via the /api/exceptions/:id route (handles exception update,
   341→        // document_field sync, shipment confidence recompute, and audit log
   342→        // in one server-side transaction).
   343→        if (isSupabaseConfigured()) {
   344→          apiFetch(`/api/exceptions/${exceptionId}`, {
   345→            method: 'PATCH',
   346→            body: JSON.stringify({
   347→              status,
   348→              correctedValue: status === 'Corrected' ? newValue : undefined,
   349→            }),
   350→          }).catch((err) => {
   351→            console.warn('[ctx] exception update failed:', err instanceof Error ? err.message : err);
   352→          });
   353→        }
   354→      }
   355→    },
   356→    [currentUser, addAuditLog]
   357→  );
   358→
   359→  // --- Undo ---
   360→  const undoLastAction = React.useCallback(() => {
   361→    if (undoStack.length === 0) return;
   362→
   363→    const [lastAction, ...remainingStack] = undoStack;
   364→    setUndoStack(remainingStack);
   365→
   366→    setEntries(prevEntries => {
   367→      return prevEntries.map(entry => {
   368→        if (entry.id !== lastAction.entryId) return entry;
   369→
   370→        const restored = lastAction.previousState;
   371→        const updatedExceptions = entry.exceptions.map(ex =>
   372→          ex.id === lastAction.exceptionId ? restored : ex
   373→        );
   374→
   375→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   376→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   377→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   378→
   379→        const updatedFields = entry.fields.map(f => {
   380→          if (f.exceptionId === lastAction.exceptionId) {
   381→            return {
   382→              ...f,
   383→              value: restored.status === 'Corrected' ? (restored.correctedValue || f.value) : restored.extractedValue,
   384→              isFlagged: restored.status === 'Unresolved',
   385→            };
   386→          }
   387→          return f;
   388→        });
   389→
   390→        const updatedEntry = {
   391→          ...entry,
   392→          status: newStatus,
   393→          exceptions: updatedExceptions,
   394→          fields: updatedFields,
   395→          currentConfidence: newConfidence,
   396→        };
   397→
   398→        if (isSupabaseConfigured() && supabase) {
   399→          supabase.from('exceptions').update({
   400→            status: restored.status,
   401→            corrected_value: restored.correctedValue || null,
   402→            resolved_at: restored.resolvedAt || null,
   403→            resolved_by: restored.resolvedBy || null,
   404→            history: restored.history,
   405→          }).eq('id', lastAction.exceptionId).then(({ error }) => {
   406→            if (error) console.warn('[ctx] undo exception failed:', error.message);
   407→          });
   408→
   409→          supabase.from('shipments').update({
   410→            status: newStatus,
   411→            current_confidence: newConfidence,
   412→            updated_at: new Date().toISOString(),
   413→          }).eq('id', entry.id).then(({ error }) => {
   414→            if (error) console.warn('[ctx] undo shipment failed:', error.message);
   415→          });
   416→        }
   417→
   418→        return updatedEntry;
   419→      });
   420→    });
   421→
   422→    addAuditLog(`Undo applied for last review action on shipment ${lastAction.entryId}`, 'info', lastAction.entryId);
   423→  }, [undoStack, addAuditLog]);
   424→
   425→  // --- Batch accept (uses configured threshold) ---
   426→  const acceptAllHighConfidence = React.useCallback((entryId: string) => {
   427→    setEntries(prevEntries => {
   428→      return prevEntries.map(entry => {
   429→        if (entry.id !== entryId) return entry;
   430→
   431→        const cutoff = rules.invoiceThreshold;
   432→        let acceptedCount = 0;
   433→
   434→        const updatedExceptions = entry.exceptions.map(ex => {
   435→          if (ex.status === 'Unresolved' && ex.confidence >= cutoff) {
   436→            acceptedCount++;
   437→            const historyItem: ExceptionHistoryEntry = {
   438→              user: currentUser,
   439→              oldValue: ex.extractedValue,
   440→              newValue: ex.extractedValue,
   441→              timestamp: new Date().toISOString(),
   442→              action: 'Accepted' as ReviewerAction,
   443→            };
   444→            return {
   445→              ...ex,
   446→              status: 'Accepted' as ExceptionStatus,
   447→              history: [historyItem, ...ex.history],
   448→              resolvedAt: new Date().toISOString(),
   449→              resolvedBy: currentUser,
   450→            };
   451→          }
   452→          return ex;
   453→        });
   454→
   455→        if (acceptedCount === 0) return entry;
   456→
   457→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   458→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   459→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   460→
   461→        const updatedFields = entry.fields.map(f => {
   462→          const exc = updatedExceptions.find(e => e.id === f.exceptionId);
   463→          if (exc) return { ...f, isFlagged: exc.status === 'Unresolved' };
   464→          return f;
   465→        });
   466→
   467→        const updatedEntry = {
   468→          ...entry,
   469→          status: newStatus,
   470→          exceptions: updatedExceptions,
   471→          fields: updatedFields,
   472→          currentConfidence: newConfidence,
   473→        };
   474→
   475→        addAuditLog(
   476→          `Batch action: Approved ${acceptedCount} high-confidence exceptions (≥${cutoff}%) in ${entryId}.`,
   477→          'success',
   478→          entryId
   479→        );
   480→
   481→        // Persist via the /api/exceptions/batch-accept route (accepts each
   482→        // qualifying exception, syncs document_fields, recomputes shipment
   483→        // confidence + status, and writes a batch-level audit log).
   484→        if (isSupabaseConfigured()) {
   485→          apiFetch('/api/exceptions/batch-accept', {
   486→            method: 'POST',
   487→            body: JSON.stringify({
   488→              shipmentId: entryId,
   489→              threshold: cutoff,
   490→            }),
   491→          }).catch((err) => {
   492→            console.warn('[ctx] batch accept failed:', err instanceof Error ? err.message : err);
   493→          });
   494→        }
   495→
   496→        return updatedEntry;
   497→      });
   498→    });
   499→  }, [rules.invoiceThreshold, currentUser, addAuditLog]);
   500→
   501→  // --- Upload documents ---
   502→  const uploadDocuments = React.useCallback(async (files: File[]): Promise<{ shipmentId: string; success: boolean; error?: string }> => {
   503→    const shipmentId = `SHIP-${new Date().getFullYear()}-${Math.floor(1000 + Math.random() * 9000)}`;
   504→
   505→    try {
   506→      if (!isSupabaseConfigured()) {
   507→        // Fallback
   508→        const newEntry: ShipmentEntry = {
   509→          id: shipmentId,
   510→          shipper: 'Titanium Castings KK',
   511→          consignee: 'Ironclad Logistics Inc.',
   512→          status: 'Under Review',
   513→          docsCount: files.length,
   514→          urgency: '08:30:00',
   515→          initialConfidence: 70,
   516→          currentConfidence: 70,
   517→          createdAt: new Date().toISOString(),
   518→          documents: [],
   519→          exceptions: [],
   520→          fields: [],
   521→        };
   522→
   523→        const excId = typeof crypto !== 'undefined' ? crypto.randomUUID() : `${shipmentId}-address`;
   524→        newEntry.exceptions = [{
   525→          id: excId,
   526→          fieldName: 'Consignee Corporate Address',
   527→          fieldKey: 'consigneeAddress',
   528→          originalValue: 'Suite 200, Seattle, WA',
   529→          extractedValue: 'Suite 200, Seattle, WA',
   530→          crossDocValue: 'Suite 201, Seattle, WA',
   531→          confidence: 62,
   532→          reason: 'Address abbreviation mismatch: Commercial Invoice lists "Suite 200, Seattle, WA" while Bill of Lading shows "Suite 201, Seattle, WA".',
   533→          exceptionType: 'cross_doc_mismatch',
   534→          docType: 'Bill of Lading',
   535→          boundingBox: { x: 10, y: 35, w: 32, h: 5 },
   536→          status: 'Unresolved',
   537→          history: [],
   538→          createdAt: new Date().toISOString(),
   539→        }];
   540→
   541→        newEntry.fields = [
   542→          { id: 'n1', key: 'invoiceNo', label: 'Commercial Invoice #', value: 'TC-9921-KK', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 90 },
   543→          { id: 'n2', key: 'invoiceDate', label: 'Invoice Date', value: new Date().toISOString().split('T')[0], sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 88 },
   544→          { id: 'n3', key: 'shipper', label: 'Shipper / Exporter', value: 'Titanium Castings KK', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 95 },
   545→          { id: 'n4', key: 'consignee', label: 'Consignee / Importer', value: 'Ironclad Logistics Inc.', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 94 },
   546→          { id: 'n5', key: 'consigneeAddress', label: 'Consignee Address', value: 'Suite 200, Seattle, WA', sourceDoc: 'Commercial Invoice', isFlagged: true, confidence: 62, exceptionId: excId, crossDocValue: 'Suite 201, Seattle, WA' },
   547→          { id: 'n6', key: 'declaredValue', label: 'Total Declared Value', value: '$45,210.00', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 87 },
   548→          { id: 'n7', key: 'htsCode', label: 'HTS Code (Primary Line)', value: '7308.90.0000', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 91 },
   549→          { id: 'n8', key: 'netWeight', label: 'Total Net Weight', value: '8,410 lbs', sourceDoc: 'Bill of Lading', isFlagged: false, confidence: 89 },
   550→        ];
   551→
   552→        setEntries(prev => [newEntry, ...prev]);
   553→        setSelectedEntryId(shipmentId);
   554→        setSelectedExceptionId(excId);
   555→        addAuditLog(`New entry ${shipmentId} ingestion and auto-analysis completed with 1 exception.`, 'info', shipmentId);
   556→
   557→        return { shipmentId, success: true };
   558→      }
   559→
   560→      // Real pipeline
   561→      await ensureAuthenticated();
   562→
   563→      // Step 1: Upload each file to Storage via edge function
   564→      for (const file of files) {
   565→        const formData = new FormData();
   566→        formData.append('file', file);
   567→        formData.append('shipment_id', shipmentId);
   568→
   569→        try {
   570→          if (supabase) {
   571→            const { error } = await supabase.functions.invoke('upload-document', {
   572→              body: formData,
   573→            });
   574→            if (error) throw error;
   575→          }
   576→        } catch (err) {
   577→          console.warn('[upload] edge function failed for file:', err);
   578→          // Continue — extraction will still try to work on whatever was uploaded
   579→        }
   580→      }
   581→
   582→      // Step 2: Create the shipment row in DB (if not already created by upload-document)
   583→      if (supabase) {
   584→        await supabase.from('shipments').upsert({
   585→          id: shipmentId,
   586→          shipper: 'Pending Extraction',
   587→          consignee: 'Pending Extraction',
   588→          status: 'Under Review',
   589→          docs_count: files.length,
   590→          urgency: '08:30:00',
   591→          initial_confidence: 0,
   592→          current_confidence: 0,
   593→        }).then(({ error }) => {
   594→          if (error) console.warn('[upload] shipment upsert:', error.message);
   595→        });
   596→      }
   597→
   598→      // Step 3: Extract fields via Gemini edge function
   599→      let extractedFields: ExtractedField[] = [];
   600→      let shipper = 'Unknown Shipper';
   601→      let consignee = 'Unknown Consignee';
   602→
   603→      try {
   604→        const extractResponse = await invokeEdgeFunction<any>('extract-document', { shipmentId });
   605→        if (extractResponse?.success && extractResponse.fields && extractResponse.fields.length > 0) {
   606→          extractedFields = extractResponse.fields.map((f: any) => ({
   607→            id: typeof crypto !== 'undefined' ? crypto.randomUUID() : `f-${Date.now()}`,
   608→            key: f.field_key,
   609→            label: f.field_label,
   610→            value: f.extracted_value,
   611→            sourceDoc: 'Commercial Invoice',
   612→            isFlagged: false,
   613→            confidence: f.confidence,
   614→            boundingBox: f.bounding_box,
   615→          }));
   616→          if (extractResponse.shipper) shipper = extractResponse.shipper;
   617→          if (extractResponse.consignee) consignee = extractResponse.consignee;
   618→        }
   619→      } catch (err) {
   620→        console.warn('[extract] edge function failed:', err);
   621→      }
   622→
   623→      // Step 4: Call the validation chain IN PARALLEL (not sequential)
   624→      // schema-validate, math-validate, cross-validate are independent → run together
   625→      // flag-exceptions runs after (it needs the validated fields)
   626→      // Fire-and-forget — don't block the UI on validation
   627→      const validationChain = (async () => {
   628→        try {
   629→          // Run the 3 independent validators in parallel
   630→          await Promise.allSettled([
   631→            invokeEdgeFunction('schema-validate', { shipmentId }),
   632→            invokeEdgeFunction('math-validate', { shipmentId }),
   633→            invokeEdgeFunction('cross-validate', { shipmentId }),
   634→          ]);
   635→          // Then run flag-exceptions (depends on validated fields)
   636→          await invokeEdgeFunction('flag-exceptions', { shipmentId }).catch(() => {});
   637→        } catch (err) {
   638→          console.warn('[validation-chain] error:', err);
   639→        }
   640→      })();
   641→      // Don't await — let validation run in the background
   642→
   643→      // Step 5: Reload from API to get the real DB state (with fields, exceptions, documents)
   644→      if (extractedFields.length > 0) {
   645→        // If we have extracted fields, build the entry from extraction response
   646→        const newExceptions: Exception[] = [];
   647→        const updatedFields = extractedFields.map(f => {
   648→          let threshold = rules.invoiceThreshold;
   649→          if (f.key === 'htsCode') threshold = rules.htsThreshold;
   650→          else if (f.key === 'shipper' || f.key === 'consignee' || f.key === 'consigneeAddress') threshold = rules.partiesThreshold;
   651→
   652→          if (f.confidence < threshold) {
   653→            const excId = crypto.randomUUID();
   654→            const exc: Exception = {
   655→              id: excId,
   656→              fieldName: f.label,
   657→              fieldKey: f.key,
   658→              originalValue: f.value,
   659→              extractedValue: f.value,
   660→              confidence: f.confidence,
   661→              reason: `Extracted confidence (${f.confidence}%) is below threshold (${threshold}%).`,
   662→              exceptionType: 'low_confidence',
   663→              docType: f.sourceDoc,
   664→              boundingBox: f.boundingBox || { x: 10, y: 35, w: 32, h: 5 },
   665→              status: 'Unresolved',
   666→              history: [],
   667→              createdAt: new Date().toISOString(),
   668→            };
   669→            newExceptions.push(exc);
   670→            return { ...f, isFlagged: true, exceptionId: excId };
   671→          }
   672→          return f;
   673→        });
   674→
   675→        const initialConfidence = Math.round(
   676→          updatedFields.reduce((acc, f) => acc + f.confidence, 0) / Math.max(updatedFields.length, 1)
   677→        );
   678→
   679→        const newEntry: ShipmentEntry = {
   680→          id: shipmentId,
   681→          shipper,
   682→          consignee,
   683→          status: 'Under Review',
   684→          docsCount: files.length,
   685→          urgency: '08:30:00',
   686→          initialConfidence,
   687→          currentConfidence: initialConfidence,
   688→          createdAt: new Date().toISOString(),
   689→          documents: [],
   690→          exceptions: newExceptions,
   691→          fields: updatedFields,
   692→        };
   693→
   694→        setEntries(prev => [newEntry, ...prev]);
   695→        setSelectedEntryId(shipmentId);
   696→        setSelectedExceptionId(newExceptions[0]?.id || '');
   697→      } else {
   698→        // No fields extracted — reload from API to get whatever the DB has
   699→        await loadData();
   700→        setSelectedEntryId(shipmentId);
   701→      }
   702→
   703→      addAuditLog(
   704→        `New entry ${shipmentId} ingestion completed: ${files.length} files, ${extractedFields.length} fields extracted.`,
   705→        extractedFields.length > 0 ? 'success' : 'warning',
   706→        shipmentId
   707→      );
   708→
   709→      return { shipmentId, success: true };
   710→    } catch (err) {
   711→      const errMsg = err instanceof Error ? err.message : 'Upload pipeline failed';
   712→      addAuditLog(`Upload failed for ${shipmentId}: ${errMsg}`, 'error', shipmentId);
   713→      return { shipmentId, success: false, error: errMsg };
   714→    }
   715→  }, [rules, addAuditLog]);
   716→
   717→  // --- Update rules ---
   718→  const updateRules = React.useCallback((newRules: Partial<OperationalRules>) => {
   719→    setRules(prev => {
   720→      const updated = { ...prev, ...newRules };
   721→
   722→      // Persist via the /api/rules route (upserts the user's rules row).
   723→      if (isSupabaseConfigured()) {
   724→        apiFetch('/api/rules', {
   725→          method: 'PATCH',
   726→          body: JSON.stringify(newRules),
   727→        }).catch((err) => {
   728→          console.warn('[ctx] rules update failed:', err instanceof Error ? err.message : err);
   729→        });
   730→      }
   731→
   732→      return updated;
   733→    });
   734→    addAuditLog('Compliance operational thresholds updated.', 'info');
   735→  }, [addAuditLog]);
   736→
   737→  // --- Real CSV export ---
   738→  const exportToCSV = React.useCallback(async (entryId: string) => {
   739→    const entry = entries.find(e => e.id === entryId);
   740→    if (!entry) return;
   741→
   742→    // Try the /api/export/:id route first (generates CSV server-side from the
   743→    // DB-backed shipment data and returns it as text/csv).
   744→    if (isSupabaseConfigured()) {
   745→      try {
   746→        const res = await apiFetch<Response>(`/api/export/${entryId}`, { raw: true });
   747→        if (res.ok) {
   748→          const csv = await res.text();
   749→          if (csv) {
   750→            const blob = new Blob([csv], { type: 'text/csv' });
   751→            const url = URL.createObjectURL(blob);
   752→            const a = document.createElement('a');
   753→            a.href = url;
   754→            a.download = `ClearPort_Audit_${entryId}.csv`;
   755→            document.body.appendChild(a);
   756→            a.click();
   757→            document.body.removeChild(a);
   758→            URL.revokeObjectURL(url);
   759→
   760→            addAuditLog(`Audit logs exported to CSV for ${entryId}.`, 'success', entryId);
   761→            return;
   762→          }
   763→        }
   764→      } catch (err) {
   765→        console.warn('[csv] /api/export failed, generating locally:', err);
   766→      }
   767→    }
   768→
   769→    // Fallback: local CSV generation from in-memory entry
   770→    const rows: string[] = [];
   771→    rows.push('Field Key,Field Label,Value,Source Document,Confidence,Flagged,Status');
   772→    entry.fields.forEach(f => {
   773→      const exc = entry.exceptions.find(e => e.id === f.exceptionId);
   774→      const status = exc ? exc.status : 'SECURE';
   775→      const escaped = [f.key, f.label, f.value, f.sourceDoc, f.confidence, f.isFlagged, status]
   776→        .map(v => `"${String(v).replace(/"/g, '""')}"`)
   777→        .join(',');
   778→      rows.push(escaped);
   779→    });
   780→
   781→    rows.push('');
   782→    rows.push('Exception ID,Field Name,Reason,Confidence,Status,Resolved By');
   783→    entry.exceptions.forEach(e => {
   784→      const escaped = [e.id, e.fieldName, e.reason, e.confidence, e.status, e.resolvedBy || '']
   785→        .map(v => `"${String(v).replace(/"/g, '""')}"`)
   786→        .join(',');
   787→      rows.push(escaped);
   788→    });
   789→
   790→    const csv = rows.join('\n');
   791→    const blob = new Blob([csv], { type: 'text/csv' });
   792→    const url = URL.createObjectURL(blob);
   793→    const a = document.createElement('a');
   794→    a.href = url;
   795→    a.download = `ClearPort_Audit_${entryId}.csv`;
   796→    document.body.appendChild(a);
   797→    a.click();
   798→    document.body.removeChild(a);
   799→    URL.revokeObjectURL(url);
   800→
   801→    addAuditLog(`Audit logs exported to CSV for ${entryId}.`, 'success', entryId);
   802→  }, [entries, addAuditLog]);
   803→
   804→  const value: ClearPortContextType = {
   805→    entries,
   806→    selectedEntryId,
   807→    selectedEntry,
   808→    selectedExceptionId,
   809→    selectedException,
   810→    activeTab,
   811→    rules,
   812→    undoStack,
   813→    auditLogs,
   814→    theme,
   815→    supabaseStatus,
   816→    edgeFunctionStatus,
   817→    currentUser,
   818→    currentTime,
   819→    userRole,
   820→    selectEntry,
   821→    selectException,
   822→    setActiveTab,
   823→    updateException,
   824→    undoLastAction,
   825→    acceptAllHighConfidence,
   826→    uploadDocuments,
   827→    updateRules,
   828→    exportToCSV,
   829→    toggleTheme,
   830→    refreshData,
   831→  };
   832→
   833→  return (
   834→    <ClearPortContext.Provider value={value}>
   835→      {children}
   836→    </ClearPortContext.Provider>
   837→  );
   838→};
   839→
   840→export function useClearPort() {
   841→  const context = React.useContext(ClearPortContext);
   842→  if (context === undefined) {
   843→    throw new Error('useClearPort must be used within a ClearPortProvider');
   844→  }
   845→  return context;
   846→}

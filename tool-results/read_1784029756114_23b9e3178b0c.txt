   231→export const ClearPortProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
   232→  const [entries, setEntries] = React.useState<ShipmentEntry[]>(initialEntries);
   233→  const [selectedEntryId, setSelectedEntryId] = React.useState<string>('SHIP-2026-8802');
   234→  const [theme, setTheme] = React.useState<'dark' | 'light'>('dark');
   235→
   236→  const toggleTheme = React.useCallback(() => {
   237→    setTheme((prev) => (prev === 'dark' ? 'light' : 'dark'));
   238→  }, []);
   239→  const [selectedExceptionId, setSelectedExceptionId] = React.useState<string>('8802-hts');
   240→  const [activeTab, setActiveTab] = React.useState<string>('exception-desk');
   241→  const [rules, setRules] = React.useState<OperationalRules>({
   242→    invoiceThreshold: 80,
   243→    htsThreshold: 85,
   244→    partiesThreshold: 75,
   245→  });
   246→  const [undoStack, setUndoStack] = React.useState<{ entryId: string; exceptionId: string; previousState: Exception }[]>([]);
   247→  const [auditLogs, setAuditLogs] = React.useState(initialLogs);
   248→  const [supabaseStatus, setSupabaseStatus] = React.useState<'connected' | 'unconfigured' | 'error_tables' | 'loading'>(
   249→    isSupabaseConfigured() ? 'loading' : 'unconfigured'
   250→  );
   251→
   252→  const loadFromSupabase = React.useCallback(async () => {
   253→    if (!isSupabaseConfigured()) {
   254→      setSupabaseStatus('unconfigured');
   255→      return;
   256→    }
   257→    
   258→    try {
   259→      setSupabaseStatus('loading');
   260→      const dbShipments = await getSupabaseShipments();
   261→      
   262→      if (dbShipments === null) {
   263→        setSupabaseStatus('error_tables');
   264→        return;
   265→      }
   266→
   267→      // Fetch document fields to enrich shipments
   268→      let fieldsData: any[] | null = null;
   269→      try {
   270→        fieldsData = await getSupabaseDocumentFields();
   271→      } catch (err) {
   272→        console.warn('Failed to load document fields, falling back to shipments JSON:', err);
   273→      }
   274→
   275→      const enrichShipmentsList = (shipments: ShipmentEntry[]) => {
   276→        if (!fieldsData || fieldsData.length === 0) return shipments;
   277→        return shipments.map(ship => {
   278→          const shipFields = fieldsData!.filter(f => f.documents?.shipment_id === ship.id);
   279→          if (shipFields.length > 0) {
   280→            const fields: ExtractedField[] = shipFields.map(f => ({
   281→              key: f.field_key,
   282→              label: f.field_label,
   283→              value: f.corrected_value || f.extracted_value,
   284→              sourceDoc: f.documents?.doc_type || 'Document',
   285→              isFlagged: f.is_flagged,
   286→              exceptionId: f.is_flagged ? String(f.id) : undefined,
   287→            }));
   288→
   289→            const exceptions: Exception[] = shipFields
   290→              .filter(f => f.is_flagged)
   291→              .map(f => ({
   292→                id: String(f.id),
   293→                fieldName: f.field_label,
   294→                fieldKey: f.field_key,
   295→                originalValue: f.extracted_value,
   296→                extractedValue: f.extracted_value,
   297→                confidence: f.confidence,
   298→                reason: f.exception_reason || `Extracted confidence (${f.confidence}%) is below threshold.`,
   299→                docType: f.documents?.doc_type || 'Document',
   300→                boundingBox: { x: 10, y: 10, w: 20, h: 4 },
   301→                status: f.reviewer_action ? (f.reviewer_action as any) : 'Unresolved',
   302→                correctedValue: f.corrected_value || undefined,
   303→                history: [],
   304→              }));
   305→
   306→            return {
   307→              ...ship,
   308→              fields,
   309→              exceptions,
   310→            };
   311→          }
   312→          return ship;
   313→        });
   314→      };
   315→
   316→      // If table exists but has no data, seed it!
   317→      if (dbShipments.length === 0) {
   318→        for (const entry of initialEntries) {
   319→          await upsertSupabaseShipment(entry);
   320→        }
   321→        const seededShipments = await getSupabaseShipments();
   322→        if (seededShipments) {
   323→          setEntries(enrichShipmentsList(seededShipments));
   324→        }
   325→      } else {
   326→        setEntries(enrichShipmentsList(dbShipments));
   327→      }
   328→
   329→      // Load Rules
   330→      const dbRules = await getSupabaseRules();
   331→      if (dbRules) {
   332→        setRules(dbRules);
   333→      } else {
   334→        // Seed initial rules if empty
   335→        await upsertSupabaseRules({
   336→          invoiceThreshold: 80,
   337→          htsThreshold: 85,
   338→          partiesThreshold: 75,
   339→        });
   340→      }
   341→
   342→      // Load Logs
   343→      const dbLogs = await getSupabaseLogs();
   344→      if (dbLogs && dbLogs.length > 0) {
   345→        setAuditLogs(dbLogs.map(l => ({
   346→          id: l.id,
   347→          text: l.text,
   348→          timestamp: l.timestamp,
   349→          type: l.type as 'info' | 'success' | 'warning',
   350→        })));
   351→      } else {
   352→        // Seed initial logs
   353→        for (const log of initialLogs) {
   354→          await insertSupabaseLog({
   355→            id: log.id,
   356→            text: log.text,
   357→            timestamp: log.timestamp,
   358→            type: log.type as 'info' | 'success' | 'warning',
   359→          });
   360→        }
   361→      }
   362→
   363→      setSupabaseStatus('connected');
   364→    } catch (err) {
   365→      console.error('Error loading data from Supabase:', err);
   366→      setSupabaseStatus('error_tables');
   367→    }
   368→  }, []);
   369→
   370→  React.useEffect(() => {
   371→    const timer = setTimeout(() => {
   372→      loadFromSupabase();
   373→    }, 0);
   374→    return () => clearTimeout(timer);
   375→  }, [loadFromSupabase]);
   376→
   377→  const refreshSupabaseData = React.useCallback(async () => {
   378→    await loadFromSupabase();
   379→  }, [loadFromSupabase]);
   380→
   381→  // Computed values
   382→  const selectedEntry = React.useMemo(() => {
   383→    return entries.find((e) => e.id === selectedEntryId);
   384→  }, [entries, selectedEntryId]);
   385→
   386→  const selectedException = React.useMemo(() => {
   387→    if (!selectedEntry) return undefined;
   388→    return selectedEntry.exceptions.find((ex) => ex.id === selectedExceptionId);
   389→  }, [selectedEntry, selectedExceptionId]);
   390→
   391→  // Adjust selections automatically if shipment changes
   392→  const selectEntry = React.useCallback((id: string) => {
   393→    setSelectedEntryId(id);
   394→    const entry = entries.find((e) => e.id === id);
   395→    if (entry && entry.exceptions.length > 0) {
   396→      // Find first unresolved exception if any
   397→      const unresolved = entry.exceptions.find(e => e.status === 'Unresolved');
   398→      setSelectedExceptionId(unresolved ? unresolved.id : entry.exceptions[0].id);
   399→    } else {
   400→      setSelectedExceptionId('');
   401→    }
   402→  }, [entries]);
   403→
   404→  const selectException = React.useCallback((id: string) => {
   405→    setSelectedExceptionId(id);
   406→  }, []);
   407→
   408→  const updateRules = React.useCallback((newRules: Partial<OperationalRules>) => {
   409→    setRules((prev) => {
   410→      const updated = { ...prev, ...newRules };
   411→      if (isSupabaseConfigured()) {
   412→        upsertSupabaseRules(updated);
   413→      }
   414→      return updated;
   415→    });
   416→    setAuditLogs((prev) => {
   417→      const newLog = {
   418→        id: `log-${Date.now()}`,
   419→        text: `Compliance operational thresholds updated.`,
   420→        timestamp: new Date().toISOString(),
   421→        type: 'info' as const,
   422→      };
   423→      if (isSupabaseConfigured()) {
   424→        insertSupabaseLog(newLog);
   425→      }
   426→      return [newLog, ...prev];
   427→    });
   428→  }, []);
   429→
   430→  // Accept, edit, or reject exception
   431→  const updateException = React.useCallback(
   432→    (entryId: string, exceptionId: string, status: 'Accepted' | 'Corrected' | 'Rejected', newValue?: string) => {
   433→      setEntries((prevEntries) => {
   434→        return prevEntries.map((entry) => {
   435→          if (entry.id !== entryId) return entry;
   436→
   437→          const previousException = entry.exceptions.find((ex) => ex.id === exceptionId);
   438→          if (!previousException) return entry;
   439→
   440→          // Push to undo stack
   441→          setUndoStack((prevStack) => [
   442→            { entryId, exceptionId, previousState: { ...previousException } },
   443→            ...prevStack,
   444→          ]);
   445→
   446→          const updatedExceptions = entry.exceptions.map((ex) => {
   447→            if (ex.id !== exceptionId) return ex;
   448→
   449→            const oldVal = ex.correctedValue || ex.extractedValue;
   450→            const updatedVal = newValue !== undefined ? newValue : ex.extractedValue;
   451→
   452→            const historyItem = {
   453→              user: 'syednasirbukhari033@gmail.com',
   454→              oldValue: oldVal,
   455→              newValue: updatedVal,
   456→              timestamp: new Date().toISOString(),
   457→              action: status,
   458→            };
   459→
   460→            return {
   461→              ...ex,
   462→              status,
   463→              correctedValue: status === 'Corrected' ? newValue : undefined,
   464→              history: [historyItem, ...ex.history],
   465→            };
   466→          });
   467→
   468→          // Check if all exceptions are resolved
   469→          const allResolved = updatedExceptions.every((ex) => ex.status !== 'Unresolved');
   470→          const newStatus = (allResolved ? 'Approved' : 'Under Review') as 'Under Review' | 'Approved' | 'Exported';
   471→
   472→          // Recalculate confidence score dynamically
   473→          // Every resolved exception increases confidence score towards 100%
   474→          const resolvedCount = updatedExceptions.filter((ex) => ex.status !== 'Unresolved').length;
   475→          const totalExc = updatedExceptions.length;
   476→          const confidenceBoost = totalExc > 0 ? ((100 - entry.initialConfidence) * (resolvedCount / totalExc)) : 0;
   477→          const currentConfidence = Math.min(100, Math.round(entry.initialConfidence + confidenceBoost));
   478→
   479→          // Also update fields values to sync values inline!
   480→          const updatedFields = entry.fields.map((f) => {
   481→            if (f.exceptionId === exceptionId) {
   482→              const updatedVal = status === 'Corrected' ? (newValue || f.value) : f.value;
   483→              return {
   484→                ...f,
   485→                value: updatedVal,
   486→                isFlagged: false, // resolving clears the active error flag
   487→              };
   488→            }
   489→            return f;
   490→          });
   491→
   492→          // Create audit log
   493→          const logText = `Exception "${previousException.fieldName}" for ${entryId} was ${status.toLowerCase()}${
   494→            status === 'Corrected' ? ` to "${newValue}"` : ''
   495→          } by syednasirbukhari033@gmail.com.`;
   496→          
   497→          const newLog = {
   498→            id: `log-${Date.now()}`,
   499→            text: logText,
   500→            timestamp: new Date().toISOString(),
   501→            type: status === 'Rejected' ? 'warning' as const : 'success' as const,
   502→          };
   503→
   504→          if (isSupabaseConfigured()) {
   505→            insertSupabaseLog(newLog);
   506→          }
   507→
   508→          setAuditLogs((prevLogs) => [newLog, ...prevLogs]);
   509→
   510→          const updatedEntry = {
   511→            ...entry,
   512→            status: newStatus,
   513→            exceptions: updatedExceptions,
   514→            fields: updatedFields,
   515→            currentConfidence,
   516→          };
   517→
   518→          if (isSupabaseConfigured()) {
   519→            upsertSupabaseShipment(updatedEntry);
   520→            if (exceptionId && !isNaN(Number(exceptionId))) {
   521→              updateSupabaseDocumentField(Number(exceptionId), status, newValue);
   522→            }
   523→          }
   524→
   525→          return updatedEntry;
   526→        });
   527→      });
   528→    },
   529→    []
   530→  );
   531→
   532→  const undoLastAction = React.useCallback(() => {
   533→    if (undoStack.length === 0) return;

   543→          if (ex.id !== lastAction.exceptionId) return ex;
   544→          return lastAction.previousState;
   545→        });
   546→
   547→        const allResolved = updatedExceptions.every((ex) => ex.status !== 'Unresolved');
   548→        const newStatus = (allResolved ? 'Approved' : 'Under Review') as 'Under Review' | 'Approved' | 'Exported';

   551→        const totalExc = updatedExceptions.length;
   552→        const confidenceBoost = totalExc > 0 ? ((100 - entry.initialConfidence) * (resolvedCount / totalExc)) : 0;
   553→        const currentConfidence = Math.min(100, Math.round(entry.initialConfidence + confidenceBoost));
   554→
   555→        // Restore field flags
   556→        const restoredException = lastAction.previousState;
   557→        const updatedFields = entry.fields.map((f) => {
   558→          if (f.exceptionId === lastAction.exceptionId) {
   559→            return {
   560→              ...f,
   561→              value: restoredException.status === 'Corrected' ? (restoredException.correctedValue || f.value) : restoredException.extractedValue,
   562→              isFlagged: restoredException.status === 'Unresolved',
   563→            };
   564→          }
   565→          return f;
   566→        });
   567→
   568→        const updatedEntry = {
   569→          ...entry,
   570→          status: newStatus,
   571→          exceptions: updatedExceptions,
   572→          fields: updatedFields,
   573→          currentConfidence,
   574→        };
   575→
   576→        if (isSupabaseConfigured()) {
   577→          upsertSupabaseShipment(updatedEntry);
   578→        }
   579→
   580→        return updatedEntry;
   581→      });
   582→    });
   583→
   584→    const newLog = {
   585→      id: `log-${Date.now()}`,
   586→      text: `Undo applied for last review action on shipment ${lastAction.entryId}`,
   587→      timestamp: new Date().toISOString(),
   588→      type: 'info' as const,
   589→    };
   590→
   591→    if (isSupabaseConfigured()) {
   592→      insertSupabaseLog(newLog);
   593→    }
   594→
   595→    setAuditLogs((prevLogs) => [newLog, ...prevLogs]);
   596→  }, [undoStack]);

   598→  const acceptAllHighConfidence = React.useCallback((entryId: string) => {

   619→          }

   621→        });
   622→
   623→        if (acceptedCount === 0) return entry;

   626→        const newStatus = (allResolved ? 'Approved' : 'Under Review') as 'Under Review' | 'Approved' | 'Exported';
   627→
   628→        const resolvedCount = updatedExceptions.filter((ex) => ex.status !== 'Unresolved').length;
   629→        const totalExc = updatedExceptions.length;
   630→        const confidenceBoost = totalExc > 0 ? ((100 - entry.initialConfidence) * (resolvedCount / totalExc)) : 0;
   631→        const currentConfidence = Math.min(100, Math.round(entry.initialConfidence + confidenceBoost));
   632→
   633→        // Sync field values
   634→        const updatedFields = entry.fields.map((f) => {
   635→          const exception = updatedExceptions.find((ex) => ex.id === f.exceptionId);
   636→          if (exception) {
   637→            return {
   638→              ...f,
   639→              isFlagged: exception.status === 'Unresolved',
   640→            };
   641→          }
   642→          return f;
   643→        });
   644→
   645→        const newLog = {
   646→          id: `log-${Date.now()}`,
   647→          text: `Batch action: Approved ${acceptedCount} high-confidence exceptions in ${entryId}.`,
   648→          timestamp: new Date().toISOString(),
   649→          type: 'success' as const,
   650→        };
   651→
   652→        if (isSupabaseConfigured()) {
   653→          insertSupabaseLog(newLog);
   654→          updatedExceptions.forEach((ex) => {
   655→            if (ex.status === 'Accepted' && ex.id && !isNaN(Number(ex.id))) {
   656→              if (ex.confidence >= 70) {
   657→                updateSupabaseDocumentField(Number(ex.id), 'Accepted');
   658→              }
   659→            }
   660→          });
   661→        }
   662→
   663→        setAuditLogs((prevLogs) => [newLog, ...prevLogs]);
   664→
   665→        const updatedEntry = {
   666→          ...entry,
   667→          status: newStatus,
   668→          exceptions: updatedExceptions,
   669→          fields: updatedFields,
   670→          currentConfidence,
   671→        };
   672→
   673→        if (isSupabaseConfigured()) {
   674→          upsertSupabaseShipment(updatedEntry);
   675→        }
   676→
   677→        return updatedEntry;
   678→      });
   679→    });
   680→  }, []);

    76→const GEMINI_PROMPT = `You are a strict customs document OCR engine. Extract structured fields from this document.
    77→
    78→STRICT OUTPUT CONTRACT — read carefully:
    79→1. Return ONLY a JSON array. Do NOT wrap it in markdown fences. Do NOT add any prose, explanation, or commentary before or after the array.
    80→2. Every object in the array MUST have EXACTLY these four keys (no more, no less):
    81→   - "field_key": string (one of the allowed keys listed below)
    82→   - "field_label": string (the human-readable label listed below)
    83→   - "extracted_value": string (the verbatim value read from the document) OR null
    84→   - "confidence": integer between 0 and 100
    85→3. If a field is NOT present in the document, OMIT it from the array entirely. Do NOT include it with a null value.
    86→4. If you cannot clearly read a value, OMIT that field. Do NOT guess, infer, hallucinate, or fabricate values. Only extract fields that are clearly visible in the document.
    87→5. Confidence must reflect your actual certainty about BOTH the field's presence AND the accuracy of the extracted value:
    88→   - 95-100: very clear, unambiguous, legible text, no possible doubt
    89→   - 80-94:  clear, but minor ambiguity (e.g. partial obscuring, slight blur)
    90→   - 60-79:  somewhat clear, value readable but with some uncertainty
    91→   - below 60: uncertain — you should usually OMIT the field rather than return it
    92→6. Use the EXACT field_key strings below (case-sensitive). Do not invent new keys.
    93→
    94→SPECIAL HANDLING RULES:
    95→
    96→A. MULTI-LINE TABLE ROWS:
    97→   - Commercial invoices often have line-item tables where a single item spans multiple rows.
    98→   - Secondary rows under a line item (e.g. "Shipping Cost", "Insurance", "Handling Fee") belong to the PRECEDING line item, NOT as separate items.
    99→   - If a table row has no QTY or HS Code, it is a secondary row — merge it with the line item above.
   100→   - Extract only the PRIMARY line item's HTS Code (field_key: "htsCode"), not secondary rows.
   101→
   102→B. CURRENCY & NUMBERS:
   103→   - Always include the currency symbol in extracted_value (e.g. "$52,150.75", not "52150.75").
   104→   - Accept $, €, £, ¥ as valid currency prefixes.
   105→   - If the total is labeled "Total", "Grand Total", "Invoice Total", or "Total Declared Value", extract it as field_key "declaredValue".
   106→
   107→C. FOREIGN CHARACTERS (UTF-8):
   108→   - Documents may contain foreign characters: German ß, ü, ö, ä; French é, è; Spanish ñ; etc.
   109→   - Preserve the original UTF-8 characters in extracted_value (e.g. "Industriestraße 14" stays as-is).
   110→   - Do NOT transliterate or strip foreign characters — the system handles UTF-8 normalization downstream.
   111→   - If a name contains special characters, extract it verbatim.
   112→
   113→D. BOUNDING BOX ISOLATION — "OFFICIAL CBP USE" AREA:
   114→   - Many customs documents have an "Official CBP Use Only" box (usually bottom-left or bottom-right).
   115→   - This box is for CBP officers to stamp/sign — it is NOT data to extract.
   116→   - Do NOT extract signatures, stamps, or handwriting from the "Official CBP Use" box.
   117→   - If a signature appears NEXT TO but OUTSIDE the CBP box, do not associate it with the box.
   118→   - Only extract typed/printed field values from the main document body, not from official-use boxes.

   120→Allowed fields (field_key → field_label):

   145→// Convert ArrayBuffer to base64 string (UTF-8 safe via chunked encoding)
   146→function bufToBase64(buf: ArrayBuffer): string {
   147→  const bytes = new Uint8Array(buf);
   148→  let binary = "";
   149→  const chunkSize = 0x8000; // 32 KB
   150→  for (let i = 0; i < bytes.length; i += chunkSize) {
   151→    const chunk = bytes.subarray(i, i + chunkSize);
   152→    binary += String.fromCharCode.apply(null, chunk as unknown as number[]);
   153→  }
   154→  return btoa(binary);
   155→}

   157→// Strip ```json fences and trailing prose from Gemini's reply.
   158→// Handles arrays, single objects, and objects with a wrapper like { fields: [...] }.
   159→function extractJsonArray(text: string): any[] {
   160→  if (!text) return [];
   161→  let cleaned = text.trim();
   162→
   163→  // Remove ```json ... ``` fences if present
   164→  const fenceMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/i);
   165→  if (fenceMatch) {
   166→    cleaned = fenceMatch[1].trim();
   167→  }
   168→
   169→  // Try direct parse first (fast path)
   170→  try {
   171→    const direct = JSON.parse(cleaned);
   172→    if (Array.isArray(direct)) return direct;
   173→    if (direct && typeof direct === "object") {
   174→      // Could be a wrapper object
   175→      if (Array.isArray(direct.fields)) return direct.fields;
   176→      if (Array.isArray(direct.data)) return direct.data;
   177→      // Single field object?
   178→      if (direct.field_key) return [direct];
   179→    }
   180→  } catch {
   181→    // Fall through to manual extraction
   182→  }
   183→
   184→  // Find the first '[' and matching last ']'
   185→  const start = cleaned.indexOf("[");
   186→  const end = cleaned.lastIndexOf("]");
   187→  if (start !== -1 && end !== -1 && end > start) {
   188→    const slice = cleaned.slice(start, end + 1);
   189→    try {
   190→      const parsed = JSON.parse(slice);
   191→      return Array.isArray(parsed) ? parsed : [];
   192→    } catch {
   193→      // Fall through
   194→    }
   195→  }
   196→
   197→  // Try finding a JSON object instead
   198→  const objStart = cleaned.indexOf("{");
   199→  const objEnd = cleaned.lastIndexOf("}");
   200→  if (objStart !== -1 && objEnd !== -1 && objEnd > objStart) {
   201→    const slice = cleaned.slice(objStart, objEnd + 1);
   202→    try {
   203→      const parsed = JSON.parse(slice);
   204→      if (Array.isArray(parsed)) return parsed;
   205→      if (parsed && typeof parsed === "object") {
   206→        if (Array.isArray(parsed.fields)) return parsed.fields;
   207→        if (parsed.field_key) return [parsed];
   208→      }
   209→    } catch {
   210→      // Give up
   211→    }
   212→  }
   213→
   214→  return [];
   215→}

   281→// Call Gemini with the file bytes. Returns an array of field objects.
   282→// Handles both binary files (via inlineData) and text files (via text prompt).
   283→// Returns { fields, debug, model, rawResponse } so the caller can surface
   284→// diagnostic info and persist the raw AI response for audit.
   285→//
   286→// MODEL CASCADE (most accurate first):
   287→//   gemini-2.5-pro  → gemini-2.0-flash → gemini-1.5-flash → regex fallback
   288→//
   289→// NOTE: Gemini handles multi-page PDFs natively via inlineData — the entire
   290→// PDF is sent as a single inlineData part and the model reads every page.
   291→// No special chunking or per-page calls are required.
   292→//
   293→// WALL-CLOCK BUDGET: a `deadline` (epoch ms) is threaded in from the caller.
   294→// Before every model attempt we check the remaining budget; if exhausted we
   295→// stop immediately so the caller can drop to manual review instead of
   296→// silently retrying for minutes. Each model call's race timeout is also
   297→// clamped to the remaining budget so a single slow model can't blow the
   298→// whole wall.
   299→async function callGeminiExtraction(
   300→  ai: GoogleGenAI,
   301→  mimeType: string,
   302→  base64Data: string,
   303→  rawText: string | undefined,
   304→  deadline: number
   305→): Promise<{ fields: any[]; debug: any; model: string | null; rawResponse: string | null; budgetExhausted: boolean }> {
   306→  // Ordered model cascade — Pro first (most accurate), then Flash variants.
   307→  const models = ["gemini-2.5-pro", "gemini-2.0-flash", "gemini-1.5-flash"];
   308→  const debug: any = { modelsTried: [], errors: [], rawResponses: [], retries: [] };
   309→
   310→  // Strict, deterministic config — temperature 0 + topP 0 + topK 1 means the
   311→  // model always picks the single highest-probability token, ensuring that
   312→  // the same document yields the same extraction on every run.
   313→  const generationConfig = { temperature: 0, topP: 0, topK: 1 };
   314→
   315→  for (const model of models) {
   316→    // ── Budget check before each model ──
   317→    const remaining = deadline - Date.now();
   318→    if (remaining <= 500) {
   319→      debug.errors.push(`budget exhausted before ${model} (remaining ${remaining}ms) — aborting Gemini cascade`);
   320→      debug.budgetExhausted = true;
   321→      return { fields: [], debug, model: null, rawResponse: null, budgetExhausted: true };
   322→    }
   323→    debug.modelsTried.push(model);
   324→    let lastErr: any = null;
   325→
   326→    // Retry up to 2 times on transient errors (429 rate-limit, 503 server
   327→    // error). Exponential backoff: 1s, then 2s.
   328→    for (let attempt = 0; attempt <= 2; attempt++) {
   329→      // ── Budget check before each attempt ──
   330→      const rem = deadline - Date.now();
   331→      if (rem <= 500) {
   332→        debug.errors.push(`budget exhausted before ${model} attempt ${attempt + 1} (remaining ${rem}ms)`);
   333→        debug.budgetExhausted = true;
   334→        return { fields: [], debug, model: null, rawResponse: null, budgetExhausted: true };
   335→      }
   336→      try {
   337→        const parts: any[] = [];
   338→
   339→        if (rawText && rawText.length > 0) {
   340→          parts.push({ text: `Document text content:\n\n${rawText}\n\n---\n${GEMINI_PROMPT}` });
   341→        } else {
   342→          parts.push({ inlineData: { mimeType, data: base64Data } });
   343→          parts.push({ text: GEMINI_PROMPT });
   344→        }
   345→
   346→        // Per-call timeout — clamped to the remaining wall-clock budget so a
   347→        // single hung model can't burn the whole budget. Floor of 2s so we
   348→        // still give the model a real chance; ceiling of 15s.
   349→        const callTimeout = Math.max(2000, Math.min(15000, rem));
   350→        const response = await Promise.race([
   351→          ai.models.generateContent({
   352→            model,
   353→            contents: [{ role: "user", parts }],
   354→            config: generationConfig,
   355→          }),
   356→          new Promise<never>((_, reject) =>
   357→            setTimeout(() => reject(new Error(`TIMEOUT_${callTimeout}ms after ${callTimeout}ms`)), callTimeout)
   358→          ),
   359→        ]);
   360→
   361→        // Try multiple ways to get text from the response
   362→        let text = "";
   363→        if (typeof response.text === "string") {
   364→          text = response.text;
   365→        } else if (typeof response.text === "function") {
   366→          text = response.text();
   367→        } else if (response?.candidates?.[0]?.content?.parts?.[0]?.text) {
   368→          text = response.candidates[0].content.parts[0].text;
   369→        } else if (response?.response?.text) {
   370→          text = response.response.text;
   371→        }
   372→
   373→        debug.rawResponses.push({ model, attempt, textLength: text.length, preview: text.substring(0, 300) });
   374→
   375→        const parsed = extractJsonArray(text);
   376→        if (parsed.length > 0) {
   377→          debug.success = true;
   378→          debug.successModel = model;
   379→          return { fields: parsed, debug, model, rawResponse: text, budgetExhausted: false };
   380→        } else {
   381→          debug.errors.push(`${model}: 0 parseable fields from ${text.length} chars`);
   382→          lastErr = new Error(`0 parseable fields from ${text.length} chars`);
   383→          // Non-retryable: model responded but produced nothing useful.
   384→          // Move on to the next model.
   385→          break;
   386→        }
   387→      } catch (err: any) {
   388→        const errMsg = err?.message || String(err);
   389→        lastErr = err;
   390→        debug.errors.push(`${model} (attempt ${attempt + 1}): ${errMsg}`);
   391→        debug.rawResponses.push({ model, attempt, error: errMsg });
   392→
   393→        const isRetryable =
   394→          /429/.test(errMsg) ||            // rate limit
   395→          /503/.test(errMsg) ||            // server error
   396→          /overloaded/i.test(errMsg) ||
   397→          /rate.?limit/i.test(errMsg) ||
   398→          /SERVICE_UNAVAILABLE/i.test(errMsg) ||
   399→          /RESOURCE_EXHAUSTED/i.test(errMsg);
   400→
   401→        // If quota exhausted (429), skip ALL remaining models — they share the same quota.
   402→        // Go straight to regex fallback to avoid wasting 15+ seconds on retries.
   403→        const isQuotaExhausted = /429/.test(errMsg) || /RESOURCE_EXHAUSTED/i.test(errMsg) || /quota/i.test(errMsg);
   404→        if (isQuotaExhausted) {
   405→          debug.errors.push(`${model}: quota exhausted — skipping remaining models, using regex fallback`);
   406→          debug.quotaExhausted = true;
   407→          return { fields: [], debug, model: null, rawResponse: null, budgetExhausted: false };
   408→        }
   409→
   410→        // ── Budget-aware retry: only sleep if the delay still fits the budget ──
   411→        if (isRetryable && attempt < 1) {
   412→          const delay = 1000; // Fixed 1s delay
   413→          const remAfterDelay = deadline - Date.now() - delay;
   414→          if (remAfterDelay <= 500) {
   415→            debug.errors.push(`${model}: retry skipped — budget would be exhausted (${remAfterDelay}ms left after ${delay}ms delay)`);
   416→            debug.budgetExhausted = true;
   417→            return { fields: [], debug, model: null, rawResponse: null, budgetExhausted: true };
   418→          }
   419→          debug.retries.push({ model, attempt: attempt + 1, delayMs: delay, reason: errMsg });
   420→          await new Promise((r) => setTimeout(r, delay));
   421→          continue;
   422→        }
   423→
   424→        // Non-retryable error (incl. TIMEOUT) — break out of retry loop
   425→        // and fall through to the next model.
   426→        if (errMsg.startsWith("TIMEOUT_")) {
   427→          debug.errors.push(`${model}: timed out, falling back to next model`);
   428→        }
   429→        break;
   430→      }
   431→    }
   432→    if (lastErr) {
   433→      // Already logged — continue to next model.
   434→    }
   435→  }
   436→
   437→  return { fields: [], debug, model: null, rawResponse: null, budgetExhausted: false };
   438→}

   442→// Sanitize a currency string: "$52,150.75" → "52150.75" (numeric only)
   443→function sanitizeCurrency(val: string): { display: string; numeric: number | null } {
   444→  const display = val.trim();
   445→  const cleaned = display.replace(/[^0-9.]/g, "");
   446→  const numeric = cleaned ? parseFloat(cleaned) : null;
   447→  return { display, numeric: numeric && Number.isFinite(numeric) ? numeric : null };
   448→}

   450→// Normalize UTF-8 foreign characters for ASCII-only compliance systems.
   451→function normalizeUtf8(val: string): { utf8: string; ascii: string } {
   452→  const utf8 = val.trim();
   453→  const ascii = utf8
   454→    .replace(/ä/g, "ae").replace(/ö/g, "oe").replace(/ü/g, "ue")
   455→    .replace(/Ä/g, "Ae").replace(/Ö/g, "Oe").replace(/Ü/g, "Ue")
   456→    .replace(/ß/g, "ss")
   457→    .replace(/é/g, "e").replace(/è/g, "e").replace(/ê/g, "e").replace(/ë/g, "e")
   458→    .replace(/á/g, "a").replace(/à/g, "a").replace(/â/g, "a")
   459→    .replace(/í/g, "i").replace(/ì/g, "i").replace(/î/g, "i")
   460→    .replace(/ó/g, "o").replace(/ò/g, "o").replace(/ô/g, "o")
   461→    .replace(/ú/g, "u").replace(/ù/g, "u").replace(/û/g, "u")
   462→    .replace(/ñ/g, "n").replace(/ç/g, "c")
   463→    .replace(/[^\x00-\x7F]/g, "")
   464→    .trim();
   465→  return { utf8, ascii };
   466→}

   468→// Parse multi-line table rows. Groups secondary lines with preceding line item.
   469→function parseTableRows(text: string): { lineItems: any[]; totalValue: string | null } {
   470→  const lines = text.split(/\n/).map(l => l.trim()).filter(l => l.length > 0);
   471→  const lineItems: any[] = [];
   472→  let totalValue: string | null = null;
   473→
   474→  const itemPattern = /^(.+?)\s+(\d+)\s+(\d{4}\.\d{2}\.\d{4}|\d{8,})\s+([$€£¥]?[\d,]+\.?\d*)\s*$/i;
   475→  const totalPattern = /^(?:total|grand\s*total|invoice\s*total|total\s*value)\s*[:\-]?\s*([$€£¥][\d,]+\.?\d*)/i;
   476→
   477→  for (const line of lines) {
   478→    const totalMatch = line.match(totalPattern);
   479→    if (totalMatch && totalMatch[1]) {
   480→      totalValue = totalMatch[1].trim();
   481→      continue;
   482→    }
   483→    const itemMatch = line.match(itemPattern);
   484→    if (itemMatch) {
   485→      const [, desc, qty, hts, value] = itemMatch;
   486→      lineItems.push({ description: desc.trim(), qty: parseInt(qty), htsCode: hts.trim(), value: value.trim(), secondaryLines: [] as string[] });
   487→      continue;
   488→    }
   489→    const isSecondary = /^(shipping\s*(?:cost|fee)?|insurance(?:\s*cost)?|freight|handling|duty|tax|discount|subtotal|other)\s*[:\-]?\s*([$€£¥]?[\d,]+\.?\d*)?/i.test(line);
   490→    if (isSecondary && lineItems.length > 0) {
   491→      lineItems[lineItems.length - 1].secondaryLines.push(line);
   492→    }
   493→  }
   494→  return { lineItems, totalValue };
   495→}

   497→// --- CSV Parser ----------------------------------------------------------------
   498→// Parse CSV content as structured fields. Expects rows with headers or key-value pairs.
   499→function parseCSV(text: string): any[] {
   500→  const fields: any[] = [];
   501→  const lines = text.split(/\n/).map(l => l.trim()).filter(l => l.length > 0);
   502→  if (lines.length === 0) return fields;
   503→
   504→  // Detect delimiter (comma, semicolon, tab)
   505→  const firstLine = lines[0];
   506→  const delimiter = firstLine.includes('\t') ? '\t' : firstLine.includes(';') ? ';' : ',';
   507→
   508→  // Check if it's a header-based CSV
   509→  const headers = firstLine.split(delimiter).map(h => h.trim().toLowerCase().replace(/[^a-z0-9]/g, ''));
   510→  
   511→  // Map CSV headers to our field keys
   512→  const headerMap: Record<string, string> = {
   513→    invoice: 'invoiceNo', invoicenumber: 'invoiceNo', invoiceno: 'invoiceNo', invoicenum: 'invoiceNo',
   514→    date: 'invoiceDate', invoicedate: 'invoiceDate',
   515→    shipper: 'shipper', exporter: 'shipper', shipperexporter: 'shipper',
   516→    consignee: 'consignee', importer: 'consignee', consigneeimporter: 'consignee',
   517→    value: 'declaredValue', totalvalue: 'declaredValue', declaredvalue: 'declaredValue', total: 'declaredValue', amount: 'declaredValue',
   518→    hts: 'htsCode', htscode: 'htsCode', tariff: 'htsCode', hscode: 'htsCode',
   519→    weight: 'netWeight', netweight: 'netWeight', net: 'netWeight',
   520→    grossweight: 'grossWeight', gross: 'grossWeight',
   521→    origin: 'countryOfOrigin', countryoforigin: 'countryOfOrigin', country: 'countryOfOrigin',
   522→    carrier: 'carrier',
   523→    port: 'portOfEntry', portofentry: 'portOfEntry',
   524→    bol: 'billOfLading', billoflading: 'billOfLading',
   525→    address: 'consigneeAddress', consigneeaddress: 'consigneeAddress',
   526→  };
   527→
   528→  // If first line looks like headers
   529→  const hasHeaders = headers.some(h => headerMap[h]);
   530→  
   531→  if (hasHeaders && lines.length >= 2) {
   532→    // Header-based CSV: use first data row
   533→    const values = lines[1].split(delimiter).map(v => v.trim().replace(/^["']|["']$/g, ''));
   534→    headers.forEach((header, idx) => {
   535→      const fieldKey = headerMap[header];
   536→      if (fieldKey && values[idx]) {
   537→        const def = FIELD_DEFINITIONS.find(d => d.key === fieldKey);
   538→        fields.push({
   539→          field_key: fieldKey,
   540→          field_label: def?.label || fieldKey,
   541→          extracted_value: values[idx],
   542→          confidence: 85,
   543→        });
   544→      }
   545→    });
   546→  } else {
   547→    // Key-value CSV: "Invoice Number,INV-001" format
   548→    for (const line of lines) {
   549→      const parts = line.split(delimiter).map(p => p.trim().replace(/^["']|["']$/g, ''));
   550→      if (parts.length >= 2) {
   551→        const key = parts[0].toLowerCase().replace(/[^a-z0-9]/g, '');
   552→        const value = parts[1];
   553→        const fieldKey = headerMap[key];
   554→        if (fieldKey && value) {
   555→          const def = FIELD_DEFINITIONS.find(d => d.key === fieldKey);
   556→          fields.push({
   557→            field_key: fieldKey,
   558→            field_label: def?.label || fieldKey,
   559→            extracted_value: value,
   560→            confidence: 80,
   561→          });
   562→        }
   563→      }
   564→    }
   565→  }
   566→
   567→  return fields;
   568→}

   791→  if (fields.length < 3) {

   793→    if (!fields.find(f => f.field_key === "invoiceNo")) {
   794→      const invMatch = fullText.match(/\b(INV[\-A-Z0-9]+)\b/i);

   805→      }
   806→    }

   810→      const htsMatch = fullText.match(/\b(\d{4}\.\d{2}\.\d{4})\b/);

   812→        fields.push({ field_key: "htsCode", field_label: "HTS Code", extracted_value: htsMatch[1], confidence: 75 });
   813→      }
   814→    }
   815→
   816→    // Look for bare country code (2 uppercase letters on their own line)
   817→    if (!fields.find(f => f.field_key === "countryOfOrigin")) {
   818→      const countryMatch = fullText.match(/(?:^|\n)\s*([A-Z]{2})\s*(?:\n|$)/m);
   819→      if (countryMatch) {
   820→        fields.push({ field_key: "countryOfOrigin", field_label: "Country of Origin", extracted_value: countryMatch[1], confidence: 60 });
   821→      }
   822→    }
   823→
   824→    // Look for bare weight
   825→    if (!fields.find(f => f.field_key === "netWeight")) {
   826→      const weightMatch = fullText.match(/(\d[\d,]*\.?\d*\s*(?:lbs?|kg|kgs|pounds?))/i);
   827→      if (weightMatch) {
   828→        fields.push({ field_key: "netWeight", field_label: "Net Weight", extracted_value: weightMatch[1], confidence: 65 });
   829→      }
   830→    }
   831→
   832→    // Look for company name (first line that's not a keyword/number)
   833→    if (!fields.find(f => f.field_key === "shipper")) {
   834→      for (const line of lines) {
   835→        const trimmed = line.trim();
   836→        // Skip empty lines, numbers, dates, currency values, keywords
   837→        if (trimmed.length < 3 || trimmed.length > 60) continue;
   838→        if (/^\d/.test(trimmed)) continue;
   839→        if (/[$€£¥]/.test(trimmed)) continue;
   840→        if (/^(invoice|inv|date|shipper|consignee|total|hts|net|gross|country|carrier|port|bill)/i.test(trimmed)) continue;
   841→        if (/^(INV|SHIP|DOC)/i.test(trimmed)) continue;
   842→        // Looks like a company name
   843→        if (/^[A-Z][a-zA-Z\s&.,]+$/.test(trimmed)) {
   844→          fields.push({ field_key: "shipper", field_label: "Shipper/Exporter", extracted_value: trimmed, confidence: 55 });
   845→          break;
   846→        }
   847→      }
   848→    }
   849→  }
   850→
   851→  return fields;
   852→}

   854→// --- PDF text-layer extraction (Tier 2) ---------------------------------------
   855→// Extracts embedded text from PDF files that have a real text layer
   856→// (not scanned images). Uses a simple regex-based approach to find text
   857→// between BT/ET markers in the PDF content stream.
   858→function extractPdfTextLayer(arrayBuffer: ArrayBuffer): string | null {
   859→  try {
   860→    const bytes = new Uint8Array(arrayBuffer);
   861→    // Check if it's actually a PDF
   862→    const header = new TextDecoder().decode(bytes.slice(0, 5));
   863→    if (header !== '%PDF-') return null;
   864→
   865→    // Extract text from PDF content streams
   866→    // Look for text in parentheses within BT...ET blocks
   867→    const text = new TextDecoder('latin1').decode(bytes);
   868→    const textMatches: string[] = [];
   869→
   870→    // Match text in parentheses (PDF text objects)
   871→    const textRegex = /\(([^)]+)\)/g;
   872→    let match;
   873→    while ((match = textRegex.exec(text)) !== null) {
   874→      const t = match[1];
   875→      // Filter out non-text content (binary data, short strings)
   876→      if (t.length > 2 && /[a-zA-Z]/.test(t) && !/\\[nrt]/.test(t)) {
   877→        textMatches.push(t);
   878→      }
   879→    }
   880→
   881→    if (textMatches.length < 3) return null; // Not enough text to be useful
   882→
   883→    // Join and clean up
   884→    const result = textMatches.join('\n');
   885→    return result.length > 50 ? result : null;
   886→  } catch {
   887→    return null;
   888→  }
   889→}

   891→// --- Tesseract OCR (Tier 3) ------------------------------------------------
   892→// Self-hosted tesseract.js running in a separate Node.js API route
   893→// (/api/internal/ocr) inside the Next.js app. The edge function (Deno) can't
   894→// host tesseract.js directly, so it base64-encodes the file and ships it over
   895→// HTTPS to the Node route, which runs OCR and returns the text.
   896→//
   897→// Config: set OCR_SERVICE_URL + INTERNAL_OCR_SECRET as Supabase secrets. If
   898→// either is missing, this function returns null and the cascade falls through
   899→// to Tier 4 (needs_manual_review) — it NEVER throws.
   900→//
   901→// The 25s timeout here is longer than the edge function's 18s wall-clock
   902→// budget on purpose: the budget will kick in first if needed, which is the
   903→// correct behaviour (budget is the global ceiling, the per-call timeout is
   904→// the local safety net for misconfigured / hung services).
   905→async function tesseractOCR(arrayBuffer: ArrayBuffer, mimeType: string): Promise<string | null> {
   906→  const ocrUrl = Deno.env.get("OCR_SERVICE_URL");
   907→  const secret = Deno.env.get("INTERNAL_OCR_SECRET");
   908→  if (!ocrUrl || !secret) return null; // misconfigured — falls through to Tier 4, never crashes
   909→  try {
   910→    const res = await fetch(ocrUrl, {
   911→      method: "POST",
   912→      headers: { "Content-Type": "application/json", "X-Internal-Secret": secret },
   913→      body: JSON.stringify({ data: bufToBase64(arrayBuffer), mimeType }),
   914→      signal: AbortSignal.timeout(25000),
   915→    });
   916→    if (!res.ok) {
   917→      console.error(`[extract-document] OCR service returned ${res.status}`);
   918→      return null;
   919→    }
   920→    const { text } = await res.json();
   921→    return text || null;
   922→  } catch (err) {
   923→    console.error("[extract-document] OCR service call failed", err);
   924→    return null; // never throw — Tier 4 manual-review is the safety net
   925→  }
   926→}

   928→// --- Main handler with 4-tier extraction fallback chain ----------------------
   929→Deno.serve(async (req) => {
   930→  if (req.method === "OPTIONS") {
   931→    return new Response("ok", { headers: corsHeaders });
   932→  }
   933→
   934→  try {
   935→    // 1. Verify JWT
   936→    const userClient = createUserClient(req);
   937→    if (!userClient) return jsonRes({ error: "Missing Authorization header" }, 401);
   938→    const user = await getUser(userClient);
   939→    if (!user) return jsonRes({ error: "Unauthorized — invalid JWT" }, 401);
   940→
   941→    // 1b. Rate limit check — 50 extractions per hour per org
   942→    // Get the user's org from organization_members
   943→    const { data: orgMember } = await userClient
   944→      .from("organization_members")
   945→      .select("org_id")
   946→      .eq("user_id", user.id)
   947→      .limit(1)
   948→      .maybeSingle();
   949→
   950→    if (orgMember?.org_id) {
   951→      const { data: allowed, error: rlError } = await userClient.rpc("check_extraction_rate_limit", {
   952→        p_org_id: orgMember.org_id,
   953→        p_max_requests: 50,
   954→      });
   955→      if (rlError || allowed === false) {
   956→        return jsonRes(
   957→          { error: "Rate limit exceeded. Maximum 50 extractions per hour per organization. Please try again later." },
   958→          429,
   959→        );
   960→      }
   961→    }
   962→
   963→    // 2. Parse JSON body
   964→    let body: any = {};
   965→    try {
   966→      body = await req.json();
   967→    } catch {
   968→      return jsonRes({ error: "Invalid JSON body" }, 400);
   969→    }
   970→    const { shipmentId, documentId } = body;
   971→    if (!shipmentId) {
   972→      return jsonRes({ error: "Missing 'shipmentId'" }, 400);
   973→    }
   974→
   975→    // 3. Fetch documents for shipment
   976→    let docQuery = userClient
   977→      .from("documents")
   978→      .select("id, shipment_id, doc_type, file_name, storage_path, mime_type")
   979→      .eq("shipment_id", shipmentId);
   980→
   981→    if (documentId) {
   982→      docQuery = docQuery.eq("id", documentId);
   983→    }
   984→
   985→    const { data: docs, error: docsErr } = await docQuery.order("uploaded_at", {
   986→      ascending: true,
   987→    });
   988→
   989→    if (docsErr) {
   990→      console.error("[extract-document] docs query error:", docsErr);
   991→      return jsonRes(
   992→        { error: "Failed to fetch documents", detail: docsErr.message },
   993→        500
   994→      );
   995→    }
   996→    if (!docs || docs.length === 0) {
   997→      return jsonRes(
   998→        { error: "No documents found for this shipment" },
   999→        404
  1000→      );
  1001→    }
  1002→
  1003→    const admin = createAdminClient();
  1004→    const ai = getGeminiClient();
  1005→
  1006→    if (!ai) {
  1007→      console.warn(
  1008→        "[extract-document] GEMINI_API_KEY not set — using mock extraction"
  1009→      );
  1010→    }
  1011→
  1012→    // ─── GLOBAL WALL-CLOCK BUDGET ───
  1013→    // 18 seconds total across ALL documents and ALL tiers. If we blow past it,
  1014→    // every remaining document is dropped straight to needs_manual_review with
  1015→    // a clear "extraction timeout" message — never a silent multi-minute retry.
  1016→    // The budget is shared across the whole shipment so a single huge PDF can't
  1017→    // starve the rest. Configurable via EXTRACTION_BUDGET_MS env (for tests).
  1018→    const BUDGET_MS = Number(Deno.env.get("EXTRACTION_BUDGET_MS") || 18000);
  1019→    const deadline = Date.now() + BUDGET_MS;
  1020→    let budgetExhausted = false;
  1021→    const budgetRemaining = () => deadline - Date.now();
  1022→
  1023→    const allFields: any[] = [];
  1024→    const perDocResults: any[] = [];
  1025→    let latestShipper: string | null = null;
  1026→    let latestConsignee: string | null = null;
  1027→    let geminiDebug: any = null;
  1028→    // Track whether extraction failed completely for ANY document so we can
  1029→    // return a clear error response at the end (rather than silently succeeding
  1030→    // with zero fields).
  1031→    let totalExtractionFailures = 0;
  1032→    const failureDetails: any[] = [];
  1033→
  1034→    // 4. Process each document through the 4-tier extraction fallback chain
  1035→    for (const doc of docs) {
  1036→      let extracted: any[] = [];
  1037→      let usedModel: string | null = null;
  1038→      let rawAiResponse: string | null = null;
  1039→      let extractionSource: string = "none";
  1040→      let extractionTier: number = 0;
  1041→      let docBudgetExhausted = false;
  1042→
  1043→      // ── Global budget check before starting this document ──
  1044→      if (budgetRemaining() <= 500) {
  1045→        budgetExhausted = true;
  1046→        docBudgetExhausted = true;
  1047→        console.warn(`[extract-document] BUDGET EXHAUSTED before doc ${doc.file_name} (${budgetRemaining()}ms left) — routing to manual review`);
  1048→      }
  1049→
  1050→      // Mark document as 'extracting'
  1051→      await admin.from("documents").update({
  1052→        processing_status: "extracting",
  1053→        processing_started_at: new Date().toISOString(),
  1054→      }).eq("id", doc.id);
  1055→
  1056→      // 4a. Download the file from Storage
  1057→      const { data: fileData, error: downloadErr } = await admin.storage
  1058→        .from("documents")
  1059→        .download(doc.storage_path);
  1060→
  1061→      if (downloadErr || !fileData) {
  1062→        console.error("[extract-document] file download failed for", doc.storage_path, downloadErr);
  1063→        await admin.from("documents").update({ processing_status: "failed" }).eq("id", doc.id);
  1064→        perDocResults.push({ documentId: doc.id, file_name: doc.file_name, status: "download_failed", fields: [] });
  1065→        continue;
  1066→      }
  1067→
  1068→      const ab = await fileData.arrayBuffer();
  1069→      const base64 = bufToBase64(ab);
  1070→      const mimeType = doc.mime_type || "application/octet-stream";
  1071→
  1072→      // Extract raw text for text-based files (used by regex fallback)
  1073→      let rawText: string | undefined;
  1074→      if (mimeType.startsWith("text/")) {
  1075→        try { rawText = new TextDecoder().decode(ab); } catch {}
  1076→      }
  1077→
  1078→      // ─── TIER 1: Gemini Vision (primary) ───
  1079→      if (ai && !docBudgetExhausted) {
  1080→        extractionTier = 1;
  1081→        const geminiResult = await callGeminiExtraction(ai, mimeType, base64, rawText, deadline);
  1082→        extracted = geminiResult.fields;
  1083→        geminiDebug = geminiResult.debug;
  1084→        usedModel = geminiResult.model;
  1085→        rawAiResponse = geminiResult.rawResponse;
  1086→        if (geminiResult.budgetExhausted) {
  1087→          docBudgetExhausted = true;
  1088→          budgetExhausted = true;
  1089→          console.warn(`[extract-document] BUDGET EXHAUSTED during Gemini cascade for ${doc.file_name} — dropping to manual review`);
  1090→        }
  1091→        if (extracted.length > 0) {
  1092→          extractionSource = usedModel || "gemini";
  1093→        }
  1094→      }
  1095→
  1096→      // ─── TIER 2: PDF text-layer extraction (if Gemini failed and it's a PDF) ───
  1097→      if (extracted.length === 0 && !docBudgetExhausted && mimeType === "application/pdf") {
  1098→        extractionTier = 2;
  1099→        console.log("[extract-document] Tier 2: trying PDF text-layer extraction");
  1100→        const pdfText = extractPdfTextLayer(ab);
  1101→        if (pdfText) {
  1102→          rawText = pdfText; // Use the extracted PDF text for regex
  1103→          console.log("[extract-document] PDF text-layer found, running regex on it");
  1104→        }
  1105→      }
  1106→
  1107→      // ─── TIER 3: Tesseract OCR (self-hosted, free) ───
  1108→      // Calls the /api/internal/ocr Node route which runs tesseract.js.
  1109→      // Returns null silently when OCR_SERVICE_URL/INTERNAL_OCR_SECRET are
  1110→      // unset or the service is unreachable — Tier 4 is the safety net.
  1111→      if (extracted.length === 0 && !docBudgetExhausted && !rawText && budgetRemaining() > 500) {
  1112→        extractionTier = 3;
  1113→        console.log("[extract-document] Tier 3: trying Tesseract OCR");
  1114→        const tesseractText = await tesseractOCR(ab, mimeType);
  1115→        if (tesseractText) {
  1116→          rawText = tesseractText;
  1117→          console.log("[extract-document] Tesseract produced text, running regex on it");
  1118→        }
  1119→      }
  1120→
  1121→      // ─── Regex extraction on whatever text we have (works for Tiers 2, 3) ───
  1122→      if (extracted.length === 0 && !docBudgetExhausted && rawText) {
  1123→        console.log(`[extract-document] Running regex extraction (tier ${extractionTier})`);
  1124→        extracted = regexExtract(rawText);
  1125→        if (extracted.length > 0) {
  1126→          extractionSource = extractionTier === 2 ? "pdf_text_layer" :
  1127→                             extractionTier === 3 ? "tesseract" : "regex_fallback";
  1128→        }
  1129→      }
  1130→
  1131→      // ─── TIER 4: Mark as 'needs_manual_review' — NEVER silent zero ───
  1132→      // This branch handles BOTH (a) all tiers genuinely failing and (b) the
  1133→      // wall-clock budget being exhausted. In the budget case we use a clear
  1134→      // "extraction timeout" reason so the user knows WHY it was dropped.
  1135→      if (extracted.length === 0 || docBudgetExhausted) {
  1136→        extractionTier = 4;
  1137→        extractionSource = docBudgetExhausted ? "timeout_manual_review" : "needs_manual_review";
  1138→
  1139→        const reviewReason = docBudgetExhausted
  1140→          ? `Extraction timed out after ${BUDGET_MS / 1000}s wall-clock budget — the document could not be processed within the time limit. Please review manually.`
  1141→          : `All extraction tiers failed for ${doc.file_name}. Document requires manual review.`;
  1142→
  1143→        // Mark the document for manual review
  1144→        await admin.from("documents").update({
  1145→          processing_status: "needs_manual_review",
  1146→          extraction_source: extractionSource,
  1147→        }).eq("id", doc.id);
  1148→
  1149→        // Create an exception for manual review
  1150→        await userClient.from("exceptions").insert({
  1151→          shipment_id: shipmentId,
  1152→          org_id: orgMember?.org_id || null,
  1153→          user_id: user.id,
  1154→          field_key: "_document",
  1155→          field_name: docBudgetExhausted
  1156→            ? `Extraction timeout — manual review required: ${doc.file_name}`
  1157→            : `Document needs manual review: ${doc.file_name}`,
  1158→          original_value: "",
  1159→          extracted_value: "",
  1160→          confidence: 0,
  1161→          reason: reviewReason,
  1162→          exception_type: "missing_field",
  1163→          doc_type: doc.doc_type || "Unknown",
  1164→          status: "Unresolved",
  1165→          history: [],
  1166→        }).then(({ error }: any) => {
  1167→          if (error) console.warn("[extract-document] failed to create manual review exception:", error.message);
  1168→        });
  1169→
  1170→        totalExtractionFailures++;
  1171→        failureDetails.push({
  1172→          documentId: doc.id,
  1173→          file_name: doc.file_name,
  1174→          reason: reviewReason,
  1175→          budgetExhausted: docBudgetExhausted,
  1176→          tiersTried: [1, 2, 3],
  1177→          modelsTried: geminiDebug?.modelsTried || [],
  1178→          errors: geminiDebug?.errors || [],
  1179→        });
  1180→
  1181→        perDocResults.push({
  1182→          documentId: doc.id,
  1183→          file_name: doc.file_name,
  1184→          doc_type: doc.doc_type,
  1185→          status: docBudgetExhausted ? "timeout_manual_review" : "needs_manual_review",
  1186→          extractionSource,
  1187→          extractionTier: 4,
  1188→          fieldsCount: 0,
  1189→          fields: [],
  1190→        });
  1191→
  1192→        // Audit log the manual review flag
  1193→        await userClient.from("audit_logs").insert({
  1194→          shipment_id: shipmentId,
  1195→          user_id: user.id,
  1196→          text: docBudgetExhausted
  1197→            ? `[extraction] Document ${doc.file_name} dropped to manual review — ${BUDGET_MS / 1000}s wall-clock budget exhausted`
  1198→            : `[extraction] Document ${doc.file_name} marked for manual review — all 4 tiers failed`,
  1199→          type: "warning",
  1200→        }).then(({ error }: any) => { if (error) console.warn("[extract-document] audit log failed:", error.message); });
  1201→
  1202→        continue;
  1203→      }
  1204→
  1205→      // ─── Success: mark document as completed ───
  1206→      await admin.from("documents").update({
  1207→        processing_status: "completed",
  1208→        extraction_source: extractionSource,
  1209→      }).eq("id", doc.id);
  1210→
  1211→      // 4a-bis. Persist raw AI response to audit_logs for compliance/debug.
  1212→      // We truncate to 500 chars to avoid bloating the audit log.
  1213→      if (rawAiResponse) {
  1214→        const truncated = rawAiResponse.length > 500
  1215→          ? rawAiResponse.substring(0, 500) + `…(+${rawAiResponse.length - 500} chars)`
  1216→          : rawAiResponse;
  1217→        await userClient.from("audit_logs").insert({
  1218→          shipment_id: shipmentId,
  1219→          user_id: user.id,
  1220→          text: `Gemini raw response (model: ${usedModel}, fields: ${extracted.length}) — doc ${doc.file_name}: ${truncated}`,
  1221→          type: "info",
  1222→        }).then(({ error: alErr }: any) => {
  1223→          if (alErr) {
  1224→            console.warn("[extract-document] audit_log insert failed for raw response:", alErr.message);
  1225→          }
  1226→        });
  1227→      }
  1228→
  1229→      // 4a-ter. If Gemini AND regex both returned nothing, record the failure.
  1230→      // We still proceed to write whatever fields we DID get (likely none for
  1231→      // this doc), and continue with the next document. We surface the
  1232→      // aggregate failure at the end of the response.
  1233→      if (extracted.length === 0) {
  1234→        totalExtractionFailures++;
  1235→        failureDetails.push({
  1236→          documentId: doc.id,
  1237→          file_name: doc.file_name,
  1238→          reason: "no fields extracted from any model or regex",
  1239→          modelsTried: geminiDebug?.modelsTried || [],
  1240→          errors: geminiDebug?.errors || [],
  1241→        });
  1242→        perDocResults.push({
  1243→          documentId: doc.id,
  1244→          file_name: doc.file_name,
  1245→          doc_type: doc.doc_type,
  1246→          status: "extraction_failed",
  1247→          fieldsCount: 0,
  1248→          fields: [],
  1249→        });
  1250→        continue;
  1251→      }
  1252→
  1253→      // 4b. Normalize + write to document_fields
  1254→      const writtenFields: any[] = [];
  1255→      for (const f of extracted) {
  1256→        const key = String(f.field_key || "").trim();
  1257→        const value = String(f.extracted_value ?? "").trim();
  1258→        if (!key || !value) continue;
  1259→
  1260→        // Map label if missing
  1261→        const def = FIELD_DEFINITIONS.find((d) => d.key === key);
  1262→        const label = String(f.field_label || def?.label || key);
  1263→        const conf = Math.max(
  1264→          0,
  1265→          Math.min(100, Number(f.confidence ?? 70) || 70)
  1266→        );
  1267→
  1268→        const insertRow = {
  1269→          document_id: doc.id,
  1270→          shipment_id: shipmentId,
  1271→          user_id: user.id,
  1272→          field_key: key,
  1273→          field_label: label,
  1274→          extracted_value: value,
  1275→          confidence: conf,
  1276→          is_flagged: false,
  1277→          validation_errors: [],
  1278→          bounding_box: f.bounding_box || null,
  1279→          extraction_source: extractionSource,
  1280→        };
  1281→
  1282→        const { data: inserted, error: insErr } = await userClient
  1283→          .from("document_fields")
  1284→          .insert(insertRow)
  1285→          .select("id, field_key, field_label, extracted_value, confidence")
  1286→          .single();
  1287→
  1288→        if (insErr) {
  1289→          console.warn(
  1290→            "[extract-document] field insert failed:",
  1291→            key,
  1292→            insErr.message
  1293→          );
  1294→          continue;
  1295→        }
  1296→
  1297→        writtenFields.push(inserted);
  1298→        allFields.push(inserted);
  1299→
  1300→        // Track shipper / consignee for shipment row update
  1301→        if (key === "shipper" && !latestShipper) latestShipper = value;
  1302→        if (key === "consignee" && !latestConsignee) latestConsignee = value;
  1303→      }
  1304→
  1305→      perDocResults.push({
  1306→        documentId: doc.id,
  1307→        file_name: doc.file_name,
  1308→        doc_type: doc.doc_type,
  1309→        status: "extracted",
  1310→        extractionSource,
  1311→        model: usedModel,
  1312→        fieldsCount: writtenFields.length,
  1313→        fields: writtenFields,
  1314→      });
  1315→    }
  1316→
  1317→    // 5. Update shipment shipper / consignee if extracted
  1318→    const shipUpdate: any = {};
  1319→    if (latestShipper) shipUpdate.shipper = latestShipper;
  1320→    if (latestConsignee) shipUpdate.consignee = latestConsignee;
  1321→
  1322→    if (Object.keys(shipUpdate).length > 0) {
  1323→      const { error: shipUpdErr } = await userClient
  1324→        .from("shipments")
  1325→        .update(shipUpdate)
  1326→        .eq("id", shipmentId);
  1327→      if (shipUpdErr) {
  1328→        console.warn(
  1329→          "[extract-document] shipment shipper/consignee update failed:",
  1330→          shipUpdErr
  1331→        );
  1332→      }
  1333→    }
  1334→
  1335→    // 6. Audit log
  1336→    await userClient.from("audit_logs").insert({
  1337→      shipment_id: shipmentId,
  1338→      user_id: user.id,
  1339→      text: `Gemini extraction completed for ${docs.length} document(s) — ${allFields.length} fields extracted.`,
  1340→      type: "success",
  1341→    });
  1342→
  1343→    // 7. Respond — the response mirrors ExtractDocumentResponse shape.
  1344→    // If ALL models failed AND regex returned nothing for every document,
  1345→    // return a clear error instead of silently succeeding with 0 fields.
  1346→    const totalDocsAttempted = docs.length;
  1347→    if (
  1348→      ai &&
  1349→      allFields.length === 0 &&
  1350→      totalExtractionFailures === totalDocsAttempted &&
  1351→      totalDocsAttempted > 0
  1352→    ) {
  1353→      // Every document failed extraction — surface a clear, actionable error.
  1354→      await userClient.from("audit_logs").insert({
  1355→        shipment_id: shipmentId,
  1356→        user_id: user.id,
  1357→        text: `Extraction failed: all models unavailable across ${totalDocsAttempted} document(s).`,
  1358→        type: "error",
  1359→      });
  1360→
  1361→      return jsonRes(
  1362→        {
  1363→          success: false,
  1364→          error: "Extraction failed: all models unavailable",
  1365→          details: {
  1366→            documentsAttempted: totalDocsAttempted,
  1367→            failures: failureDetails,
  1368→            debug: geminiDebug,
  1369→          },
  1370→        },
  1371→        502
  1372→      );
  1373→    }
  1374→
  1375→    return jsonRes({
  1376→      success: true,
  1377→      shipmentId,
  1378→      documentId: documentId || docs[0]?.id,
  1379→      fields: allFields.map((f) => ({
  1380→        field_key: f.field_key,
  1381→        field_label: f.field_label,
  1382→        extracted_value: f.extracted_value,
  1383→        confidence: f.confidence,
  1384→        bounding_box: f.bounding_box || null,
  1385→      })),
  1386→      documents: perDocResults,
  1387→      shipper: latestShipper || undefined,
  1388→      consignee: latestConsignee || undefined,
  1389→      geminiUsed: !!ai,
  1390→      debug: geminiDebug,
  1391→      // Surface the wall-clock budget exhaustion so the frontend can show a
  1392→      // clear "extraction timed out" message instead of a generic failure.
  1393→      budgetExhausted,
  1394→      budgetMs: BUDGET_MS,
  1395→      ...(totalExtractionFailures > 0
  1396→        ? {
  1397→            partialFailure: {
  1398→              failedDocuments: totalExtractionFailures,
  1399→              totalDocuments: totalDocsAttempted,
  1400→              details: failureDetails,
  1401→              budgetExhausted,
  1402→            },
  1403→          }
  1404→        : {}),
  1405→    });
  1406→  } catch (err: any) {
  1407→    console.error("[extract-document] unhandled error:", err);
  1408→    return jsonRes(
  1409→      { error: "Internal server error", detail: String(err?.message || err) },
  1410→      500
  1411→    );
  1412→  }
  1413→});
  1414→

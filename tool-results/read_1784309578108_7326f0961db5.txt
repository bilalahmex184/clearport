    31→    31→// ============================================================================
    32→    32→// Context Type
    33→    33→// ============================================================================
    34→    34→
    35→    35→export type SupabaseStatus = 'connected' | 'unconfigured' | 'error_tables' | 'loading';
    36→    36→export type EdgeFunctionStatus = 'live' | 'fallback' | 'unknown';
    37→    37→
    38→    38→interface ClearPortContextType {
    39→    39→  entries: ShipmentEntry[];
    40→    40→  selectedEntryId: string;
    41→    41→  selectedEntry: ShipmentEntry | undefined;
    42→    42→  selectedExceptionId: string;
    43→    43→  selectedException: Exception | undefined;
    44→    44→  activeTab: string;
    45→    45→  rules: OperationalRules;
    46→    46→  undoStack: { entryId: string; exceptionId: string; previousState: Exception }[];
    47→    47→  auditLogs: AuditLog[];
    48→    48→  theme: 'dark' | 'light';
    49→    49→  supabaseStatus: SupabaseStatus;
    50→    50→  edgeFunctionStatus: EdgeFunctionStatus;
    51→    51→  currentUser: string;
    52→    52→  currentTime: string;
    53→    53→  // RBAC role for the current user — fetched from /api/organizations on
    54→    54→  // load (reflects the user's role in their active org). Defaults to
    55→    55→  // 'operator' so the no-login UX continues to work before the role is
    56→    56→  // resolved. Components use the can* helpers (canUpload, canResolve,
    57→    57→  // canManageRules, canExport) from rbac.service to gate their UI.
    58→    58→  userRole: UserRole;
    59→    59→  // Organizations the user belongs to (populated from
    60→    60→  // GET /api/organizations). Empty until the first successful fetch.
    61→    61→  userOrgs: Array<{ org_id: string; org_name: string; role: UserRole }>;
    62→    62→  // The currently active org id. Sent as the X-Org-Id header on every
    63→    63→  // API call so the backend resolves the same org context via
    64→    64→  // requireOrgRole(). Null until the first org list load completes.
    65→    65→  currentOrgId: string | null;
    66→    66→
    67→    67→  // API helper — same as apiFetch but injects X-Org-Id header
    68→    68→  apiFetchOrg: <T = any>(path: string, options?: RequestInit & { raw?: boolean }) => Promise<T>;
    69→    69→
    70→    70→  // Actions
    71→    71→  selectEntry: (id: string) => void;
    72→    72→  selectException: (id: string) => void;
    73→    73→  setActiveTab: (tab: string) => void;
    74→    74→  updateException: (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => void;
    75→    75→  undoLastAction: () => void;
    76→    76→  acceptAllHighConfidence: (entryId: string) => void;
    77→    77→  uploadDocuments: (files: File[]) => Promise<{ shipmentId: string; success: boolean; error?: string }>;
    78→    78→  updateRules: (newRules: Partial<OperationalRules>) => void;
    79→    79→  exportToCSV: (entryId: string) => Promise<void>;
    80→    80→  toggleTheme: () => void;
    81→    81→  refreshData: () => Promise<void>;
    82→    82→  // Switch the active org context. Updates currentOrgId + userRole, then
    83→    83→  // reloads all org-scoped data (shipments, rules, logs) via the API.
    84→    84→  switchOrg: (orgId: string) => void;
    85→    85→}
    86→    86→

    88→    88→
    89→    89→// ============================================================================
    90→    90→// Provider
    91→    91→// ============================================================================
    92→    92→
    93→    93→export const ClearPortProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
    94→    94→  const [entries, setEntries] = React.useState<ShipmentEntry[]>(seedEntries);
    95→    95→  const [selectedEntryId, setSelectedEntryId] = React.useState<string>('SHIP-2026-8802');
    96→    96→  const [selectedExceptionId, setSelectedExceptionId] = React.useState<string>('8802-hts');
    97→    97→  const [activeTab, setActiveTab] = React.useState<string>('exception-desk');
    98→    98→  const [theme, setTheme] = React.useState<'dark' | 'light'>('dark');
    99→    99→  const [rules, setRules] = React.useState<OperationalRules>(seedRules);
   100→   100→  const [undoStack, setUndoStack] = React.useState<{ entryId: string; exceptionId: string; previousState: Exception }[]>([]);
   101→   101→  const [auditLogs, setAuditLogs] = React.useState<AuditLog[]>(seedLogs);
   102→   102→  const [supabaseStatus, setSupabaseStatus] = React.useState<SupabaseStatus>(
   103→   103→    isSupabaseConfigured() ? 'loading' : 'unconfigured'
   104→   104→  );
   105→   105→  const [edgeFunctionStatus, setEdgeFunctionStatus] = React.useState<EdgeFunctionStatus>('unknown');
   106→   106→  const [currentUser, setCurrentUser] = React.useState<string>('Broker');
   107→   107→  const [currentTime, setCurrentTime] = React.useState<string>('');
   108→   108→  // Initial value 'operator' preserves the no-login UX. The real role is
   109→   109→  // fetched from /api/organizations on load (see loadData below).
   110→   110→  const [userRole, setUserRole] = React.useState<UserRole>('operator');
   111→   111→  // Orgs the user belongs to + the currently active org id. Populated by
   112→   112→  // loadData on first run; switchOrg() updates currentOrgId and reloads.
   113→   113→  const [userOrgs, setUserOrgs] = React.useState<Array<{ org_id: string; org_name: string; role: UserRole }>>([]);
   114→   114→  const [currentOrgId, setCurrentOrgId] = React.useState<string | null>(null);
   115→   115→
   116→   116→  // Refs mirror the org state so loadData / apiFetchOrg / switchOrg can read
   117→   117→  // the latest values without being recreated on every state change. This
   118→   118→  // keeps loadData's deps stable ([]) so the initial mount effect only fires
   119→   119→  // once, and switchOrg can explicitly trigger a reload.
   120→   120→  const currentOrgIdRef = React.useRef<string | null>(null);
   121→   121→  currentOrgIdRef.current = currentOrgId;
   122→   122→  const userOrgsRef = React.useRef<typeof userOrgs>([]);
   123→   123→  userOrgsRef.current = userOrgs;
   124→   124→
   125→   125→  // --- apiFetch wrapper that injects the X-Org-Id header for org-scoped routes ---
   126→   126→  // Uses the ref so the callback identity is stable (no dependency on state).
   127→   127→  // Routes that DON'T need an org context (e.g. /api/organizations itself)
   128→   128→  // should call the base `apiFetch` directly.
   129→   129→  const apiFetchOrg = React.useCallback(
   130→   130→    <T = any>(path: string, options: RequestInit & { raw?: boolean } = {}): Promise<T> => {
   131→   131→      const headers: Record<string, string> = {
   132→   132→        ...((options.headers as Record<string, string>) || {}),
   133→   133→      };
   134→   134→      if (currentOrgIdRef.current) {
   135→   135→        headers['X-Org-Id'] = currentOrgIdRef.current;
   136→   136→      }
   137→   137→      return apiFetch<T>(path, { ...options, headers });
   138→   138→    },
   139→   139→    [],
   140→   140→  );
   141→   141→
   142→   142→  // --- Real-time clock ---
   143→   143→  React.useEffect(() => {
   144→   144→    const update = () => {
   145→   145→      const now = new Date();
   146→   146→      const utc = now.toISOString().slice(0, 16).replace('T', ' ') + ' UTC';
   147→   147→      setCurrentTime(utc);
   148→   148→    };
   149→   149→    update();
   150→   150→    const interval = setInterval(update, 1000);
   151→   151→    return () => clearInterval(interval);
   152→   152→  }, []);
   153→   153→
   154→   154→  // --- Load user email ---
   155→   155→  React.useEffect(() => {
   156→   156→    getCurrentUserEmail().then(email => setCurrentUser(email));
   157→   157→  }, []);
   158→   158→
   159→   159→  // --- Persist theme ---
   160→   160→  React.useEffect(() => {
   161→   161→    const saved = typeof window !== 'undefined' ? localStorage.getItem('clearport-theme') : null;
   162→   162→    if (saved === 'light' || saved === 'dark') setTheme(saved);
   163→   163→  }, []);
   164→   164→
   165→   165→  const toggleTheme = React.useCallback(() => {
   166→   166→    setTheme(prev => {
   167→   167→      const next = prev === 'dark' ? 'light' : 'dark';
   168→   168→      if (typeof window !== 'undefined') localStorage.setItem('clearport-theme', next);
   169→   169→      return next;
   170→   170→    });
   171→   171→  }, []);
   172→   172→
   173→   173→  // --- Load data ---
   174→   174→  const selectedEntryIdRef = React.useRef(selectedEntryId);
   175→   175→  selectedEntryIdRef.current = selectedEntryId;
   176→   176→
   177→   177→  const loadData = React.useCallback(async () => {
   178→   178→    if (!isSupabaseConfigured()) {
   179→   179→      setSupabaseStatus('unconfigured');
   180→   180→      setEdgeFunctionStatus('fallback');
   181→   181→      return;
   182→   182→    }
   183→   183→
   184→   184→    setSupabaseStatus('loading');
   185→   185→
   186→   186→    try {
   187→   187→      await ensureAuthenticated();
   188→   188→      const email = await getCurrentUserEmail();
   189→   189→      setCurrentUser(email);
   190→   190→
   191→   191→      // --- Bootstrap org context (only on first run, when currentOrgId is null) ---
   192→   192→      // Subsequent reloads (e.g. after switchOrg) skip this block and reuse
   193→   193→      // the already-selected org. Uses the base apiFetch (no X-Org-Id header)
   194→   194→      // because GET /api/organizations is the bootstrap path that doesn't
   195→   195→      // require an org context.
   196→   196→      if (!currentOrgIdRef.current) {
   197→   197→        try {
   198→   198→          const orgsRes = await apiFetch<{
   199→   199→            organizations: Array<{ org_id: string; org_name: string; role: UserRole }>;
   200→   200→          }>('/api/organizations');
   201→   201→
   202→   202→          const orgs = orgsRes?.organizations ?? [];
   203→   203→          if (orgs.length > 0) {
   204→   204→            setUserOrgs(orgs);
   205→   205→            const first = orgs[0];
   206→   206→            setCurrentOrgId(first.org_id);
   207→   207→            currentOrgIdRef.current = first.org_id;
   208→   208→            setUserRole(first.role);
   209→   209→          } else {
   210→   210→            // No org memberships — fall back to seed data with a warning.
   211→   211→            // The user can still see the demo data; they just can't mutate it.
   212→   212→            console.warn('[ctx] No org memberships found — falling back to seed data. Create or join an organization to enable live mode.');
   213→   213→            setSupabaseStatus('connected');
   214→   214→            setEdgeFunctionStatus('fallback');
   215→   215→            return;
   216→   216→          }
   217→   217→        } catch (orgErr) {
   218→   218→          // 403 (no org membership) or network / schema error — fall back to
   219→   219→          // seed data so the demo UX keeps working.
   220→   220→          console.warn('[ctx] /api/organizations failed, falling back to seed data:', orgErr);
   221→   221→          setSupabaseStatus('connected');
   222→   222→          setEdgeFunctionStatus('fallback');
   223→   223→          return;
   224→   224→        }
   225→   225→      }
   226→   226→
   227→   227→      // --- Fetch shipments for the active org (uses X-Org-Id header via apiFetchOrg) ---
   228→   228→      try {
   229→   229→        const response = await apiFetchOrg<{
   230→   230→          data: ShipmentEntry[];
   231→   231→          pagination: { total: number };
   232→   232→        }>('/api/shipments?page=1&limit=100');
   233→   233→
   234→   234→        if (response?.data && Array.isArray(response.data)) {
   235→   235→          // Only use DB shipments if they have fields/exceptions data (new
   236→   236→          // schema). Old-schema shipments have empty arrays — fall back to
   237→   237→          // seed data so the demo UX is preserved.
   238→   238→          const hasRealData = response.data.some(
   239→   239→            (s) => s.fields.length > 0 || s.exceptions.length > 0,
   240→   240→          );
   241→   241→
   242→   242→          if (hasRealData) {
   243→   243→            setEntries(response.data);
   244→   244→            // Select first shipment if current selection doesn't exist
   245→   245→            if (!response.data.find((s) => s.id === selectedEntryIdRef.current)) {
   246→   246→              const first = response.data[0];
   247→   247→              setSelectedEntryId(first.id);
   248→   248→              if (first.exceptions?.length > 0) {
   249→   249→                const unresolved = first.exceptions.find((e) => e.status === 'Unresolved');
   250→   250→                setSelectedExceptionId(unresolved ? unresolved.id : first.exceptions[0].id);
   251→   251→              }
   252→   252→            }
   253→   253→          }
   254→   254→          // else: keep seed entries (demo mode)
   255→   255→
   256→   256→          setEdgeFunctionStatus('live');
   257→   257→          setSupabaseStatus('connected');
   258→   258→
   259→   259→          // Fetch rules + logs in parallel via the org-scoped API routes.
   260→   260→          const [rulesRes, logsRes] = await Promise.all([
   261→   261→            apiFetchOrg<{ rules: OperationalRules }>('/api/rules').catch(() => null),
   262→   262→            apiFetchOrg<{ logs: AuditLog[] }>('/api/audit-logs').catch(() => null),
   263→   263→          ]);
   264→   264→          if (rulesRes?.rules) setRules(rulesRes.rules);
   265→   265→          if (logsRes?.logs && logsRes.logs.length > 0) setAuditLogs(logsRes.logs);
   266→   266→          return;
   267→   267→        }
   268→   268→      } catch (apiErr) {
   269→   269→        console.warn('[ctx] /api/shipments failed, falling back to seed data:', apiErr);
   270→   270→      }
   271→   271→
   272→   272→      // Fallback: seed data (demo mode)
   273→   273→      setSupabaseStatus('connected');
   274→   274→      setEdgeFunctionStatus('fallback');
   275→   275→    } catch (err) {
   276→   276→      console.error('[ctx] loadData error:', err);
   277→   277→      setSupabaseStatus('error_tables');
   278→   278→      setEdgeFunctionStatus('fallback');
   279→   279→    }
   280→   280→  }, [apiFetchOrg]);
   281→   281→
   282→   282→  // --- switchOrg: change the active org context and reload data ---
   283→   283→  const switchOrg = React.useCallback((orgId: string) => {
   284→   284→    const org = userOrgsRef.current.find((o) => o.org_id === orgId);
   285→   285→    if (!org) {
   286→   286→      console.warn('[ctx] switchOrg: org not found in userOrgs', { orgId });
   287→   287→      return;
   288→   288→    }
   289→   289→    // Update both state + ref so apiFetchOrg picks up the new org immediately
   290→   290→    // (the ref is read synchronously; the state triggers a re-render).
   291→   291→    setCurrentOrgId(orgId);
   292→   292→    currentOrgIdRef.current = orgId;
   293→   293→    setUserRole(org.role);
   294→   294→    // Reload all org-scoped data for the new org.
   295→   295→    loadData();
   296→   296→  }, [loadData]);
   297→   297→
   298→   298→  React.useEffect(() => {
   299→   299→    loadData();
   300→   300→  }, [loadData]);
   301→   301→
   302→   302→  const refreshData = React.useCallback(async () => {
   303→   303→    await loadData();
   304→   304→  }, [loadData]);
   305→   305→
   306→   306→  // --- Computed ---
   307→   307→  const selectedEntry = React.useMemo(
   308→   308→    () => entries.find(e => e.id === selectedEntryId),
   309→   309→    [entries, selectedEntryId]
   310→   310→  );
   311→   311→
   312→   312→  const selectedException = React.useMemo(() => {
   313→   313→    if (!selectedEntry) return undefined;
   314→   314→    return selectedEntry.exceptions.find(ex => ex.id === selectedExceptionId);
   315→   315→  }, [selectedEntry, selectedExceptionId]);
   316→   316→
   317→   317→  // --- Selection (fixes stale closure — uses functional update) ---
   318→   318→  const selectEntry = React.useCallback((id: string) => {
   319→   319→    setSelectedEntryId(id);
   320→   320→    setEntries(prev => {
   321→   321→      const entry = prev.find(e => e.id === id);
   322→   322→      if (entry && entry.exceptions.length > 0) {
   323→   323→        const unresolved = entry.exceptions.find(e => e.status === 'Unresolved');
   324→   324→        setSelectedExceptionId(unresolved ? unresolved.id : entry.exceptions[0].id);
   325→   325→      } else {
   326→   326→        setSelectedExceptionId('');
   327→   327→      }
   328→   328→      return prev;
   329→   329→    });
   330→   330→  }, []);
   331→   331→
   332→   332→  const selectException = React.useCallback((id: string) => {
   333→   333→    setSelectedExceptionId(id);
   334→   334→  }, []);
   335→   335→
   336→   336→  // --- Audit log helper ---
   337→   337→  const addAuditLog = React.useCallback((text: string, type: AuditLog['type'] = 'info', shipmentId?: string) => {
   338→   338→    const newLog: AuditLog = {
   339→   339→      id: typeof crypto !== 'undefined' ? crypto.randomUUID() : `log-${Date.now()}`,
   340→   340→      text,
   341→   341→      timestamp: new Date().toISOString(),
   342→   342→      type,
   343→   343→      shipmentId,
   344→   344→    };
   345→   345→    setAuditLogs(prev => [newLog, ...prev]);
   346→   346→
   347→   347→    if (isSupabaseConfigured() && supabase) {
   348→   348→      supabase.from('audit_logs').insert({
   349→   349→        id: newLog.id,
   350→   350→        text: newLog.text,
   351→   351→        timestamp: newLog.timestamp,
   352→   352→        type: newLog.type,
   353→   353→        shipment_id: shipmentId || null,
   354→   354→      }).then(({ error }) => {
   355→   355→        if (error) console.warn('[ctx] audit log insert failed:', error.message);
   356→   356→      });
   357→   357→    }
   358→   358→  }, []);
   359→   359→
   360→   360→  // --- Update exception (fixes setState-in-setState) ---
   361→   361→  const updateException = React.useCallback(
   362→   362→    (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => {
   363→   363→      let previousException: Exception | undefined;
   364→   364→      let updatedEntry: ShipmentEntry | undefined;
   365→   365→
   366→   366→      setEntries(prevEntries => {
   367→   367→        return prevEntries.map(entry => {
   368→   368→          if (entry.id !== entryId) return entry;
   369→   369→
   370→   370→          const prev = entry.exceptions.find(ex => ex.id === exceptionId);
   371→   371→          if (!prev) return entry;
   372→   372→          previousException = { ...prev };
   373→   373→
   374→   374→          const updatedExceptions = entry.exceptions.map(ex => {
   375→   375→            if (ex.id !== exceptionId) return ex;
   376→   376→
   377→   377→            const oldVal = ex.correctedValue || ex.extractedValue;
   378→   378→            const updatedVal = newValue !== undefined ? newValue : ex.extractedValue;
   379→   379→
   380→   380→            const historyItem: ExceptionHistoryEntry = {
   381→   381→              user: currentUser,
   382→   382→              oldValue: oldVal,
   383→   383→              newValue: updatedVal,
   384→   384→              timestamp: new Date().toISOString(),
   385→   385→              action: status,
   386→   386→            };
   387→   387→
   388→   388→            return {
   389→   389→              ...ex,
   390→   390→              status: status as ExceptionStatus,
   391→   391→              correctedValue: status === 'Corrected' ? newValue : undefined,
   392→   392→              history: [historyItem, ...ex.history],
   393→   393→              resolvedAt: new Date().toISOString(),
   394→   394→              resolvedBy: currentUser,
   395→   395→            };
   396→   396→          });
   397→   397→
   398→   398→          const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   399→   399→          const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   400→   400→          const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   401→   401→
   402→   402→          const updatedFields = entry.fields.map(f => {
   403→   403→            if (f.exceptionId === exceptionId) {
   404→   404→              const val = status === 'Corrected' ? (newValue || f.value) : f.value;
   405→   405→              return { ...f, value: val, isFlagged: false };
   406→   406→            }
   407→   407→            return f;
   408→   408→          });
   409→   409→
   410→   410→          updatedEntry = {
   411→   411→            ...entry,
   412→   412→            status: newStatus,
   413→   413→            exceptions: updatedExceptions,
   414→   414→            fields: updatedFields,
   415→   415→            currentConfidence: newConfidence,
   416→   416→          };
   417→   417→
   418→   418→          return updatedEntry;
   419→   419→        });
   420→   420→      });
   421→   421→
   422→   422→      // Side effects AFTER setState
   423→   423→      if (previousException && updatedEntry) {
   424→   424→        setUndoStack(prevStack => [
   425→   425→          { entryId, exceptionId, previousState: { ...previousException! } },
   426→   426→          ...prevStack,
   427→   427→        ]);
   428→   428→
   429→   429→        const logText = `Exception "${previousException.fieldName}" for ${entryId} was ${status.toLowerCase()}${
   430→   430→          status === 'Corrected' ? ` to "${newValue}"` : ''
   431→   431→        } by ${currentUser}.`;
   432→   432→        addAuditLog(logText, status === 'Rejected' ? 'warning' : 'success', entryId);
   433→   433→
   434→   434→        // Persist via the /api/exceptions/:id route (handles exception update,
   435→   435→        // document_field sync, shipment confidence recompute, and audit log
   436→   436→        // in one server-side transaction). Uses the org-scoped wrapper so
   437→   437→        // the X-Org-Id header is sent.
   438→   438→        if (isSupabaseConfigured()) {
   439→   439→          apiFetchOrg(`/api/exceptions/${exceptionId}`, {
   440→   440→            method: 'PATCH',
   441→   441→            body: JSON.stringify({
   442→   442→              status,
   443→   443→              correctedValue: status === 'Corrected' ? newValue : undefined,
   444→   444→            }),
   445→   445→          }).catch((err) => {
   446→   446→            console.warn('[ctx] exception update failed:', err instanceof Error ? err.message : err);
   447→   447→          });
   448→   448→        }
   449→   449→      }
   450→   450→    },
   451→   451→    [currentUser, addAuditLog, apiFetchOrg]

   889→   889→        }
   890→   890→      } catch (err) {
   891→   891→        // Outer catch — should never reach here since allSettled doesn't throw,
   892→   892→        // but if it does, persist the failure (§2: no silent catches)
   893→   893→        const errMsg = err instanceof Error ? err.message : String(err);
   894→   894→        console.error('[validation-chain] unexpected error:', errMsg);
   895→   895→        addAuditLog(`[pipeline_error] Unexpected validation error for ${shipmentId}: ${errMsg}`, 'error', shipmentId);
   896→   896→        try {
   897→   897→          await apiFetchOrg('/api/shipments/' + shipmentId, {
   898→   898→            method: 'PATCH',
   899→   899→            body: JSON.stringify({ validation_status: 'failed' }),
   900→   900→          });
   901→   901→        } catch (e) {
   902→   902→          console.error('[validation-chain] failed to set validation_status=failed:', e);
   903→   903→        }
   904→   904→      }
   905→   905→
   906→   906→      // Step 5: Reload from API to get the real DB state (with fields, exceptions, documents)
   907→   907→      if (extractedFields.length > 0) {
   908→   908→        // If we have extracted fields, build the entry from extraction response
   909→   909→        const newExceptions: Exception[] = [];
   910→   910→        const updatedFields = extractedFields.map(f => {
   911→   911→          let threshold = rules.invoiceThreshold;
   912→   912→          if (f.key === 'htsCode') threshold = rules.htsThreshold;
   913→   913→          else if (f.key === 'shipper' || f.key === 'consignee' || f.key === 'consigneeAddress') threshold = rules.partiesThreshold;
   914→   914→
   915→   915→          if (f.confidence < threshold) {
   916→   916→            const excId = crypto.randomUUID();
   917→   917→            const exc: Exception = {
   918→   918→              id: excId,
   919→   919→              fieldName: f.label,
   920→   920→              fieldKey: f.key,
   921→   921→              originalValue: f.value,
   922→   922→              extractedValue: f.value,
   923→   923→              confidence: f.confidence,
   924→   924→              reason: `Extracted confidence (${f.confidence}%) is below threshold (${threshold}%).`,
   925→   925→              exceptionType: 'low_confidence',
   926→   926→              docType: f.sourceDoc,
   927→   927→              boundingBox: f.boundingBox || { x: 10, y: 35, w: 32, h: 5 },
   928→   928→              status: 'Unresolved',
   929→   929→              history: [],
   930→   930→              createdAt: new Date().toISOString(),
   931→   931→            };
   932→   932→            newExceptions.push(exc);
   933→   933→            return { ...f, isFlagged: true, exceptionId: excId };
   934→   934→          }
   935→   935→          return f;
   936→   936→        });
   937→   937→
   938→   938→        const initialConfidence = Math.round(
   939→   939→          updatedFields.reduce((acc, f) => acc + f.confidence, 0) / Math.max(updatedFields.length, 1)
   940→   940→        );
   941→   941→
   942→   942→        const newEntry: ShipmentEntry = {
   943→   943→          id: shipmentId,
   944→   944→          shipper,
   945→   945→          consignee,
   946→   946→          status: 'Under Review',
   947→   947→          docsCount: files.length,
   948→   948→          urgency: '08:30:00',
   949→   949→          initialConfidence,
   950→   950→          currentConfidence: initialConfidence,
   951→   951→          createdAt: new Date().toISOString(),
   952→   952→          documents: uploadedDocuments,  // Use fetched documents, not empty array
   953→   953→          exceptions: newExceptions,
   954→   954→          fields: updatedFields,
   955→   955→        };
   956→   956→
   957→   957→        setEntries(prev => [newEntry, ...prev]);
   958→   958→        setSelectedEntryId(shipmentId);
   959→   959→        setSelectedExceptionId(newExceptions[0]?.id || '');
   960→   960→      } else {
   961→   961→        // No fields extracted — reload from API to get whatever the DB has
   962→   962→        await loadData();
   963→   963→        setSelectedEntryId(shipmentId);
   964→   964→      }
   965→   965→
   966→   966→      addAuditLog(
   967→   967→        `New entry ${shipmentId} ingestion completed: ${files.length} files, ${extractedFields.length} fields extracted.`,
   968→   968→        extractedFields.length > 0 ? 'success' : 'warning',
   969→   969→        shipmentId
   970→   970→      );
   971→   971→
   972→   972→      return { shipmentId, success: true };
   973→   973→    } catch (err) {
   974→   974→      const errMsg = err instanceof Error ? err.message : 'Upload pipeline failed';
   975→   975→      addAuditLog(`Upload failed for ${shipmentId}: ${errMsg}`, 'error', shipmentId);
   976→   976→      return { shipmentId, success: false, error: errMsg };
   977→   977→    }
   978→   978→  }, [rules, addAuditLog]);
   979→   979→
   980→   980→  // --- Update rules ---
   981→   981→  const updateRules = React.useCallback((newRules: Partial<OperationalRules>) => {
   982→   982→    setRules(prev => {
   983→   983→      const updated = { ...prev, ...newRules };
   984→   984→
   985→   985→      // Persist via the /api/rules route (upserts the org's rules row).
   986→   986→      // Uses the org-scoped wrapper so the X-Org-Id header is sent.
   987→   987→      if (isSupabaseConfigured()) {
   988→   988→        apiFetchOrg('/api/rules', {
   989→   989→          method: 'PATCH',
   990→   990→          body: JSON.stringify(newRules),
   991→   991→        }).catch((err) => {
   992→   992→          console.warn('[ctx] rules update failed:', err instanceof Error ? err.message : err);
   993→   993→        });
   994→   994→      }
   995→   995→
   996→   996→      return updated;
   997→   997→    });
   998→   998→    addAuditLog('Compliance operational thresholds updated.', 'info');
   999→   999→  }, [addAuditLog, apiFetchOrg]);
  1000→  1000→
  1001→  1001→  // --- Real CSV export ---
  1002→  1002→  const exportToCSV = React.useCallback(async (entryId: string) => {
  1003→  1003→    const entry = entries.find(e => e.id === entryId);
  1004→  1004→    if (!entry) return;
  1005→  1005→
  1006→  1006→    // Try the /api/export/:id route first (generates CSV server-side from the
  1007→  1007→    // DB-backed shipment data and returns it as text/csv). Uses the org-scoped
  1008→  1008→    // wrapper so the X-Org-Id header is sent.
  1009→  1009→    if (isSupabaseConfigured()) {
  1010→  1010→      try {
  1011→  1011→        const res = await apiFetchOrg<Response>(`/api/export/${entryId}`, { raw: true });
  1012→  1012→        if (res.ok) {
  1013→  1013→          const csv = await res.text();
  1014→  1014→          if (csv) {
  1015→  1015→            const blob = new Blob([csv], { type: 'text/csv' });
  1016→  1016→            const url = URL.createObjectURL(blob);
  1017→  1017→            const a = document.createElement('a');
  1018→  1018→            a.href = url;
  1019→  1019→            a.download = `ClearPort_Audit_${entryId}.csv`;
  1020→  1020→            document.body.appendChild(a);
  1021→  1021→            a.click();
  1022→  1022→            document.body.removeChild(a);
  1023→  1023→            URL.revokeObjectURL(url);
  1024→  1024→
  1025→  1025→            addAuditLog(`Audit logs exported to CSV for ${entryId}.`, 'success', entryId);
  1026→  1026→            return;
  1027→  1027→          }
  1028→  1028→        }
  1029→  1029→      } catch (err) {
  1030→  1030→        console.warn('[csv] /api/export failed, generating locally:', err);
  1031→  1031→      }
  1032→  1032→    }
  1033→  1033→
  1034→  1034→    // Fallback: local CSV generation from in-memory entry
  1035→  1035→    const rows: string[] = [];
  1036→  1036→    rows.push('Field Key,Field Label,Value,Source Document,Confidence,Flagged,Status');
  1037→  1037→    entry.fields.forEach(f => {
  1038→  1038→      const exc = entry.exceptions.find(e => e.id === f.exceptionId);
  1039→  1039→      const status = exc ? exc.status : 'SECURE';
  1040→  1040→      const escaped = [f.key, f.label, f.value, f.sourceDoc, f.confidence, f.isFlagged, status]
  1041→  1041→        .map(v => `"${String(v).replace(/"/g, '""')}"`)
  1042→  1042→        .join(',');
  1043→  1043→      rows.push(escaped);
  1044→  1044→    });
  1045→  1045→
  1046→  1046→    rows.push('');
  1047→  1047→    rows.push('Exception ID,Field Name,Reason,Confidence,Status,Resolved By');
  1048→  1048→    entry.exceptions.forEach(e => {
  1049→  1049→      const escaped = [e.id, e.fieldName, e.reason, e.confidence, e.status, e.resolvedBy || '']
  1050→  1050→        .map(v => `"${String(v).replace(/"/g, '""')}"`)
  1051→  1051→        .join(',');
  1052→  1052→      rows.push(escaped);
  1053→  1053→    });
  1054→  1054→
  1055→  1055→    const csv = rows.join('\n');
  1056→  1056→    const blob = new Blob([csv], { type: 'text/csv' });
  1057→  1057→    const url = URL.createObjectURL(blob);
  1058→  1058→    const a = document.createElement('a');
  1059→  1059→    a.href = url;
  1060→  1060→    a.download = `ClearPort_Audit_${entryId}.csv`;
  1061→  1061→    document.body.appendChild(a);
  1062→  1062→    a.click();
  1063→  1063→    document.body.removeChild(a);
  1064→  1064→    URL.revokeObjectURL(url);
  1065→  1065→
  1066→  1066→    addAuditLog(`Audit logs exported to CSV for ${entryId}.`, 'success', entryId);
  1067→  1067→  }, [entries, addAuditLog, apiFetchOrg]);
  1068→  1068→
  1069→  1069→  const value: ClearPortContextType = {
  1070→  1070→    entries,
  1071→  1071→    selectedEntryId,
  1072→  1072→    selectedEntry,

  1081→  1081→    edgeFunctionStatus,

  1088→  1088→    selectEntry,

  1090→  1090→    setActiveTab,
  1091→  1091→    updateException,

  1093→  1093→    acceptAllHighConfidence,

  1096→  1096→    exportToCSV,

  1102→  1102→  return (

  1104→  1104→      {children}

     1→'use client';
     2→
     3→import * as React from 'react';
     4→import { useClearPort } from '@/context/ClearPortContext';
     5→import {
     6→  AlertCircle, CheckCircle2, ChevronRight, Eye, RefreshCw, XCircle, FileText,
     7→  ZoomIn, ZoomOut, RotateCw, Sparkles, Undo2, Loader2,
     8→} from 'lucide-react';
     9→import { motion, AnimatePresence } from 'framer-motion';
    10→import { supabase, invokeEdgeFunction } from '@/lib/supabase';
    11→
    12→export default function ExceptionDesk() {
    13→  const {
    14→    entries,
    15→    selectedEntry,
    16→    selectedExceptionId,
    17→    selectedException,
    18→    selectException,
    19→    updateException,
    20→    undoLastAction,
    21→    acceptAllHighConfidence,
    22→    rules,
    23→  } = useClearPort();
    24→
    25→  const [severityFilter, setSeverityFilter] = React.useState<'All' | 'High' | 'Medium' | 'Low'>('All');
    26→  const [sortBy, setSortBy] = React.useState<'severity' | 'confidence' | 'fieldName'>('severity');
    27→  const [isEditing, setIsEditing] = React.useState(false);
    28→  const [editValue, setEditValue] = React.useState('');
    29→  const [activeDocTab, setActiveDocTab] = React.useState('');
    30→  const [zoomLevel, setZoomLevel] = React.useState(100);
    31→  const [rotation, setRotation] = React.useState(0);
    32→  const [showHistory, setShowHistory] = React.useState(false);
    33→  const [documentUrl, setDocumentUrl] = React.useState<string | null>(null);
    34→  const [isLoadingUrl, setIsLoadingUrl] = React.useState(false);
    35→  const inputRef = React.useRef<HTMLInputElement>(null);
    36→
    37→  // Fetch signed URL for the active document when tab changes
    38→  React.useEffect(() => {
    39→    if (!selectedEntry || !activeDocTab) {
    40→      setDocumentUrl(null);
    41→      return;
    42→    }
    43→
    44→    // Find the document matching the active tab
    45→    const doc = selectedEntry.documents.find(d => d.docType === activeDocTab);
    46→    if (!doc) {
    47→      setDocumentUrl(null);
    48→      return;
    49→    }
    50→
    51→    let cancelled = false;
    52→    setIsLoadingUrl(true);
    53→    setDocumentUrl(null);
    54→
    55→    (async () => {
    56→      try {
    57→        // Try the get-document-url edge function first
    58→        const response = await invokeEdgeFunction<any>('get-document-url', {
    59→          storagePath: doc.storagePath,
    60→        });
    61→        if (!cancelled && response?.success && response.signedUrl) {
    62→          setDocumentUrl(response.signedUrl);
    63→        }
    64→      } catch (err) {
    65→        // Fallback: try direct Supabase storage signed URL
    66→        if (supabase && !cancelled) {
    67→          try {
    68→            const { data } = await supabase.storage
    69→              .from('documents')
    70→              .createSignedUrl(doc.storagePath, 3600);
    71→            if (data?.signedUrl) {
    72→              setDocumentUrl(data.signedUrl);
    73→            }
    74→          } catch (e) {
    75→            console.warn('[doc-url] failed to get signed URL:', e);
    76→          }
    77→        }
    78→      } finally {
    79→        if (!cancelled) setIsLoadingUrl(false);
    80→      }
    81→    })();
    82→
    83→    return () => { cancelled = true; };
    84→  }, [selectedEntry?.id, activeDocTab]);
    85→
    86→  // Fix: use useEffect instead of setState-during-render
    87→  React.useEffect(() => {
    88→    if (selectedException) {
    89→      setEditValue(selectedException.correctedValue ?? selectedException.extractedValue);
    90→      setActiveDocTab(selectedException.docType);
    91→      setIsEditing(false);
    92→    }
    93→  }, [selectedException?.id]);
    94→
    95→  // Keyboard shortcuts (scoped — only when not typing in input)
    96→  React.useEffect(() => {
    97→    const handleKeyDown = (e: KeyboardEvent) => {
    98→      const activeEl = document.activeElement;
    99→      if (activeEl?.tagName === 'INPUT' || activeEl?.tagName === 'TEXTAREA') {
   100→        if (e.key === 'Escape') {
   101→          inputRef.current?.blur();
   102→          setIsEditing(false);
   103→        }
   104→        return;
   105→      }
   106→
   107→      if (!selectedException || !selectedEntry) return;
   108→
   109→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'z') {
   110→        e.preventDefault();
   111→        undoLastAction();
   112→        return;
   113→      }
   114→
   115→      if (e.key === ' ') {
   116→        e.preventDefault();
   117→        updateException(selectedEntry.id, selectedException.id, 'Accepted');
   118→      }
   119→
   120→      if (e.key.toLowerCase() === 'e') {
   121→        e.preventDefault();
   122→        setIsEditing(true);
   123→        setTimeout(() => inputRef.current?.focus(), 50);
   124→      }
   125→
   126→      if (e.key.toLowerCase() === 'r') {
   127→        e.preventDefault();
   128→        updateException(selectedEntry.id, selectedException.id, 'Rejected');
   129→      }
   130→    };
   131→
   132→    window.addEventListener('keydown', handleKeyDown);
   133→    return () => window.removeEventListener('keydown', handleKeyDown);
   134→  }, [selectedException, selectedEntry, updateException, undoLastAction]);
   135→
   136→  if (!selectedEntry) {
   137→    return (
   138→      <div className="h-full flex flex-col items-center justify-center text-gray-500 p-6">
   139→        <AlertCircle className="w-12 h-12 mb-4 text-gray-600 animate-pulse" />
   140→        <p className="text-sm font-medium">No shipment selected</p>
   141→        <p className="text-xs text-gray-600 mt-1">Please select an active batch from the Dashboard or sidebar.</p>
   142→      </div>
   143→    );
   144→  }
   145→
   146→  const exceptions = selectedEntry.exceptions;
   147→
   148→  // Filter & Sort
   149→  const filteredExceptions = exceptions.filter(ex => {
   150→    if (severityFilter === 'All') return true;
   151→    if (severityFilter === 'High') return ex.confidence < 60;
   152→    if (severityFilter === 'Medium') return ex.confidence >= 60 && ex.confidence < 85;
   153→    if (severityFilter === 'Low') return ex.confidence >= 85;
   154→    return true;
   155→  });
   156→
   157→  const sortedExceptions = [...filteredExceptions].sort((a, b) => {
   158→    if (sortBy === 'severity') return a.confidence - b.confidence;
   159→    if (sortBy === 'confidence') return b.confidence - a.confidence;
   160→    return a.fieldName.localeCompare(b.fieldName);
   161→  });
   162→
   163→  const totalExceptions = exceptions.length;
   164→  const resolvedExceptions = exceptions.filter(ex => ex.status !== 'Unresolved').length;
   165→  const highConfidenceCount = exceptions.filter(ex => ex.status === 'Unresolved' && ex.confidence >= rules.invoiceThreshold).length;
   166→
   167→  const getConfidenceColor = (conf: number) => {
   168→    if (conf < 60) return 'text-red-400 bg-red-950/40 border-red-900/50';
   169→    if (conf < 85) return 'text-amber-400 bg-amber-950/40 border-amber-900/50';
   170→    return 'text-green-400 bg-green-950/40 border-green-900/50';
   171→  };
   172→
   173→  const getConfidenceBadge = (conf: number) => {
   174→    if (conf < 60) return 'RED / CRITICAL';
   175→    if (conf < 85) return 'AMBER / WARNING';
   176→    return 'GREEN / COMPLIANT';
   177→  };
   178→
   179→  const handleSaveEdit = () => {
   180→    if (selectedException && selectedEntry) {
   181→      updateException(selectedEntry.id, selectedException.id, 'Corrected', editValue);
   182→      setIsEditing(false);
   183→    }
   184→  };
   185→
   186→  // Data-driven document tabs (derive from shipment fields, not hardcoded IDs)
   187→  const availableDocTypes = Array.from(new Set(selectedEntry.fields.map(f => f.sourceDoc).filter(Boolean)));
   188→
   189→  return (
   190→    <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 h-full overflow-hidden p-6 font-sans">
   191→      {/* LEFT PANEL: EXCEPTION LIST */}
   192→      <div className="lg:col-span-3 bg-[#0c0d12] border border-gray-900 rounded-xl flex flex-col overflow-hidden">
   193→        <div className="p-4 border-b border-gray-900 bg-[#0e1017]">
   194→          <div className="flex items-center justify-between mb-2">
   195→            <span className="font-mono text-xs text-gray-500 tracking-wider">EXCEPTIONS DESK</span>
   196→            <span className="font-mono text-xs text-emerald-400 bg-emerald-950/30 px-2 py-0.5 rounded border border-emerald-900/40">
   197→              {resolvedExceptions} / {totalExceptions} RESOLVED
   198→            </span>
   199→          </div>
   200→          <h2 className="text-sm font-medium text-gray-200 tracking-tight flex items-center gap-1.5">
   201→            <FileText className="w-4 h-4 text-gray-400" />
   202→            {selectedEntry.id}
   203→          </h2>
   204→          <p className="text-xs text-gray-500 mt-1 truncate">{selectedEntry.shipper}</p>
   205→        </div>
   206→
   207→        {/* Filters */}
   208→        <div className="p-3 border-b border-gray-900 bg-[#090a0f] space-y-2.5">
   209→          <div className="flex items-center gap-1">
   210→            {(['All', 'High', 'Medium', 'Low'] as const).map(filter => (
   211→              <button
   212→                key={filter}
   213→                onClick={() => setSeverityFilter(filter)}
   214→                className={`text-[11px] px-2.5 py-1 rounded-md font-medium transition-all ${
   215→                  severityFilter === filter
   216→                    ? 'bg-gray-800 text-white'
   217→                    : 'text-gray-500 hover:text-gray-300 hover:bg-gray-950'
   218→                }`}
   219→              >
   220→                {filter}
   221→              </button>
   222→            ))}
   223→          </div>
   224→
   225→          <div className="flex items-center justify-between gap-2 pt-1">
   226→            <span className="text-[10px] font-mono text-gray-500 uppercase tracking-widest">SORT BY</span>
   227→            <select
   228→              value={sortBy}
   229→              onChange={e => setSortBy(e.target.value as any)}
   230→              className="bg-black border border-gray-800 text-gray-300 rounded px-1.5 py-0.5 text-xs focus:outline-none focus:border-gray-700 font-medium"
   231→            >
   232→              <option value="severity">Severity (Low Conf)</option>
   233→              <option value="confidence">Confidence (High)</option>
   234→              <option value="fieldName">Field Name</option>
   235→            </select>
   236→          </div>
   237→
   238→          {highConfidenceCount > 0 && (
   239→            <button
   240→              onClick={() => acceptAllHighConfidence(selectedEntry.id)}
   241→              className="w-full flex items-center justify-center gap-1.5 text-xs bg-emerald-950/30 text-emerald-400 border border-emerald-900/50 hover:bg-emerald-900/20 py-2 rounded-md transition-all cursor-pointer font-medium mt-1"
   242→            >
   243→              <Sparkles className="w-3.5 h-3.5" />
   244→              Accept {highConfidenceCount} Fields ≥ {rules.invoiceThreshold}% Conf
   245→            </button>
   246→          )}
   247→        </div>
   248→
   249→        {/* Exceptions list */}
   250→        <div className="flex-1 overflow-y-auto divide-y divide-gray-950 p-2 space-y-1.5">
   251→          <AnimatePresence mode="popLayout">
   252→            {sortedExceptions.map(ex => {
   253→              const isSelected = ex.id === selectedExceptionId;
   254→              const severityClass = getConfidenceColor(ex.confidence);
   255→              const isDone = ex.status !== 'Unresolved';
   256→
   257→              return (
   258→                <motion.div
   259→                  key={ex.id}
   260→                  layout
   261→                  initial={{ opacity: 0, y: 10 }}
   262→                  animate={{ opacity: 1, y: 0 }}
   263→                  exit={{ opacity: 0, scale: 0.95 }}
   264→                  transition={{ duration: 0.15 }}
   265→                  onClick={() => selectException(ex.id)}
   266→                  className={`p-3 rounded-lg cursor-pointer transition-all border ${
   267→                    isSelected
   268→                      ? 'bg-gray-950 border-gray-700 shadow-lg shadow-black/40'
   269→                      : 'bg-transparent border-transparent hover:bg-gray-950/40 hover:border-gray-900'
   270→                  }`}
   271→                >
   272→                  <div className="flex items-start justify-between gap-1.5 mb-1.5">
   273→                    <span className="text-xs font-semibold text-gray-200 tracking-tight leading-tight">
   274→                      {ex.fieldName}
   275→                    </span>
   276→                    {isDone ? (
   277→                      <span className={`text-[9px] font-mono font-bold px-1.5 py-0.5 rounded flex items-center gap-1 shrink-0 ${
   278→                        ex.status === 'Accepted' ? 'bg-emerald-950 text-emerald-400 border border-emerald-900/30' :
   279→                        ex.status === 'Rejected' ? 'bg-red-950 text-red-400 border border-red-900/30' :
   280→                        'bg-blue-950 text-blue-400 border border-blue-900/30'
   281→                      }`}>
   282→                        {ex.status.toUpperCase()}
   283→                      </span>
   284→                    ) : (
   285→                      <span className={`text-[10px] font-mono font-bold px-1.5 py-0.5 rounded border shrink-0 ${severityClass}`}>
   286→                        {ex.confidence}%
   287→                      </span>
   288→                    )}
   289→                  </div>
   290→
   291→                  <p className="text-[11px] text-gray-400 leading-normal line-clamp-2">{ex.reason}</p>
   292→
   293→                  <div className="flex items-center justify-between mt-2 pt-2 border-t border-gray-900/30 text-[10px] text-gray-500 font-mono">
   294→                    <span>{ex.docType}</span>
   295→                    <span className="text-gray-600 uppercase">{ex.exceptionType.replace(/_/g, ' ')}</span>
   296→                  </div>
   297→                </motion.div>
   298→              );
   299→            })}
   300→          </AnimatePresence>
   301→
   302→          {sortedExceptions.length === 0 && (
   303→            <div className="p-8 text-center text-gray-500 text-xs">
   304→              No exceptions match the selected filter.
   305→            </div>
   306→          )}
   307→        </div>
   308→      </div>
   309→
   310→      {/* CENTER PANEL: DOCUMENT VIEWER */}
   311→      <div className="lg:col-span-5 bg-[#0c0d12] border border-gray-900 rounded-xl flex flex-col overflow-hidden relative">
   312→        {/* Document Tabs (data-driven) */}
   313→        <div className="px-4 py-2 bg-[#0e1017] border-b border-gray-900 flex items-center justify-between shrink-0">
   314→          <div className="flex items-center gap-1.5 overflow-x-auto">
   315→            {availableDocTypes.map(doc => (
   316→              <button
   317→                key={doc}
   318→                onClick={() => setActiveDocTab(doc)}
   319→                className={`text-xs px-3 py-1.5 rounded-md font-medium transition-all whitespace-nowrap ${
   320→                  activeDocTab === doc
   321→                    ? 'bg-gray-800 text-gray-100'
   322→                    : 'text-gray-500 hover:text-gray-300'
   323→                }`}
   324→              >
   325→                {doc === 'Commercial Invoice' ? 'Invoice' : doc === 'Bill of Lading' ? 'BOL' : doc}
   326→              </button>
   327→            ))}
   328→            {availableDocTypes.length === 0 && (
   329→              <span className="text-xs text-gray-600">No documents available</span>
   330→            )}
   331→          </div>
   332→
   333→          <div className="flex items-center gap-1 shrink-0">
   334→            <button
   335→              onClick={() => setZoomLevel(prev => Math.max(50, prev - 10))}
   336→              title="Zoom Out"
   337→              className="p-1 rounded text-gray-500 hover:text-gray-300 hover:bg-gray-950 transition-all cursor-pointer"
   338→            >
   339→              <ZoomOut className="w-4 h-4" />
   340→            </button>
   341→            <span className="text-xs font-mono text-gray-500 w-10 text-center">{zoomLevel}%</span>
   342→            <button
   343→              onClick={() => setZoomLevel(prev => Math.min(200, prev + 10))}
   344→              title="Zoom In"
   345→              className="p-1 rounded text-gray-500 hover:text-gray-300 hover:bg-gray-950 transition-all cursor-pointer"
   346→            >
   347→              <ZoomIn className="w-4 h-4" />
   348→            </button>
   349→            <button
   350→              onClick={() => setRotation(prev => (prev + 90) % 360)}
   351→              title="Rotate 90°"
   352→              className="p-1 rounded text-gray-500 hover:text-gray-300 hover:bg-gray-950 transition-all cursor-pointer"
   353→            >
   354→              <RotateCw className="w-4 h-4" />
   355→            </button>
   356→          </div>
   357→        </div>
   358→
   359→        {/* Document Board — shows real uploaded file or extracted data view */}
   360→        <div className="flex-1 overflow-auto bg-[#040406] p-6 flex items-start justify-center relative">
   361→          {documentUrl ? (
   362→            /* Real file viewer — show the actual uploaded document */
   363→            <div
   364→              style={{
   365→                transform: `scale(${zoomLevel / 100}) rotate(${rotation}deg)`,
   366→                transformOrigin: 'top center',
   367→                transition: 'transform 0.15s ease-out',
   368→              }}
   369→              className="rounded border border-gray-700 shadow-2xl overflow-hidden bg-white relative"
   370→            >
   371→              {documentUrl.endsWith('.txt') || documentUrl.includes('text/plain') ? (
   372→                /* Text file — show as preformatted text */
   373→                <iframe src={documentUrl} className="w-[500px] h-[600px] bg-white" title="Document" />
   374→              ) : documentUrl.includes('.pdf') || documentUrl.includes('application/pdf') ? (
   375→                /* PDF — show in iframe */
   376→                <iframe src={documentUrl} className="w-[500px] h-[600px] bg-white" title="Document PDF" />
   377→              ) : (
   378→                /* Image — show directly */
   379→                <img src={documentUrl} alt="Document" className="max-w-[500px] max-h-[600px] object-contain" />
   380→              )}
   381→
   382→              {/* Highlighted bounding box overlay */}
   383→              {selectedException && activeDocTab === selectedException.docType && (
   384→                <motion.div
   385→                  initial={{ opacity: 0, scale: 0.95 }}
   386→                  animate={{ opacity: 1, scale: 1 }}
   387→                  className="absolute border-[2px] border-amber-500 bg-amber-500/10 rounded pointer-events-none shadow-[0_0_12px_rgba(245,158,11,0.4)]"
   388→                  style={{
   389→                    left: `${selectedException.boundingBox.x}%`,
   390→                    top: `${selectedException.boundingBox.y}%`,
   391→                    width: `${selectedException.boundingBox.w}%`,
   392→                    height: `${selectedException.boundingBox.h}%`,
   393→                  }}
   394→                >
   395→                  <div className="absolute -top-3.5 left-0 text-[7px] font-mono text-white bg-amber-600 px-1 py-0.5 rounded shadow whitespace-nowrap uppercase tracking-wider font-bold">
   396→                    FLAGGED: {selectedException.fieldName.toUpperCase()}
   397→                  </div>
   398→                </motion.div>
   399→              )}
   400→            </div>
   401→          ) : isLoadingUrl ? (
   402→            <div className="flex items-center justify-center h-full">
   403→              <Loader2 className="w-8 h-8 text-amber-500 animate-spin" />
   404→            </div>
   405→          ) : (
   406→            /* Fallback: show extracted data as a structured view (not a fake document) */
   407→            <div
   408→              style={{
   409→                transform: `scale(${zoomLevel / 100}) rotate(${rotation}deg)`,
   410→                transformOrigin: 'top center',
   411→                transition: 'transform 0.15s ease-out',
   412→              }}
   413→              className="w-[500px] bg-[#0c0d12] border border-gray-800 rounded-xl shadow-2xl p-6 relative"
   414→            >
   415→              <div className="flex items-center justify-between mb-4 pb-3 border-b border-gray-800">
   416→                <div>
   417→                  <h3 className="text-sm font-bold text-gray-100 uppercase tracking-wider">
   418→                    {activeDocTab || 'Extracted Data'}
   419→                  </h3>
   420→                  <p className="text-[10px] text-gray-500 font-mono mt-0.5">
   421→                    {selectedEntry.id} • {selectedEntry.documents.length} file(s) uploaded
   422→                  </p>
   423→                </div>
   424→                <span className="text-[9px] font-mono bg-amber-950/40 text-amber-400 border border-amber-900/40 px-2 py-0.5 rounded uppercase">
   425→                  Structured Extract
   426→                </span>
   427→              </div>
   428→
   429→              <div className="space-y-2">
   430→                {selectedEntry.fields.map((f, idx) => (
   431→                  <div key={f.id || idx} className={`flex justify-between items-start p-2 rounded border ${
   432→                    f.isFlagged ? 'bg-red-950/20 border-red-900/40' : 'bg-black/30 border-gray-900'
   433→                  }`}>
   434→                    <div className="flex-1">
   435→                      <span className="text-[10px] font-mono text-gray-500 uppercase tracking-wider block">{f.label}</span>
   436→                      <span className={`text-xs font-mono ${f.isFlagged ? 'text-amber-400' : 'text-gray-200'}`}>
   437→                        {f.value}
   438→                      </span>
   439→                      {f.crossDocValue && (
   440→                        <span className="text-[9px] text-red-400 block mt-0.5">↔ {f.crossDocValue}</span>
   441→                      )}
   442→                    </div>
   443→                    <span className={`text-[9px] font-mono px-1.5 py-0.5 rounded ${
   444→                      f.confidence < 60 ? 'bg-red-950 text-red-400' :
   445→                      f.confidence < 85 ? 'bg-amber-950 text-amber-400' :
   446→                      'bg-emerald-950 text-emerald-400'
   447→                    }`}>
   448→                      {f.confidence}%
   449→                    </span>
   450→                  </div>
   451→                ))}
   452→              </div>
   453→
   454→              {selectedEntry.fields.length === 0 && (
   455→                <div className="text-center py-8 text-gray-600 text-xs">
   456→                  No fields extracted for this shipment.
   457→                </div>
   458→              )}
   459→
   460→              {selectedException && activeDocTab === selectedException.docType && (
   461→                <motion.div
   462→                  initial={{ opacity: 0 }}
   463→                  animate={{ opacity: 1 }}
   464→                  className="mt-4 p-3 bg-amber-950/20 border border-amber-900/40 rounded-lg"
   465→                >
   466→                  <div className="text-[10px] font-mono text-amber-400 uppercase tracking-wider mb-1">
   467→                    ⚠ FLAGGED: {selectedException.fieldName}
   468→                  </div>
   469→                  <p className="text-[11px] text-gray-300">{selectedException.reason}</p>
   470→                </motion.div>
   471→              )}
   472→            </div>
   473→          )}
   474→        </div>
   475→      </div>
   476→
   477→      {/* RIGHT PANEL: RESOLUTION DESK */}
   478→      <div className="lg:col-span-4 bg-[#0c0d12] border border-gray-900 rounded-xl flex flex-col overflow-hidden">
   479→        {selectedException ? (
   480→          <div className="flex-1 flex flex-col justify-between overflow-hidden">
   481→            <div className="p-4 space-y-4 overflow-y-auto flex-1">
   482→              {/* Header */}
   483→              <div className="border-b border-gray-900 pb-3">
   484→                <div className="flex items-center justify-between mb-1.5">
   485→                  <span className="text-[10px] font-mono text-gray-500 tracking-wider">EXCEPTION FIELD RESOLUTION</span>
   486→                  <span className={`text-[9px] font-mono font-bold px-2 py-0.5 rounded border ${
   487→                    selectedException.status === 'Unresolved'
   488→                      ? getConfidenceColor(selectedException.confidence)
   489→                      : 'bg-gray-950 text-gray-400 border-gray-900'
   490→                  }`}>
   491→                    {selectedException.status === 'Unresolved'
   492→                      ? getConfidenceBadge(selectedException.confidence)
   493→                      : selectedException.status.toUpperCase()}
   494→                  </span>
   495→                </div>
   496→                <h3 className="text-base font-semibold text-gray-100 tracking-tight leading-tight">
   497→                  {selectedException.fieldName}
   498→                </h3>
   499→                <span className="text-[10px] font-mono text-gray-600 uppercase mt-1 block">
   500→                  Type: {selectedException.exceptionType.replace(/_/g, ' ')}
   501→                </span>
   502→              </div>
   503→
   504→              {/* Discrepancy reason */}
   505→              <div className="bg-[#120f12] border border-red-950/40 p-3.5 rounded-lg">
   506→                <h4 className="text-[11px] font-mono text-red-400 tracking-wider uppercase mb-1 flex items-center gap-1.5">
   507→                  <AlertCircle className="w-3.5 h-3.5" />
   508→                  DISCREPANCY WHY:
   509→                </h4>
   510→                <p className="text-xs text-gray-300 leading-normal">{selectedException.reason}</p>
   511→              </div>
   512→
   513→              {/* Data comparison */}
   514→              <div className="grid grid-cols-2 gap-3">
   515→                <div className="bg-black border border-gray-900 p-3 rounded-lg">
   516→                  <span className="text-[10px] text-gray-500 block uppercase tracking-wider font-mono">EXTRACTED (OCR)</span>
   517→                  <span className="text-sm font-mono font-bold text-gray-200 block mt-1">
   518→                    {selectedException.extractedValue}
   519→                  </span>
   520→                  <span className="text-[9px] font-mono text-gray-600 block mt-0.5">
   521→                    Confidence: {selectedException.confidence}%
   522→                  </span>
   523→                </div>
   524→                <div className="bg-black border border-gray-900 p-3 rounded-lg">
   525→                  <span className="text-[10px] text-gray-500 block uppercase tracking-wider font-mono">CROSS-DOC SOURCE</span>
   526→                  <span className="text-sm font-mono font-bold text-gray-400 block mt-1">
   527→                    {selectedException.crossDocValue || '—'}
   528→                  </span>
   529→                  <span className="text-[9px] font-mono text-gray-600 block mt-0.5">
   530→                    Source: {selectedException.docType}
   531→                  </span>
   532→                </div>
   533→              </div>
   534→
   535→              {/* Inline editor */}
   536→              <div className="space-y-1.5">
   537→                <label className="text-[11px] text-gray-400 font-mono tracking-wider block uppercase">
   538→                  MANUAL / OVERRIDE VALUE:
   539→                </label>
   540→                {isEditing ? (
   541→                  <div className="flex gap-2">
   542→                    <input
   543→                      ref={inputRef}
   544→                      type="text"
   545→                      value={editValue}
   546→                      onChange={e => setEditValue(e.target.value)}
   547→                      onKeyDown={e => {
   548→                        if (e.key === 'Enter') handleSaveEdit();
   549→                        if (e.key === 'Escape') setIsEditing(false);
   550→                      }}
   551→                      className="flex-1 bg-black text-gray-200 border border-gray-700 px-3 py-2 text-sm rounded-lg focus:outline-none focus:border-amber-500 font-mono tracking-tight"
   552→                    />
   553→                    <button
   554→                      onClick={handleSaveEdit}
   555→                      className="bg-amber-600 hover:bg-amber-500 text-black px-3 py-1.5 rounded-lg font-medium text-xs transition-all cursor-pointer"
   556→                    >
   557→                      Apply
   558→                    </button>
   559→                    <button
   560→                      onClick={() => setIsEditing(false)}
   561→                      className="text-gray-400 hover:text-white border border-gray-800 px-3 py-1.5 rounded-lg text-xs transition-all cursor-pointer"
   562→                    >
   563→                      Cancel
   564→                    </button>
   565→                  </div>
   566→                ) : (
   567→                  <div className="flex items-center justify-between bg-black border border-gray-900 rounded-lg p-2.5">
   568→                    <span className="font-mono text-sm font-bold text-emerald-400">{editValue}</span>
   569→                    <button
   570→                      onClick={() => {
   571→                        setIsEditing(true);
   572→                        setTimeout(() => inputRef.current?.focus(), 50);
   573→                      }}
   574→                      className="text-xs text-gray-400 hover:text-white bg-gray-900 hover:bg-gray-800 border border-gray-800 px-2 py-1 rounded transition-all cursor-pointer"
   575→                    >
   576→                      Edit Field
   577→                    </button>
   578→                  </div>
   579→                )}
   580→              </div>
   581→
   582→              {/* Action buttons */}
   583→              <div className="pt-3 grid grid-cols-3 gap-2">
   584→                <button
   585→                  onClick={() => updateException(selectedEntry.id, selectedException.id, 'Accepted')}
   586→                  className={`py-2 px-1 text-xs rounded-lg font-medium border cursor-pointer transition-all flex flex-col items-center justify-center gap-1 ${
   587→                    selectedException.status === 'Accepted'
   588→                      ? 'bg-emerald-950 text-emerald-400 border-emerald-800'
   589→                      : 'bg-black border-gray-900 text-emerald-500 hover:bg-emerald-950/20'
   590→                  }`}
   591→                >
   592→                  <CheckCircle2 className="w-4 h-4" />
   593→                  <span>Accept [Space]</span>
   594→                </button>
   595→                <button
   596→                  onClick={() => {
   597→                    setIsEditing(true);
   598→                    setTimeout(() => inputRef.current?.focus(), 50);
   599→                  }}
   600→                  className={`py-2 px-1 text-xs rounded-lg font-medium border cursor-pointer transition-all flex flex-col items-center justify-center gap-1 ${
   601→                    selectedException.status === 'Corrected'
   602→                      ? 'bg-blue-950 text-blue-400 border-blue-800'
   603→                      : 'bg-black border-gray-900 text-blue-500 hover:bg-blue-950/20'
   604→                  }`}
   605→                >
   606→                  <RefreshCw className="w-4 h-4" />
   607→                  <span>Modify [E]</span>
   608→                </button>
   609→                <button
   610→                  onClick={() => updateException(selectedEntry.id, selectedException.id, 'Rejected')}
   611→                  className={`py-2 px-1 text-xs rounded-lg font-medium border cursor-pointer transition-all flex flex-col items-center justify-center gap-1 ${
   612→                    selectedException.status === 'Rejected'
   613→                      ? 'bg-red-950 text-red-400 border-red-800'
   614→                      : 'bg-black border-gray-900 text-red-500 hover:bg-red-950/20'
   615→                  }`}
   616→                >
   617→                  <XCircle className="w-4 h-4" />
   618→                  <span>Reject [R]</span>
   619→                </button>
   620→              </div>
   621→
   622→              {/* Audit trail */}
   623→              <div className="pt-4 border-t border-gray-900">
   624→                <button
   625→                  onClick={() => setShowHistory(!showHistory)}
   626→                  className="flex items-center justify-between w-full text-xs text-gray-500 font-mono tracking-wider hover:text-gray-300 py-1"
   627→                >
   628→                  <span>AUDIT TRAIL ({selectedException.history.length})</span>
   629→                  <span>{showHistory ? '[-]' : '[+]'}</span>
   630→                </button>
   631→                {showHistory && (
   632→                  <div className="mt-2 text-[11px] text-gray-400 bg-[#090a0f] p-3 rounded-lg border border-gray-900 space-y-2 max-h-32 overflow-y-auto">
   633→                    {selectedException.history.length === 0 ? (
   634→                      <p className="text-gray-600 italic">No previous reviews logged for this field.</p>
   635→                    ) : (
   636→                      selectedException.history.map((h, idx) => (
   637→                        <div key={idx} className="border-b border-gray-900/40 pb-1.5 last:border-b-0">
   638→                          <div className="flex justify-between items-center text-gray-500 font-mono text-[9px] mb-0.5">
   639→                            <span>{h.user}</span>
   640→                            <span suppressHydrationWarning>{new Date(h.timestamp).toLocaleTimeString()}</span>
   641→                          </div>
   642→                          <div className="flex items-center gap-1">
   643→                            <span className="font-semibold text-gray-300">{h.action}:</span>
   644→                            <span className="line-through text-gray-600 font-mono">{h.oldValue}</span>
   645→                            <ChevronRight className="w-3 h-3 text-gray-700" />
   646→                            <span className="text-emerald-400 font-mono font-semibold">{h.newValue}</span>
   647→                          </div>
   648→                        </div>
   649→                      ))
   650→                    )}
   651→                  </div>
   652→                )}
   653→              </div>
   654→            </div>
   655→
   656→            {/* Keyboard shortcuts footer */}
   657→            <div className="p-3 bg-black/40 border-t border-gray-900 flex justify-between items-center text-[10px] text-gray-500 font-mono select-none">
   658→              <div className="flex items-center gap-1">
   659→                <kbd className="bg-gray-950 px-1.5 py-0.5 rounded border border-gray-800 text-gray-300 font-bold uppercase">Space</kbd> Accept
   660→                <kbd className="bg-gray-950 px-1.5 py-0.5 rounded border border-gray-800 text-gray-300 font-bold uppercase ml-2">E</kbd> Edit
   661→                <kbd className="bg-gray-950 px-1.5 py-0.5 rounded border border-gray-800 text-gray-300 font-bold uppercase ml-2">R</kbd> Reject
   662→              </div>
   663→              <button
   664→                onClick={undoLastAction}
   665→                className="flex items-center gap-1 text-[11px] text-gray-400 hover:text-white hover:bg-gray-900 border border-gray-800 px-2 py-1 rounded transition-all cursor-pointer font-bold uppercase"
   666→              >
   667→                <Undo2 className="w-3.5 h-3.5 text-gray-500" />
   668→                Undo [Ctrl+Z]
   669→              </button>
   670→            </div>
   671→          </div>
   672→        ) : (
   673→          <div className="flex-1 flex flex-col items-center justify-center text-center p-6 text-gray-500">
   674→            <CheckCircle2 className="w-12 h-12 text-gray-600 mb-2" />
   675→            <p className="text-sm font-semibold text-gray-300">All exceptions cleared</p>
   676→            <p className="text-xs text-gray-600 max-w-xs mt-1">
   677→              Select another shipment or upload invoices to run compliance tests.
   678→            </p>
   679→          </div>
   680→        )}
   681→      </div>
   682→    </div>
   683→  );

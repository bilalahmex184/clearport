     1→'use client';
     2→
     3→import * as React from 'react';
     4→import {
     5→  supabase,
     6→  isSupabaseConfigured,
     7→  ensureAuthenticated,
     8→  getCurrentUserEmail,
     9→  invokeEdgeFunction,
    10→  fetchShipmentsDirect,
    11→  fetchRulesDirect,
    12→  fetchLogsDirect,
    13→  seedEntries,
    14→  seedRules,
    15→  seedLogs,
    16→  calculateConfidence,
    17→} from '@/lib/supabase';
    18→import type {
    19→  ShipmentEntry,
    20→  Exception,
    21→  ExtractedField,
    22→  OperationalRules,
    23→  AuditLog,
    24→  ShipmentStatus,
    25→  ExceptionStatus,
    26→  ReviewerAction,
    27→  ExceptionHistoryEntry,
    28→} from '@/lib/clearport-types';
    29→
    30→// ============================================================================
    31→// Context Type
    32→// ============================================================================
    33→
    34→export type SupabaseStatus = 'connected' | 'unconfigured' | 'error_tables' | 'loading';
    35→export type EdgeFunctionStatus = 'live' | 'fallback' | 'unknown';
    36→
    37→interface ClearPortContextType {
    38→  entries: ShipmentEntry[];
    39→  selectedEntryId: string;
    40→  selectedEntry: ShipmentEntry | undefined;
    41→  selectedExceptionId: string;
    42→  selectedException: Exception | undefined;
    43→  activeTab: string;
    44→  rules: OperationalRules;
    45→  undoStack: { entryId: string; exceptionId: string; previousState: Exception }[];
    46→  auditLogs: AuditLog[];
    47→  theme: 'dark' | 'light';
    48→  supabaseStatus: SupabaseStatus;
    49→  edgeFunctionStatus: EdgeFunctionStatus;
    50→  currentUser: string;
    51→  currentTime: string;
    52→
    53→  // Actions
    54→  selectEntry: (id: string) => void;
    55→  selectException: (id: string) => void;
    56→  setActiveTab: (tab: string) => void;
    57→  updateException: (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => void;
    58→  undoLastAction: () => void;
    59→  acceptAllHighConfidence: (entryId: string) => void;
    60→  uploadDocuments: (files: File[]) => Promise<{ shipmentId: string; success: boolean; error?: string }>;
    61→  updateRules: (newRules: Partial<OperationalRules>) => void;
    62→  exportToCSV: (entryId: string) => Promise<void>;
    63→  toggleTheme: () => void;
    64→  refreshData: () => Promise<void>;
    65→}
    66→
    67→const ClearPortContext = React.createContext<ClearPortContextType | undefined>(undefined);
    68→
    69→// ============================================================================
    70→// Provider
    71→// ============================================================================
    72→
    73→export const ClearPortProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
    74→  const [entries, setEntries] = React.useState<ShipmentEntry[]>(seedEntries);
    75→  const [selectedEntryId, setSelectedEntryId] = React.useState<string>('SHIP-2026-8802');
    76→  const [selectedExceptionId, setSelectedExceptionId] = React.useState<string>('8802-hts');
    77→  const [activeTab, setActiveTab] = React.useState<string>('exception-desk');
    78→  const [theme, setTheme] = React.useState<'dark' | 'light'>('dark');
    79→  const [rules, setRules] = React.useState<OperationalRules>(seedRules);
    80→  const [undoStack, setUndoStack] = React.useState<{ entryId: string; exceptionId: string; previousState: Exception }[]>([]);
    81→  const [auditLogs, setAuditLogs] = React.useState<AuditLog[]>(seedLogs);
    82→  const [supabaseStatus, setSupabaseStatus] = React.useState<SupabaseStatus>(
    83→    isSupabaseConfigured() ? 'loading' : 'unconfigured'
    84→  );
    85→  const [edgeFunctionStatus, setEdgeFunctionStatus] = React.useState<EdgeFunctionStatus>('unknown');
    86→  const [currentUser, setCurrentUser] = React.useState<string>('Broker');
    87→  const [currentTime, setCurrentTime] = React.useState<string>('');
    88→
    89→  // --- Real-time clock ---
    90→  React.useEffect(() => {
    91→    const update = () => {
    92→      const now = new Date();
    93→      const utc = now.toISOString().slice(0, 16).replace('T', ' ') + ' UTC';
    94→      setCurrentTime(utc);
    95→    };
    96→    update();
    97→    const interval = setInterval(update, 1000);
    98→    return () => clearInterval(interval);
    99→  }, []);
   100→
   101→  // --- Load user email ---
   102→  React.useEffect(() => {
   103→    getCurrentUserEmail().then(email => setCurrentUser(email));
   104→  }, []);
   105→
   106→  // --- Persist theme ---
   107→  React.useEffect(() => {
   108→    const saved = typeof window !== 'undefined' ? localStorage.getItem('clearport-theme') : null;
   109→    if (saved === 'light' || saved === 'dark') setTheme(saved);
   110→  }, []);
   111→
   112→  const toggleTheme = React.useCallback(() => {
   113→    setTheme(prev => {
   114→      const next = prev === 'dark' ? 'light' : 'dark';
   115→      if (typeof window !== 'undefined') localStorage.setItem('clearport-theme', next);
   116→      return next;
   117→    });
   118→  }, []);
   119→
   120→  // --- Load data ---
   121→  const selectedEntryIdRef = React.useRef(selectedEntryId);
   122→  selectedEntryIdRef.current = selectedEntryId;
   123→
   124→  const loadData = React.useCallback(async () => {
   125→    if (!isSupabaseConfigured()) {
   126→      setSupabaseStatus('unconfigured');
   127→      setEdgeFunctionStatus('fallback');
   128→      return;
   129→    }
   130→
   131→    setSupabaseStatus('loading');
   132→
   133→    try {
   134→      await ensureAuthenticated();
   135→      const email = await getCurrentUserEmail();
   136→      setCurrentUser(email);
   137→
   138→      // Try edge function first
   139→      try {
   140→        const response = await invokeEdgeFunction<any>('get-shipments');
   141→        if (response?.success && Array.isArray(response.shipments)) {
   142→          if (response.shipments.length > 0) {
   143→            setEntries(response.shipments);
   144→            // Select first shipment if current selection doesn't exist
   145→            if (!response.shipments.find((s: any) => s.id === selectedEntryIdRef.current)) {
   146→              const first = response.shipments[0];
   147→              setSelectedEntryId(first.id);
   148→              if (first.exceptions?.length > 0) {
   149→                const unresolved = first.exceptions.find((e: any) => e.status === 'Unresolved');
   150→                setSelectedExceptionId(unresolved ? unresolved.id : first.exceptions[0].id);
   151→              }
   152→            }
   153→          }
   154→          setEdgeFunctionStatus('live');
   155→          setSupabaseStatus('connected');
   156→
   157→          const [dbRules, dbLogs] = await Promise.all([
   158→            fetchRulesDirect(),
   159→            fetchLogsDirect(),
   160→          ]);
   161→          if (dbRules) setRules(dbRules);
   162→          if (dbLogs && dbLogs.length > 0) setAuditLogs(dbLogs);
   163→          return;
   164→        }
   165→      } catch (edgeErr) {
   166→        console.warn('[ctx] Edge function unavailable, trying direct query:', edgeErr);
   167→      }
   168→
   169→      // Fallback: direct query
   170→      const dbShipments = await fetchShipmentsDirect();
   171→      if (dbShipments === null) {
   172→        setSupabaseStatus('error_tables');
   173→        setEdgeFunctionStatus('fallback');
   174→        return;
   175→      }
   176→
   177→      // Only use DB shipments if they have fields data (new schema).
   178→      // Old schema shipments have empty fields/exceptions — fall back to seed data.
   179→      if (dbShipments.length > 0 && dbShipments.some(s => s.fields.length > 0 || s.exceptions.length > 0)) {
   180→        setEntries(dbShipments);
   181→        // Select first shipment if current selection doesn't exist
   182→        if (!dbShipments.find(s => s.id === selectedEntryIdRef.current)) {
   183→          const first = dbShipments[0];
   184→          setSelectedEntryId(first.id);
   185→          if (first.exceptions.length > 0) {
   186→            const unresolved = first.exceptions.find(e => e.status === 'Unresolved');
   187→            setSelectedExceptionId(unresolved ? unresolved.id : first.exceptions[0].id);
   188→          }
   189→        }
   190→      }
   191→      // else: keep seed entries (demo mode)
   192→
   193→      const [dbRules, dbLogs] = await Promise.all([
   194→        fetchRulesDirect(),
   195→        fetchLogsDirect(),
   196→      ]);
   197→      if (dbRules) setRules(dbRules);
   198→      if (dbLogs && dbLogs.length > 0) setAuditLogs(dbLogs);
   199→
   200→      setSupabaseStatus('connected');
   201→      setEdgeFunctionStatus('fallback');
   202→    } catch (err) {
   203→      console.error('[ctx] loadData error:', err);
   204→      setSupabaseStatus('error_tables');
   205→      setEdgeFunctionStatus('fallback');
   206→    }
   207→  }, []);
   208→
   209→  React.useEffect(() => {
   210→    loadData();
   211→  }, [loadData]);
   212→
   213→  const refreshData = React.useCallback(async () => {
   214→    await loadData();
   215→  }, [loadData]);
   216→
   217→  // --- Computed ---
   218→  const selectedEntry = React.useMemo(
   219→    () => entries.find(e => e.id === selectedEntryId),
   220→    [entries, selectedEntryId]
   221→  );
   222→
   223→  const selectedException = React.useMemo(() => {
   224→    if (!selectedEntry) return undefined;
   225→    return selectedEntry.exceptions.find(ex => ex.id === selectedExceptionId);
   226→  }, [selectedEntry, selectedExceptionId]);
   227→
   228→  // --- Selection (fixes stale closure — uses functional update) ---
   229→  const selectEntry = React.useCallback((id: string) => {
   230→    setSelectedEntryId(id);
   231→    setEntries(prev => {
   232→      const entry = prev.find(e => e.id === id);
   233→      if (entry && entry.exceptions.length > 0) {
   234→        const unresolved = entry.exceptions.find(e => e.status === 'Unresolved');
   235→        setSelectedExceptionId(unresolved ? unresolved.id : entry.exceptions[0].id);
   236→      } else {
   237→        setSelectedExceptionId('');
   238→      }
   239→      return prev;
   240→    });
   241→  }, []);
   242→
   243→  const selectException = React.useCallback((id: string) => {
   244→    setSelectedExceptionId(id);
   245→  }, []);
   246→
   247→  // --- Audit log helper ---
   248→  const addAuditLog = React.useCallback((text: string, type: AuditLog['type'] = 'info', shipmentId?: string) => {
   249→    const newLog: AuditLog = {
   250→      id: typeof crypto !== 'undefined' ? crypto.randomUUID() : `log-${Date.now()}`,
   251→      text,
   252→      timestamp: new Date().toISOString(),
   253→      type,
   254→      shipmentId,
   255→    };
   256→    setAuditLogs(prev => [newLog, ...prev]);
   257→
   258→    if (isSupabaseConfigured() && supabase) {
   259→      supabase.from('audit_logs').insert({
   260→        id: newLog.id,
   261→        text: newLog.text,
   262→        timestamp: newLog.timestamp,
   263→        type: newLog.type,
   264→        shipment_id: shipmentId || null,
   265→      }).then(({ error }) => {
   266→        if (error) console.warn('[ctx] audit log insert failed:', error.message);
   267→      });
   268→    }
   269→  }, []);
   270→
   271→  // --- Update exception (fixes setState-in-setState) ---
   272→  const updateException = React.useCallback(
   273→    (entryId: string, exceptionId: string, status: ReviewerAction, newValue?: string) => {
   274→      let previousException: Exception | undefined;
   275→      let updatedEntry: ShipmentEntry | undefined;
   276→
   277→      setEntries(prevEntries => {
   278→        return prevEntries.map(entry => {
   279→          if (entry.id !== entryId) return entry;
   280→
   281→          const prev = entry.exceptions.find(ex => ex.id === exceptionId);
   282→          if (!prev) return entry;
   283→          previousException = { ...prev };
   284→
   285→          const updatedExceptions = entry.exceptions.map(ex => {
   286→            if (ex.id !== exceptionId) return ex;
   287→
   288→            const oldVal = ex.correctedValue || ex.extractedValue;
   289→            const updatedVal = newValue !== undefined ? newValue : ex.extractedValue;
   290→
   291→            const historyItem: ExceptionHistoryEntry = {
   292→              user: currentUser,
   293→              oldValue: oldVal,
   294→              newValue: updatedVal,
   295→              timestamp: new Date().toISOString(),
   296→              action: status,
   297→            };
   298→
   299→            return {
   300→              ...ex,
   301→              status: status as ExceptionStatus,
   302→              correctedValue: status === 'Corrected' ? newValue : undefined,
   303→              history: [historyItem, ...ex.history],
   304→              resolvedAt: new Date().toISOString(),
   305→              resolvedBy: currentUser,
   306→            };
   307→          });
   308→
   309→          const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   310→          const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   311→          const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   312→
   313→          const updatedFields = entry.fields.map(f => {
   314→            if (f.exceptionId === exceptionId) {
   315→              const val = status === 'Corrected' ? (newValue || f.value) : f.value;
   316→              return { ...f, value: val, isFlagged: false };
   317→            }
   318→            return f;
   319→          });
   320→
   321→          updatedEntry = {
   322→            ...entry,
   323→            status: newStatus,
   324→            exceptions: updatedExceptions,
   325→            fields: updatedFields,
   326→            currentConfidence: newConfidence,
   327→          };
   328→
   329→          return updatedEntry;
   330→        });
   331→      });
   332→
   333→      // Side effects AFTER setState
   334→      if (previousException && updatedEntry) {
   335→        setUndoStack(prevStack => [
   336→          { entryId, exceptionId, previousState: { ...previousException! } },
   337→          ...prevStack,
   338→        ]);
   339→
   340→        const logText = `Exception "${previousException.fieldName}" for ${entryId} was ${status.toLowerCase()}${
   341→          status === 'Corrected' ? ` to "${newValue}"` : ''
   342→        } by ${currentUser}.`;
   343→        addAuditLog(logText, status === 'Rejected' ? 'warning' : 'success', entryId);
   344→
   345→        if (isSupabaseConfigured() && supabase) {
   346→          supabase.from('exceptions').update({
   347→            status: status as ExceptionStatus,
   348→            corrected_value: status === 'Corrected' ? newValue : null,
   349→            resolved_at: new Date().toISOString(),
   350→            resolved_by: currentUser,
   351→            history: updatedEntry.exceptions.find(e => e.id === exceptionId)?.history,
   352→          }).eq('id', exceptionId).then(({ error }) => {
   353→            if (error) console.warn('[ctx] exception update failed:', error.message);
   354→          });
   355→
   356→          supabase.from('shipments').update({
   357→            status: updatedEntry.status,
   358→            current_confidence: updatedEntry.currentConfidence,
   359→            updated_at: new Date().toISOString(),
   360→          }).eq('id', entryId).then(({ error }) => {
   361→            if (error) console.warn('[ctx] shipment update failed:', error.message);
   362→          });
   363→
   364→          const fieldEntry = updatedEntry.fields.find(f => f.exceptionId === exceptionId);
   365→          if (fieldEntry) {
   366→            supabase.from('document_fields').update({
   367→              is_flagged: false,
   368→              reviewer_action: status,
   369→              corrected_value: status === 'Corrected' ? newValue : null,
   370→              updated_at: new Date().toISOString(),
   371→            }).eq('id', fieldEntry.id).then(({ error }) => {
   372→              if (error) console.warn('[ctx] field update failed:', error.message);
   373→            });
   374→          }
   375→        }
   376→      }
   377→    },
   378→    [currentUser, addAuditLog]
   379→  );
   380→
   381→  // --- Undo ---
   382→  const undoLastAction = React.useCallback(() => {
   383→    if (undoStack.length === 0) return;
   384→
   385→    const [lastAction, ...remainingStack] = undoStack;
   386→    setUndoStack(remainingStack);
   387→
   388→    setEntries(prevEntries => {
   389→      return prevEntries.map(entry => {
   390→        if (entry.id !== lastAction.entryId) return entry;
   391→
   392→        const restored = lastAction.previousState;
   393→        const updatedExceptions = entry.exceptions.map(ex =>
   394→          ex.id === lastAction.exceptionId ? restored : ex
   395→        );
   396→
   397→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   398→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   399→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   400→
   401→        const updatedFields = entry.fields.map(f => {
   402→          if (f.exceptionId === lastAction.exceptionId) {
   403→            return {
   404→              ...f,
   405→              value: restored.status === 'Corrected' ? (restored.correctedValue || f.value) : restored.extractedValue,
   406→              isFlagged: restored.status === 'Unresolved',
   407→            };
   408→          }
   409→          return f;
   410→        });
   411→
   412→        const updatedEntry = {
   413→          ...entry,
   414→          status: newStatus,
   415→          exceptions: updatedExceptions,
   416→          fields: updatedFields,
   417→          currentConfidence: newConfidence,
   418→        };
   419→
   420→        if (isSupabaseConfigured() && supabase) {
   421→          supabase.from('exceptions').update({
   422→            status: restored.status,
   423→            corrected_value: restored.correctedValue || null,
   424→            resolved_at: restored.resolvedAt || null,
   425→            resolved_by: restored.resolvedBy || null,
   426→            history: restored.history,
   427→          }).eq('id', lastAction.exceptionId).then(({ error }) => {
   428→            if (error) console.warn('[ctx] undo exception failed:', error.message);
   429→          });
   430→
   431→          supabase.from('shipments').update({
   432→            status: newStatus,
   433→            current_confidence: newConfidence,
   434→            updated_at: new Date().toISOString(),
   435→          }).eq('id', entry.id).then(({ error }) => {
   436→            if (error) console.warn('[ctx] undo shipment failed:', error.message);
   437→          });
   438→        }
   439→
   440→        return updatedEntry;
   441→      });
   442→    });
   443→
   444→    addAuditLog(`Undo applied for last review action on shipment ${lastAction.entryId}`, 'info', lastAction.entryId);
   445→  }, [undoStack, addAuditLog]);
   446→
   447→  // --- Batch accept (uses configured threshold) ---
   448→  const acceptAllHighConfidence = React.useCallback((entryId: string) => {
   449→    setEntries(prevEntries => {
   450→      return prevEntries.map(entry => {
   451→        if (entry.id !== entryId) return entry;
   452→
   453→        const cutoff = rules.invoiceThreshold;
   454→        let acceptedCount = 0;
   455→
   456→        const updatedExceptions = entry.exceptions.map(ex => {
   457→          if (ex.status === 'Unresolved' && ex.confidence >= cutoff) {
   458→            acceptedCount++;
   459→            const historyItem: ExceptionHistoryEntry = {
   460→              user: currentUser,
   461→              oldValue: ex.extractedValue,
   462→              newValue: ex.extractedValue,
   463→              timestamp: new Date().toISOString(),
   464→              action: 'Accepted' as ReviewerAction,
   465→            };
   466→            return {
   467→              ...ex,
   468→              status: 'Accepted' as ExceptionStatus,
   469→              history: [historyItem, ...ex.history],
   470→              resolvedAt: new Date().toISOString(),
   471→              resolvedBy: currentUser,
   472→            };
   473→          }
   474→          return ex;
   475→        });
   476→
   477→        if (acceptedCount === 0) return entry;
   478→
   479→        const allResolved = updatedExceptions.every(ex => ex.status !== 'Unresolved');
   480→        const newStatus: ShipmentStatus = allResolved ? 'Approved' : 'Under Review';
   481→        const newConfidence = calculateConfidence(entry.initialConfidence, updatedExceptions);
   482→
   483→        const updatedFields = entry.fields.map(f => {
   484→          const exc = updatedExceptions.find(e => e.id === f.exceptionId);
   485→          if (exc) return { ...f, isFlagged: exc.status === 'Unresolved' };
   486→          return f;
   487→        });
   488→
   489→        const updatedEntry = {
   490→          ...entry,
   491→          status: newStatus,
   492→          exceptions: updatedExceptions,
   493→          fields: updatedFields,
   494→          currentConfidence: newConfidence,
   495→        };
   496→
   497→        addAuditLog(
   498→          `Batch action: Approved ${acceptedCount} high-confidence exceptions (≥${cutoff}%) in ${entryId}.`,
   499→          'success',
   500→          entryId
   501→        );
   502→
   503→        if (isSupabaseConfigured() && supabase) {
   504→          updatedExceptions.forEach(ex => {
   505→            if (ex.status === 'Accepted' && ex.resolvedAt) {
   506→              supabase.from('exceptions').update({
   507→                status: 'Accepted',
   508→                resolved_at: ex.resolvedAt,
   509→                resolved_by: currentUser,
   510→                history: ex.history,
   511→              }).eq('id', ex.id).then(({ error }) => {
   512→                if (error) console.warn('[ctx] batch accept failed:', error.message);
   513→              });
   514→            }
   515→          });
   516→
   517→          supabase.from('shipments').update({
   518→            status: newStatus,
   519→            current_confidence: newConfidence,
   520→            updated_at: new Date().toISOString(),
   521→          }).eq('id', entryId).then(({ error }) => {
   522→            if (error) console.warn('[ctx] batch shipment update failed:', error.message);
   523→          });
   524→        }
   525→
   526→        return updatedEntry;
   527→      });
   528→    });
   529→  }, [rules.invoiceThreshold, currentUser, addAuditLog]);
   530→
   531→  // --- Upload documents ---
   532→  const uploadDocuments = React.useCallback(async (files: File[]): Promise<{ shipmentId: string; success: boolean; error?: string }> => {
   533→    const shipmentId = `SHIP-${new Date().getFullYear()}-${Math.floor(1000 + Math.random() * 9000)}`;
   534→
   535→    try {
   536→      if (!isSupabaseConfigured()) {
   537→        // Fallback
   538→        const newEntry: ShipmentEntry = {
   539→          id: shipmentId,
   540→          shipper: 'Titanium Castings KK',
   541→          consignee: 'Ironclad Logistics Inc.',
   542→          status: 'Under Review',
   543→          docsCount: files.length,
   544→          urgency: '08:30:00',
   545→          initialConfidence: 70,
   546→          currentConfidence: 70,
   547→          createdAt: new Date().toISOString(),
   548→          documents: [],
   549→          exceptions: [],
   550→          fields: [],
   551→        };
   552→
   553→        const excId = typeof crypto !== 'undefined' ? crypto.randomUUID() : `${shipmentId}-address`;
   554→        newEntry.exceptions = [{
   555→          id: excId,
   556→          fieldName: 'Consignee Corporate Address',
   557→          fieldKey: 'consigneeAddress',
   558→          originalValue: 'Suite 200, Seattle, WA',
   559→          extractedValue: 'Suite 200, Seattle, WA',
   560→          crossDocValue: 'Suite 201, Seattle, WA',
   561→          confidence: 62,
   562→          reason: 'Address abbreviation mismatch: Commercial Invoice lists "Suite 200, Seattle, WA" while Bill of Lading shows "Suite 201, Seattle, WA".',
   563→          exceptionType: 'cross_doc_mismatch',
   564→          docType: 'Bill of Lading',
   565→          boundingBox: { x: 10, y: 35, w: 32, h: 5 },
   566→          status: 'Unresolved',
   567→          history: [],
   568→          createdAt: new Date().toISOString(),
   569→        }];
   570→
   571→        newEntry.fields = [
   572→          { id: 'n1', key: 'invoiceNo', label: 'Commercial Invoice #', value: 'TC-9921-KK', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 90 },
   573→          { id: 'n2', key: 'invoiceDate', label: 'Invoice Date', value: new Date().toISOString().split('T')[0], sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 88 },
   574→          { id: 'n3', key: 'shipper', label: 'Shipper / Exporter', value: 'Titanium Castings KK', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 95 },
   575→          { id: 'n4', key: 'consignee', label: 'Consignee / Importer', value: 'Ironclad Logistics Inc.', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 94 },
   576→          { id: 'n5', key: 'consigneeAddress', label: 'Consignee Address', value: 'Suite 200, Seattle, WA', sourceDoc: 'Commercial Invoice', isFlagged: true, confidence: 62, exceptionId: excId, crossDocValue: 'Suite 201, Seattle, WA' },
   577→          { id: 'n6', key: 'declaredValue', label: 'Total Declared Value', value: '$45,210.00', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 87 },
   578→          { id: 'n7', key: 'htsCode', label: 'HTS Code (Primary Line)', value: '7308.90.0000', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 91 },
   579→          { id: 'n8', key: 'netWeight', label: 'Total Net Weight', value: '8,410 lbs', sourceDoc: 'Bill of Lading', isFlagged: false, confidence: 89 },
   580→        ];
   581→
   582→        setEntries(prev => [newEntry, ...prev]);
   583→        setSelectedEntryId(shipmentId);
   584→        setSelectedExceptionId(excId);
   585→        addAuditLog(`New entry ${shipmentId} ingestion and auto-analysis completed with 1 exception.`, 'info', shipmentId);
   586→
   587→        return { shipmentId, success: true };
   588→      }
   589→
   590→      // Real pipeline
   591→      await ensureAuthenticated();
   592→
   593→      // Upload each file
   594→      for (const file of files) {
   595→        const formData = new FormData();
   596→        formData.append('file', file);
   597→        formData.append('shipment_id', shipmentId);
   598→
   599→        try {
   600→          if (supabase) {
   601→            const { error } = await supabase.functions.invoke('upload-document', {
   602→              body: formData,
   603→            });
   604→            if (error) throw error;
   605→          }
   606→        } catch (err) {
   607→          console.warn('[upload] edge function failed, continuing:', err);
   608→        }
   609→      }
   610→
   611→      // Extract
   612→      let extractedFields: ExtractedField[] = [];
   613→      let shipper = 'Titanium Castings KK';
   614→      let consignee = 'Ironclad Logistics Inc.';
   615→
   616→      try {
   617→        const extractResponse = await invokeEdgeFunction<any>('extract-document', { shipmentId });
   618→        if (extractResponse?.success && extractResponse.fields) {
   619→          extractedFields = extractResponse.fields.map((f: any) => ({
   620→            id: typeof crypto !== 'undefined' ? crypto.randomUUID() : `f-${Date.now()}`,
   621→            key: f.field_key,
   622→            label: f.field_label,
   623→            value: f.extracted_value,
   624→            sourceDoc: extractResponse.docType || 'Commercial Invoice',
   625→            isFlagged: false,
   626→            confidence: f.confidence,
   627→            boundingBox: f.bounding_box,
   628→          }));
   629→          if (extractResponse.shipper) shipper = extractResponse.shipper;
   630→          if (extractResponse.consignee) consignee = extractResponse.consignee;
   631→        }
   632→      } catch (err) {
   633→        console.warn('[extract] edge function failed, using fallback:', err);
   634→      }
   635→
   636→      if (extractedFields.length === 0) {
   637→        extractedFields = [
   638→          { id: crypto.randomUUID(), key: 'invoiceNo', label: 'Commercial Invoice #', value: 'TC-9921-KK', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 90 },
   639→          { id: crypto.randomUUID(), key: 'shipper', label: 'Shipper / Exporter', value: shipper, sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 95 },
   640→          { id: crypto.randomUUID(), key: 'consignee', label: 'Consignee / Importer', value: consignee, sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 94 },
   641→          { id: crypto.randomUUID(), key: 'consigneeAddress', label: 'Consignee Address', value: 'Suite 200, Seattle, WA', sourceDoc: 'Commercial Invoice', isFlagged: true, confidence: 62, crossDocValue: 'Suite 201, Seattle, WA' },
   642→          { id: crypto.randomUUID(), key: 'declaredValue', label: 'Total Declared Value', value: '$45,210.00', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 87 },
   643→          { id: crypto.randomUUID(), key: 'htsCode', label: 'HTS Code (Primary Line)', value: '7308.90.0000', sourceDoc: 'Commercial Invoice', isFlagged: false, confidence: 91 },
   644→          { id: crypto.randomUUID(), key: 'netWeight', label: 'Total Net Weight', value: '8,410 lbs', sourceDoc: 'Bill of Lading', isFlagged: false, confidence: 89 },
   645→        ];
   646→      }
   647→
   648→      // Flag exceptions based on thresholds
   649→      const newExceptions: Exception[] = [];
   650→      const updatedFields = extractedFields.map(f => {
   651→        let threshold = rules.invoiceThreshold;
   652→        if (f.key === 'htsCode') threshold = rules.htsThreshold;
   653→        else if (f.key === 'shipper' || f.key === 'consignee' || f.key === 'consigneeAddress') threshold = rules.partiesThreshold;
   654→
   655→        if (f.confidence < threshold || f.crossDocValue) {
   656→          const excId = crypto.randomUUID();
   657→          const exc: Exception = {
   658→            id: excId,
   659→            fieldName: f.label,
   660→            fieldKey: f.key,
   661→            originalValue: f.value,
   662→            extractedValue: f.value,
   663→            crossDocValue: f.crossDocValue,
   664→            confidence: f.confidence,
   665→            reason: f.crossDocValue
   666→              ? `Cross-document mismatch detected: extracted "${f.value}" but cross-reference shows "${f.crossDocValue}".`
   667→              : `Extracted confidence (${f.confidence}%) is below threshold (${threshold}%).`,
   668→            exceptionType: f.crossDocValue ? 'cross_doc_mismatch' : 'low_confidence',
   669→            docType: f.sourceDoc,
   670→            boundingBox: f.boundingBox || { x: 10, y: 35, w: 32, h: 5 },
   671→            status: 'Unresolved',
   672→            history: [],
   673→            createdAt: new Date().toISOString(),
   674→          };
   675→          newExceptions.push(exc);
   676→          return { ...f, isFlagged: true, exceptionId: excId };
   677→        }
   678→        return f;
   679→      });
   680→
   681→      const initialConfidence = Math.round(
   682→        updatedFields.reduce((acc, f) => acc + f.confidence, 0) / Math.max(updatedFields.length, 1)
   683→      );
   684→
   685→      const newEntry: ShipmentEntry = {
   686→        id: shipmentId,
   687→        shipper,
   688→        consignee,
   689→        status: 'Under Review',
   690→        docsCount: files.length,
   691→        urgency: '08:30:00',
   692→        initialConfidence,
   693→        currentConfidence: initialConfidence,
   694→        createdAt: new Date().toISOString(),
   695→        documents: [],
   696→        exceptions: newExceptions,
   697→        fields: updatedFields,
   698→      };
   699→
   700→      setEntries(prev => [newEntry, ...prev]);
   701→      setSelectedEntryId(shipmentId);
   702→      setSelectedExceptionId(newExceptions[0]?.id || '');
   703→
   704→      addAuditLog(
   705→        `New entry ${shipmentId} ingestion completed: ${files.length} files, ${newExceptions.length} exceptions flagged.`,
   706→        newExceptions.length > 0 ? 'warning' : 'success',
   707→        shipmentId
   708→      );
   709→
   710→      return { shipmentId, success: true };
   711→    } catch (err) {
   712→      const errMsg = err instanceof Error ? err.message : 'Upload pipeline failed';
   713→      addAuditLog(`Upload failed for ${shipmentId}: ${errMsg}`, 'error', shipmentId);
   714→      return { shipmentId, success: false, error: errMsg };
   715→    }
   716→  }, [rules, addAuditLog]);
   717→
   718→  // --- Update rules ---
   719→  const updateRules = React.useCallback((newRules: Partial<OperationalRules>) => {
   720→    setRules(prev => {
   721→      const updated = { ...prev, ...newRules };
   722→
   723→      if (isSupabaseConfigured() && supabase) {
   724→        supabase.from('operational_rules').upsert({
   725→          id: 'default_config',
   726→          invoice_threshold: updated.invoiceThreshold,
   727→          hts_threshold: updated.htsThreshold,
   728→          parties_threshold: updated.partiesThreshold,
   729→          updated_at: new Date().toISOString(),
   730→        }).then(({ error }) => {
   731→          if (error) console.warn('[ctx] rules upsert failed:', error.message);
   732→        });
   733→      }
   734→
   735→      return updated;
   736→    });
   737→    addAuditLog('Compliance operational thresholds updated.', 'info');
   738→  }, [addAuditLog]);
   739→
   740→  // --- Real CSV export ---
   741→  const exportToCSV = React.useCallback(async (entryId: string) => {
   742→    const entry = entries.find(e => e.id === entryId);
   743→    if (!entry) return;
   744→
   745→    if (isSupabaseConfigured()) {
   746→      try {
   747→        const response = await invokeEdgeFunction<any>('export-csv', { shipmentId: entryId });
   748→        if (response?.csv) {
   749→          const blob = new Blob([response.csv], { type: 'text/csv' });
   750→          const url = URL.createObjectURL(blob);
   751→          const a = document.createElement('a');
   752→          a.href = url;
   753→          a.download = `ClearPort_Audit_${entryId}.csv`;
   754→          document.body.appendChild(a);
   755→          a.click();
   756→          document.body.removeChild(a);
   757→          URL.revokeObjectURL(url);
   758→
   759→          addAuditLog(`Audit logs exported to CSV for ${entryId}.`, 'success', entryId);
   760→          return;
   761→        }
   762→      } catch (err) {
   763→        console.warn('[csv] edge function failed, generating locally:', err);
   764→      }
   765→    }
   766→
   767→    // Fallback: local CSV generation
   768→    const rows: string[] = [];
   769→    rows.push('Field Key,Field Label,Value,Source Document,Confidence,Flagged,Status');
   770→    entry.fields.forEach(f => {
   771→      const exc = entry.exceptions.find(e => e.id === f.exceptionId);
   772→      const status = exc ? exc.status : 'SECURE';
   773→      const escaped = [f.key, f.label, f.value, f.sourceDoc, f.confidence, f.isFlagged, status]
   774→        .map(v => `"${String(v).replace(/"/g, '""')}"`)
   775→        .join(',');
   776→      rows.push(escaped);
   777→    });
   778→
   779→    rows.push('');
   780→    rows.push('Exception ID,Field Name,Reason,Confidence,Status,Resolved By');
   781→    entry.exceptions.forEach(e => {
   782→      const escaped = [e.id, e.fieldName, e.reason, e.confidence, e.status, e.resolvedBy || '']
   783→        .map(v => `"${String(v).replace(/"/g, '""')}"`)
   784→        .join(',');
   785→      rows.push(escaped);
   786→    });
   787→
   788→    const csv = rows.join('\n');
   789→    const blob = new Blob([csv], { type: 'text/csv' });
   790→    const url = URL.createObjectURL(blob);
   791→    const a = document.createElement('a');
   792→    a.href = url;
   793→    a.download = `ClearPort_Audit_${entryId}.csv`;
   794→    document.body.appendChild(a);
   795→    a.click();
   796→    document.body.removeChild(a);
   797→    URL.revokeObjectURL(url);
   798→
   799→    addAuditLog(`Audit logs exported to CSV for ${entryId}.`, 'success', entryId);
   800→  }, [entries, addAuditLog]);
   801→
   802→  const value: ClearPortContextType = {
   803→    entries,
   804→    selectedEntryId,
   805→    selectedEntry,
   806→    selectedExceptionId,
   807→    selectedException,
   808→    activeTab,
   809→    rules,
   810→    undoStack,
   811→    auditLogs,
   812→    theme,
   813→    supabaseStatus,
   814→    edgeFunctionStatus,
   815→    currentUser,
   816→    currentTime,
   817→    selectEntry,
   818→    selectException,
   819→    setActiveTab,
   820→    updateException,
   821→    undoLastAction,
   822→    acceptAllHighConfidence,
   823→    uploadDocuments,
   824→    updateRules,
   825→    exportToCSV,
   826→    toggleTheme,
   827→    refreshData,
   828→  };
   829→
   830→  return (
   831→    <ClearPortContext.Provider value={value}>
   832→      {children}
   833→    </ClearPortContext.Provider>
   834→  );
   835→};
   836→
   837→export function useClearPort() {
   838→  const context = React.useContext(ClearPortContext);
   839→  if (context === undefined) {
   840→    throw new Error('useClearPort must be used within a ClearPortProvider');
   841→  }

   843→}
